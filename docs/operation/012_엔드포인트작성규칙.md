## 3. 엔드포인트 구현 템플릿

### 3.1 GET 목록 조회 (`route.ts`)

```typescript
// Generated: YYYY-MM-DD HH:MM:SS KST

import { NextRequest, NextResponse } from 'next/server';
import { getServerSession } from 'next-auth';
import { authOptions } from '@/lib/auth';
import { getDataSource } from '@/lib/db';

export const dynamic = 'force-dynamic';

export async function GET(req: NextRequest): Promise<NextResponse> {
  try {
    // 1. 인증 검증
    const session = await getServerSession(authOptions);
    if (!session?.user) {
      return NextResponse.json(
        { message: 'Unauthorized' },
        { status: 401 }
      );
    }

    const user = session.user as any;
    const userId = user.id;
    const userRole = user.role;
    const userDepartmentId = user.department;

    // 2. 쿼리 파라미터 파싱
    const searchParams = req.nextUrl.searchParams;
    const page = Math.max(1, parseInt(searchParams.get('page') || '1', 10));
    const page_size = Math.min(100, Math.max(1, parseInt(searchParams.get('page_size') || '20', 10)));

    // 3. 권한별 WHERE 절 (RLS - Row Level Security)
    const whereClauses: string[] = ['"deleted_at" IS NULL'];
    const params: any = {};

    if (userRole === 'ADMIN') {
      // 모든 데이터 접근
    } else if (userRole === 'MANAGER') {
      // 부서별 접근 제한
      whereClauses.push(`"department_id" = :departmentId`);
      params.departmentId = userDepartmentId;
    } else if (userRole === 'USER') {
      // 자신이 생성한 데이터만
      whereClauses.push(`"created_by_id" = :userId`);
      params.userId = userId;
    }

    // 4. 데이터베이스 쿼리
    const ds = await getDataSource();
    const queryRunner = ds.createQueryRunner();

    try {
      const offset = (page - 1) * page_size;

      const query = `
        SELECT * FROM [TABLE_NAME]
        WHERE ${whereClauses.join(' AND ')}
        ORDER BY "created_at" DESC
        OFFSET :offset ROWS FETCH NEXT :pageSize ROWS ONLY
      `;

      const countQuery = `
        SELECT COUNT(*) as total FROM [TABLE_NAME]
        WHERE ${whereClauses.join(' AND ')}
      `;

      const [items, countResult] = await Promise.all([
        queryRunner.query(query, { ...params, offset, pageSize: page_size }),
        queryRunner.query(countQuery, params),
      ]);

      const total = parseInt(countResult[0]?.total || '0', 10);

      // 5. 응답 반환
      return NextResponse.json({
        data: items,
        pagination: {
          page,
          page_size,
          total,
          total_pages: Math.ceil(total / page_size),
        },
      });
    } finally {
      await queryRunner.release();
    }
  } catch (error) {
    console.error('GET /api/[module] error:', error);
    return NextResponse.json(
      { message: 'Internal server error' },
      { status: 500 }
    );
  }
}
```

### 3.2 POST 생성 (`route.ts`)

```typescript
export async function POST(req: NextRequest): Promise<NextResponse> {
  try {
    // 1. 인증 검증
    const session = await getServerSession(authOptions);
    if (!session?.user) {
      return NextResponse.json({ message: 'Unauthorized' }, { status: 401 });
    }

    const user = session.user as any;
    const userId = user.id;

    // 2. 요청 본문 파싱
    const body = await req.json();

    // 3. 필드 검증
    if (!body.title || typeof body.title !== 'string') {
      return NextResponse.json(
        { message: 'Validation failed', errors: { title: 'Required' } },
        { status: 400 }
      );
    }

    // 4. 데이터 삽입
    const ds = await getDataSource();
    const queryRunner = ds.createQueryRunner();

    try {
      const now = new Date();
      const result = await queryRunner.query(
        `INSERT INTO [TABLE_NAME] ("title", "created_by_id", "created_at", "updated_at", "deleted_at")
         VALUES (:title, :createdById, :createdAt, :updatedAt, :deletedAt)
         RETURNING "id", "created_at"`,
        {
          title: body.title,
          createdById: userId,
          createdAt: now,
          updatedAt: now,
          deletedAt: null,
        }
      );

      return NextResponse.json(
        { message: 'Created successfully', data: result[0] },
        { status: 201 }
      );
    } finally {
      await queryRunner.release();
    }
  } catch (error) {
    console.error('POST /api/[module] error:', error);
    return NextResponse.json(
      { message: 'Internal server error' },
      { status: 500 }
    );
  }
}
```

## 4. 열 식별자 (Column Identifier) 규칙

### 4.1 quoted identifier 사용

모든 열 이름은 **큰따옴표**로 감싸기 (Oracle 대소문자 구분)

```typescript
// ✓ 올바른 사용
SELECT "id", "username", "created_at" FROM EMPLOYEE

// ❌ 잘못된 사용
SELECT id, username, created_at FROM EMPLOYEE  // ORA-00904 에러
```

### 4.2 테이블별 열 이름 규칙

| 테이블 | 대소문자 | 예시 |
|--------|---------|------|
| EMPLOYEE | 소문자 | `"id"`, `"username"`, `"password_hash"`, `"department_id"` |
| CUSTOMER | 소문자 | `"id"`, `"name"`, `"deleted_at"` |
| ISSUE | 대문자 | `"ID"`, `"CUSTOMER_ID"`, `"CREATED_BY_ID"`, `"DELETED_AT"` |

### 4.3 JOIN 작성 예시

```typescript
// ISSUE (대문자) ← → CUSTOMER (소문자)
const query = `
  SELECT
    "i"."ID",
    "i"."CUSTOMER_ID",
    "c"."id",
    "c"."name"
  FROM ISSUE "i"
  LEFT JOIN CUSTOMER "c" ON "c"."id" = "i"."CUSTOMER_ID" AND "c"."deleted_at" IS NULL
`;
```

## 6. 권한 검증 (RBAC)

### 6.1 기본 역할 정의

| 역할 | 권한 |
|------|------|
| ADMIN | 모든 기능 + 사용자 관리 |
| MANAGER | 부서 범위 읽기/쓰기 |
| USER | 자신의 데이터 + 공개 데이터 읽기 |

### 6.2 권한 검증 코드

```typescript
// 역할별 접근 제어
if (userRole === 'ADMIN') {
  // 모든 데이터
} else if (userRole === 'MANAGER') {
  // 부서별 필터
  whereClauses.push(`"department_id" = :dept`);
} else {
  // 개인 데이터만
  whereClauses.push(`"created_by_id" = :userId`);
}

// 역할별 작업 제어
const allowedRoles = ['ADMIN', 'MANAGER'];
if (!allowedRoles.includes(userRole)) {
  return NextResponse.json({ message: 'Forbidden' }, { status: 403 });
}
```
