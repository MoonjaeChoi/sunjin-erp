```typescript
'use client';

import { Suspense } from 'react';
import { useMaintenanceContracts } from '@/hooks/useMaintenanceContracts';
import { useMaintenanceFilter } from '@/hooks/useMaintenanceFilter';
import { useMaintenanceStats } from '@/hooks/useMaintenanceStats';
import MaintenanceFilters from './MaintenanceFilters';
import MaintenanceDataTable from './MaintenanceDataTable';
import MaintenanceStats from './MaintenanceStats';
import MaintenanceTableSkeleton from './MaintenanceTableSkeleton';
import { Button } from '@/components/ui/button';
import { Plus } from 'lucide-react';
import Link from 'next/link';

/**
 * 유지보수 계약 목록 페이지
 */
export default function MaintenanceList() {
  const { filters, updateFilters, resetFilters } = useMaintenanceFilter();
  const { data, isLoading, error } = useMaintenanceContracts(filters);
  const { data: statsData } = useMaintenanceStats();

  return (
    <div className="space-y-6">
      {/* 헤더 */}
      <div className="flex items-center justify-between">
        <div>
          <h1 className="text-3xl font-bold">유지보수 계약 관리</h1>
          <p className="text-muted-foreground">
            {data?.pagination.total || 0}건의 계약
          </p>
        </div>
        <Link href="/maintenance/new">
          <Button>
            <Plus className="w-4 h-4 mr-2" />
            신규 등록
          </Button>
        </Link>
      </div>

      {/* 통계 카드 */}
      <Suspense fallback={<div className="h-20 bg-gray-200 rounded" />}>
        <MaintenanceStats />
      </Suspense>

      {/* 필터 */}
      <MaintenanceFilters
        filters={filters}
        onFilterChange={updateFilters}
        onReset={resetFilters}
      />

      {/* 테이블 */}
      {isLoading ? (
        <MaintenanceTableSkeleton />
      ) : error ? (
        <div className="text-center py-8 text-red-500">
          목록 조회에 실패했습니다
        </div>
      ) : (
        <MaintenanceDataTable
          data={data?.data || []}
          pagination={data?.pagination}
          currentFilters={filters}
          onFilterChange={updateFilters}
        />
      )}
    </div>
  );
}
```

### 2. MaintenanceFilters.tsx - 필터 도구 모음 (CC)

```typescript
// Generated: 2026-01-26 HH:MM:SS KST

'use client';

import { useCallback } from 'react';
import { MaintenanceListFilters, ContractStatus, ContractStatusLabel } from '@/types/maintenance';
import {
  Select,
  SelectContent,
  SelectItem,
  SelectTrigger,
  SelectValue
} from '@/components/ui/select';
import { Input } from '@/components/ui/input';
import { Button } from '@/components/ui/button';
import { X } from 'lucide-react';

interface MaintenanceFiltersProps {
  filters: MaintenanceListFilters;
  onFilterChange: (filters: Partial<MaintenanceListFilters>) => void;
  onReset: () => void;
}

/**
 * 계약 목록 필터 도구 모음
 */
export default function MaintenanceFilters({
  filters,
  onFilterChange,
  onReset
}: MaintenanceFiltersProps) {
  const handleStatusChange = useCallback(
    (status: string) => {
      onFilterChange({
        status: status === 'all' ? undefined : (status as ContractStatus),
        page: 1
      });
    },
    [onFilterChange]
  );

  const handleSearchChange = useCallback(
    (e: React.ChangeEvent<HTMLInputElement>) => {
      onFilterChange({
        contractNameSearch: e.target.value,
        page: 1
      });
    },
    [onFilterChange]
  );

  const handleReset = useCallback(() => {
    onReset();
  }, [onReset]);

  const hasActiveFilters = !!(
    filters.status ||
    filters.customerId ||
    filters.contractNameSearch
  );

  return (
    <div className="rounded-lg border p-4 space-y-4">
      {/* 검색 */}
      <div>
        <label className="text-sm font-medium mb-2 block">계약명</label>
        <Input
          placeholder="계약명으로 검색..."
          value={filters.contractNameSearch || ''}
          onChange={handleSearchChange}
        />
      </div>

      {/* 필터 행 */}
      <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
        {/* 상태 필터 */}
        <div>
          <label className="text-sm font-medium mb-2 block">상태</label>
          <Select value={filters.status || 'all'} onValueChange={handleStatusChange}>
            <SelectTrigger>
              <SelectValue placeholder="상태 선택" />
            </SelectTrigger>
            <SelectContent>
              <SelectItem value="all">모두</SelectItem>
              <SelectItem value={ContractStatus.ACTIVE}>
                {ContractStatusLabel[ContractStatus.ACTIVE]}
              </SelectItem>
              <SelectItem value={ContractStatus.RENEWAL_PENDING}>
                {ContractStatusLabel[ContractStatus.RENEWAL_PENDING]}
              </SelectItem>
              <SelectItem value={ContractStatus.ENDED}>
                {ContractStatusLabel[ContractStatus.ENDED]}
              </SelectItem>
            </SelectContent>
          </Select>
        </div>

        {/* 고객 필터 - TODO: 나중에 고객 선택 드롭다운으로 변경 */}
        <div>
          <label className="text-sm font-medium mb-2 block">고객</label>
          <Input placeholder="고객명 입력..." disabled />
        </div>

        {/* 담당자 필터 - TODO: 나중에 담당자 선택 드롭다운으로 변경 */}
        <div>
          <label className="text-sm font-medium mb-2 block">담당자</label>
          <Input placeholder="담당자명 입력..." disabled />
        </div>
      </div>

      {/* 버튼 */}
      <div className="flex justify-end gap-2">
        {hasActiveFilters && (
          <Button variant="outline" onClick={handleReset}>
            <X className="w-4 h-4 mr-2" />
            필터 초기화
          </Button>
        )}
      </div>
    </div>
  );
}
```

```typescript
// Generated: 2026-01-26 HH:MM:SS KST

'use client';

import { useCallback } from 'react';
import { MaintenanceContract, MaintenanceListFilters, ContractStatusLabel } from '@/types/maintenance';
import { PaginationInfo } from '@/types/common';
import {
  Table,
  TableBody,
  TableCell,
  TableHead,
  TableHeader,
  TableRow
} from '@/components/ui/table';
import { Badge } from '@/components/ui/badge';
import { Button } from '@/components/ui/button';
import { ChevronLeft, ChevronRight } from 'lucide-react';
import { useContractStatusBadge } from '@/hooks/useContractStatusBadge';
import { useExpiryBadge } from '@/hooks/useContractExpiry';
import Link from 'next/link';

interface MaintenanceDataTableProps {
  data: MaintenanceContract[];
  pagination?: PaginationInfo;
  currentFilters: MaintenanceListFilters;
  onFilterChange: (filters: Partial<MaintenanceListFilters>) => void;
}

/**
 * 계약 데이터 테이블
 */
export default function MaintenanceDataTable({
  data,
  pagination,
  currentFilters,
  onFilterChange
}: MaintenanceDataTableProps) {
  const handlePreviousPage = useCallback(() => {
    if (pagination && currentFilters.page && currentFilters.page > 1) {
      onFilterChange({ page: currentFilters.page - 1 });
    }
  }, [pagination, currentFilters.page, onFilterChange]);

  const handleNextPage = useCallback(() => {
    if (pagination?.hasNextPage) {
      onFilterChange({ page: (currentFilters.page || 1) + 1 });
    }
  }, [pagination, currentFilters.page, onFilterChange]);

  if (data.length === 0) {
    return (
      <div className="rounded-lg border p-8 text-center text-muted-foreground">
        조회된 계약이 없습니다
      </div>
    );
  }

  return (
    <div className="space-y-4">
      <div className="rounded-lg border overflow-hidden">
        <Table>
          <TableHeader className="bg-muted">
            <TableRow>
              <TableHead>계약명</TableHead>
              <TableHead>고객</TableHead>
              <TableHead>담당자</TableHead>
              <TableHead>기간</TableHead>
              <TableHead>상태</TableHead>
              <TableHead>만료</TableHead>
              <TableHead className="text-right">작업</TableHead>
            </TableRow>
          </TableHeader>
          <TableBody>
            {data.map((contract) => {
              const statusBadge = useContractStatusBadge(contract.contract_status);
              const expiryInfo = useExpiryBadge(contract.end_date);

              return (
                <TableRow key={contract.id}>
                  <TableCell className="font-medium">
                    {contract.contract_name}
                  </TableCell>
                  <TableCell>{contract.customer.name}</TableCell>
                  <TableCell>{contract.assigned_employee.name}</TableCell>
                  <TableCell className="text-sm text-muted-foreground">
                    {contract.start_date} ~ {contract.end_date}
                  </TableCell>
                  <TableCell>
                    <Badge variant={statusBadge.variant}>
                      {statusBadge.label}
                    </Badge>
                  </TableCell>
                  <TableCell>
                    <Badge
                      variant="outline"
                      className={expiryInfo.badgeColor}
                    >
                      {expiryInfo.displayText}
                    </Badge>
                  </TableCell>
                  <TableCell className="text-right">
                    <Link href={`/maintenance/${contract.id}`}>
                      <Button variant="ghost" size="sm">
                        상세
                      </Button>
                    </Link>
                  </TableCell>
                </TableRow>
              );
            })}
          </TableBody>
        </Table>
      </div>

      {/* 페이지네이션 */}
      {pagination && pagination.total > 0 && (
        <div className="flex items-center justify-between">
          <div className="text-sm text-muted-foreground">
            {(pagination.page - 1) * pagination.limit + 1} ~{' '}
            {Math.min(pagination.page * pagination.limit, pagination.total)} / 총{' '}
            {pagination.total}건
          </div>
          <div className="flex gap-2">
            <Button
              variant="outline"
              size="sm"
              onClick={handlePreviousPage}
              disabled={pagination.page === 1}
            >
              <ChevronLeft className="w-4 h-4" />
              이전
            </Button>
            <div className="flex items-center gap-1">
              <span className="text-sm px-2">
                {pagination.page} / {Math.ceil(pagination.total / pagination.limit)}
              </span>
            </div>
            <Button
              variant="outline"
              size="sm"
              onClick={handleNextPage}
              disabled={!pagination.hasNextPage}
            >
              다음
              <ChevronRight className="w-4 h-4" />
            </Button>
          </div>
        </div>
      )}
    </div>
  );
}
```
