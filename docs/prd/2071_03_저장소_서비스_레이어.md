**파일**: `src/lib/maintenance-repository.ts`

```typescript
import { DataSource, Repository, IsNull } from 'typeorm';
import { MaintenanceContract } from '@/entities/MaintenanceContract';
import { MaintenanceContractAttachment } from '@/entities/MaintenanceContractAttachment';
import { MaintenanceContractHistory } from '@/entities/MaintenanceContractHistory';

export class MaintenanceContractRepository {
  private contractRepo: Repository<MaintenanceContract>;
  private attachmentRepo: Repository<MaintenanceContractAttachment>;
  private historyRepo: Repository<MaintenanceContractHistory>;

  constructor(dataSource: DataSource) {
    this.contractRepo = dataSource.getRepository(MaintenanceContract);
    this.attachmentRepo = dataSource.getRepository(MaintenanceContractAttachment);
    this.historyRepo = dataSource.getRepository(MaintenanceContractHistory);
  }

  // ===== Contract 쿼리 =====

  /**
   * 활성 계약 조회 (soft delete 제외)
   */
  async findActiveContracts(
    filters: {
      status?: string;
      customerId?: number;
      employeeId?: number;
      searchTerm?: string;
      startDateFrom?: Date;
      startDateTo?: Date;
      endDateFrom?: Date;
      endDateTo?: Date;
    } = {},
    pagination: { page: number; limit: number } = { page: 1, limit: 20 },
    sort: { sortBy: string; order: 'ASC' | 'DESC' } = { sortBy: 'end_date', order: 'DESC' }
  ) {
    let query = this.contractRepo
      .createQueryBuilder('mc')
      .leftJoinAndSelect('mc.customer', 'customer')
      .leftJoinAndSelect('mc.assignedEmployee', 'employee')
      .where('mc.deleted_at IS NULL');

    // 필터 적용
    if (filters.status) {
      query = query.andWhere('mc.contract_status = :status', { status: filters.status });
    }

    if (filters.customerId) {
      query = query.andWhere('mc.customer_id = :customerId', { customerId: filters.customerId });
    }

    if (filters.employeeId) {
      query = query.andWhere('mc.assigned_employee_id = :employeeId', { employeeId: filters.employeeId });
    }

    if (filters.searchTerm) {
      query = query.andWhere('LOWER(mc.contract_name) LIKE :search', {
        search: `%${filters.searchTerm.toLowerCase()}%`,
      });
    }

    if (filters.startDateFrom) {
      query = query.andWhere('mc.start_date >= :startDateFrom', { startDateFrom: filters.startDateFrom });
    }

    if (filters.startDateTo) {
      query = query.andWhere('mc.start_date <= :startDateTo', { startDateTo: filters.startDateTo });
    }

    if (filters.endDateFrom) {
      query = query.andWhere('mc.end_date >= :endDateFrom', { endDateFrom: filters.endDateFrom });
    }

    if (filters.endDateTo) {
      query = query.andWhere('mc.end_date <= :endDateTo', { endDateTo: filters.endDateTo });
    }

    // 정렬
    const orderBy: Record<string, 'ASC' | 'DESC'> = {};
    if (sort.sortBy === 'customerName') {
      orderBy['customer.name'] = sort.order;
    } else if (sort.sortBy === 'contractName') {
      orderBy['mc.contract_name'] = sort.order;
    } else {
      orderBy['mc.end_date'] = sort.order;
    }
    query = query.orderBy(orderBy);

    // 페이지네이션
    const offset = (pagination.page - 1) * pagination.limit;
    query = query.skip(offset).take(pagination.limit);

    return query.getManyAndCount();
  }

  /**
   * 계약 상세 조회 (모든 관계)
   */
  async findById(id: number) {
    return this.contractRepo
      .createQueryBuilder('mc')
      .leftJoinAndSelect('mc.customer', 'customer')
      .leftJoinAndSelect('mc.assignedEmployee', 'employee')
      .leftJoinAndSelect('mc.createdBy', 'createdBy')
      .leftJoinAndSelect('mc.updatedBy', 'updatedBy')
      .where('mc.id = :id', { id })
      .andWhere('mc.deleted_at IS NULL')
      .getOne();
  }

  /**
   * 계약 생성
   */
  async create(data: Partial<MaintenanceContract>) {
    const contract = this.contractRepo.create(data);
    return this.contractRepo.save(contract);
  }

  /**
   * 계약 수정
   */
  async update(id: number, data: Partial<MaintenanceContract>) {
    await this.contractRepo.update(
      { id, deleted_at: IsNull() },
      data
    );
    return this.findById(id);
  }

  /**
   * 계약 소프트 삭제 (첨부 + 이력도 함께)
   */
  async softDeleteContract(id: number, deletedAt: Date = new Date()) {
    // 트랜잭션으로 처리해야 함 (service에서)
    await this.contractRepo.update(
      { id, deleted_at: IsNull() },
      { deleted_at: deletedAt }
    );

    // 첨부 파일도 함께 삭제
    await this.attachmentRepo.update(
      { maintenance_contract_id: id, deleted_at: IsNull() },
      { deleted_at: deletedAt }
    );

    // 이력도 함께 삭제
    await this.historyRepo.update(
      { maintenance_contract_id: id, deleted_at: IsNull() },
      { deleted_at: deletedAt }
    );
  }

  // ===== Attachment 쿼리 =====

  /**
   * 계약별 첨부 파일 조회 (페이지네이션)
   */
  async findAttachmentsByContractId(
    contractId: number,
    pagination: { page: number; limit: number } = { page: 1, limit: 10 }
  ) {
    const offset = (pagination.page - 1) * pagination.limit;
    return this.attachmentRepo
      .createQueryBuilder('mca')
      .leftJoinAndSelect('mca.uploadedBy', 'uploadedBy')
      .where('mca.maintenance_contract_id = :contractId', { contractId })
      .andWhere('mca.deleted_at IS NULL')
      .orderBy('mca.created_at', 'DESC')
      .skip(offset)
      .take(pagination.limit)
      .getManyAndCount();
  }

  /**
   * 첨부 파일 추가
   */
  async createAttachment(data: Partial<MaintenanceContractAttachment>) {
    const attachment = this.attachmentRepo.create(data);
    return this.attachmentRepo.save(attachment);
  }

  /**
   * 첨부 파일 삭제 (soft delete)
   */
  async softDeleteAttachment(id: number) {
    await this.attachmentRepo.update(
      { id, deleted_at: IsNull() },
      { deleted_at: new Date() }
    );
  }

  // ===== History 쿼리 =====

  /**
   * 계약 갱신 이력 조회 (페이지네이션)
   */
  async findHistoriesByContractId(
    contractId: number,
    pagination: { page: number; limit: number } = { page: 1, limit: 20 }
  ) {
    const offset = (pagination.page - 1) * pagination.limit;
    return this.historyRepo
      .createQueryBuilder('mch')
      .leftJoinAndSelect('mch.changedBy', 'changedBy')
      .where('mch.maintenance_contract_id = :contractId', { contractId })
      .andWhere('mch.deleted_at IS NULL')
      .orderBy('mch.changed_at', 'DESC')
      .skip(offset)
      .take(pagination.limit)
      .getManyAndCount();
  }

  /**
   * 이력 기록 추가
   */
  async createHistory(data: Partial<MaintenanceContractHistory>) {
    const history = this.historyRepo.create(data);
    return this.historyRepo.save(history);
  }

  // ===== 통계 =====

  /**
   * 상태별 계약 집계
   */
  async getStatsByStatus() {
    const result = await this.contractRepo
      .createQueryBuilder('mc')
      .select('mc.contract_status', 'status')
      .addSelect('COUNT(*)', 'count')
      .where('mc.deleted_at IS NULL')
      .groupBy('mc.contract_status')
      .getRawMany();

    return result;
  }

  /**
   * 30일 내 만료 계약 수
   */
  async countExpiringIn30Days() {
    const thirtyDaysFromNow = new Date();
    thirtyDaysFromNow.setDate(thirtyDaysFromNow.getDate() + 30);

    return this.contractRepo.count({
      where: {
        contract_status: '활성',
        end_date: () => `end_date <= '${thirtyDaysFromNow.toISOString().split('T')[0]}'`,
        deleted_at: IsNull(),
      },
    });
  }
}
```

### 2. Service 클래스

**파일**: `src/lib/maintenance-service.ts`

```typescript
import { DataSource } from 'typeorm';
import { MaintenanceContractRepository } from './maintenance-repository';
import { MaintenanceContract } from '@/entities/MaintenanceContract';
import { MaintenanceContractHistory } from '@/entities/MaintenanceContractHistory';

export class MaintenanceContractService {
  private repository: MaintenanceContractRepository;
  private dataSource: DataSource;

  constructor(dataSource: DataSource) {
    this.dataSource = dataSource;
    this.repository = new MaintenanceContractRepository(dataSource);
  }

  /**
   * 계약 목록 조회 (필터, 정렬, 페이지네이션)
   */
  async getContracts(filters: any = {}, pagination: any = {}, sort: any = {}) {
    return this.repository.findActiveContracts(filters, pagination, sort);
  }

  /**
   * 계약 상세 조회
   */
  async getContract(id: number) {
    const contract = await this.repository.findById(id);
    if (!contract) {
      throw new Error('Contract not found');
    }
    return contract;
  }

  /**
   * 계약 생성 (트랜잭션)
   */
  async createContract(
    data: Partial<MaintenanceContract>,
    userId: number
  ) {
    const queryRunner = this.dataSource.createQueryRunner();
    await queryRunner.startTransaction();

    try {
      // 계약 생성
      const contract = await this.repository.create({
        ...data,
        created_by_id: userId,
        updated_by_id: userId,
        contract_status: '활성',
      });

      // 이력 기록
      await this.repository.createHistory({
        maintenance_contract_id: contract.id,
        change_type: '정보수정',
        reason: '계약 생성',
        changed_by_id: userId,
      });

      await queryRunner.commitTransaction();
      return contract;
    } catch (error) {
      await queryRunner.rollbackTransaction();
      throw error;
    } finally {
      await queryRunner.release();
    }
  }

  /**
   * 계약 수정 (트랜잭션)
   */
  async updateContract(
    id: number,
    data: Partial<MaintenanceContract>,
    userId: number
  ) {
    const queryRunner = this.dataSource.createQueryRunner();
    await queryRunner.startTransaction();

    try {
      // 이전 값 저장 (갱신 감지용)
      const previousContract = await this.repository.findById(id);

      // 계약 수정
      const updated = await this.repository.update(id, {
        ...data,
        updated_by_id: userId,
      });

      // end_date 변경시 갱신 이력 기록
      if (data.end_date && previousContract && previousContract.end_date !== data.end_date) {
        await this.repository.createHistory({
          maintenance_contract_id: id,
          change_type: '갱신',
          previous_end_date: previousContract.end_date,
          new_end_date: data.end_date,
          changed_by_id: userId,
        });
      }

      // 상태 변경시 이력 기록
      if (data.contract_status && previousContract && previousContract.contract_status !== data.contract_status) {
        await this.repository.createHistory({
          maintenance_contract_id: id,
          change_type: '상태변경',
          reason: data.contract_status,
          changed_by_id: userId,
        });
      }

      await queryRunner.commitTransaction();
      return updated;
    } catch (error) {
      await queryRunner.rollbackTransaction();
      throw error;
    } finally {
      await queryRunner.release();
    }
  }

  /**
   * 계약 삭제 (cascade soft delete - 계약+첨부+이력)
   */
  async deleteContract(id: number) {
    const queryRunner = this.dataSource.createQueryRunner();
    await queryRunner.startTransaction();

    try {
      await this.repository.softDeleteContract(id);
      await queryRunner.commitTransaction();
    } catch (error) {
      await queryRunner.rollbackTransaction();
      throw error;
    } finally {
      await queryRunner.release();
    }
  }

  /**
   * 첨부 파일 추가
   */
  async addAttachment(contractId: number, fileData: any, userId: number) {
    return this.repository.createAttachment({
      maintenance_contract_id: contractId,
      file_name: fileData.originalName,
      file_path: fileData.path,
      file_size: fileData.size,
      uploaded_by_id: userId,
    });
  }

  /**
   * 첨부 파일 삭제 (ADMIN만)
   */
  async deleteAttachment(id: number) {
    return this.repository.softDeleteAttachment(id);
  }

  /**
   * 갱신 이력 조회
   */
  async getContractHistory(contractId: number, pagination?: any) {
    return this.repository.findHistoriesByContractId(contractId, pagination);
  }

  /**
   * 계약 통계
   */
  async getStats() {
    const byStatus = await this.repository.getStatsByStatus();
    const expiringIn30Days = await this.repository.countExpiringIn30Days();

    return {
      byStatus,
      expiringIn30Days,
    };
  }
}
```
