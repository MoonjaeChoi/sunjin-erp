-- ============================================================
-- 2091 직원 관리 테이블 DDL
-- Database: Oracle XE 21c (XEPDB1)
-- Schema: sunjin_admin
-- Generated: 2026-01-28
-- ============================================================

-- ============================================================
-- 1. SEQUENCES
-- ============================================================

-- 직급 코드 자동 생성용 시퀀스 (Decision #10)
CREATE SEQUENCE POSITION_CODE_SEQ
  START WITH 1
  INCREMENT BY 1
  NOCACHE
  NOCYCLE;

-- ============================================================
-- 2. DEPARTMENT (부서) 테이블
-- ============================================================

CREATE TABLE DEPARTMENT (
  "id"                    NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  "name"                  VARCHAR2(100) NOT NULL,
  "code"                  VARCHAR2(20) NOT NULL,
  "description"           VARCHAR2(500),
  PARENT_DEPARTMENT_ID    NUMBER,
  "created_at"            TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
  "updated_at"            TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
  "deleted_at"            TIMESTAMP,

  -- 자기 참조 FK (ON DELETE RESTRICT)
  CONSTRAINT FK_DEPARTMENT_PARENT
    FOREIGN KEY (PARENT_DEPARTMENT_ID)
    REFERENCES DEPARTMENT("id")
);

-- 부서명 유니크 (soft delete 제외)
CREATE UNIQUE INDEX UQ_DEPARTMENT_NAME ON DEPARTMENT("name")
  WHERE "deleted_at" IS NULL;

-- 부서 코드 유니크 (soft delete 제외)
CREATE UNIQUE INDEX UQ_DEPARTMENT_CODE ON DEPARTMENT("code")
  WHERE "deleted_at" IS NULL;

-- 인덱스
CREATE INDEX IDX_DEPARTMENT_PARENT ON DEPARTMENT(PARENT_DEPARTMENT_ID);
CREATE INDEX IDX_DEPARTMENT_DELETED ON DEPARTMENT("deleted_at");

COMMENT ON TABLE DEPARTMENT IS '부서 정보';
COMMENT ON COLUMN DEPARTMENT."id" IS '부서 ID (PK)';
COMMENT ON COLUMN DEPARTMENT."name" IS '부서명';
COMMENT ON COLUMN DEPARTMENT."code" IS '부서 코드 (예: IT, SALES)';
COMMENT ON COLUMN DEPARTMENT."description" IS '부서 설명';
COMMENT ON COLUMN DEPARTMENT.PARENT_DEPARTMENT_ID IS '상위 부서 ID (계층 구조, 최대 5단계)';
COMMENT ON COLUMN DEPARTMENT."deleted_at" IS 'Soft delete 시각';

-- ============================================================
-- 3. POSITION (직급) 테이블
-- ============================================================

CREATE TABLE POSITION (
  "id"              NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  "name"            VARCHAR2(50) NOT NULL,
  "code"            VARCHAR2(20) NOT NULL,
  "level"           NUMBER(2) NOT NULL,
  "created_at"      TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
  "updated_at"      TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
  "deleted_at"      TIMESTAMP,

  -- 레벨 제약 (1-10)
  CONSTRAINT CHK_POSITION_LEVEL CHECK ("level" BETWEEN 1 AND 10)
);

-- 직급명 유니크 (soft delete 제외)
CREATE UNIQUE INDEX UQ_POSITION_NAME ON POSITION("name")
  WHERE "deleted_at" IS NULL;

-- 직급 코드 유니크
CREATE UNIQUE INDEX UQ_POSITION_CODE ON POSITION("code");

-- 인덱스
CREATE INDEX IDX_POSITION_LEVEL ON POSITION("level");
CREATE INDEX IDX_POSITION_DELETED ON POSITION("deleted_at");

COMMENT ON TABLE POSITION IS '직급 정보';
COMMENT ON COLUMN POSITION."id" IS '직급 ID (PK)';
COMMENT ON COLUMN POSITION."name" IS '직급명 (예: 사원, 대리, 과장)';
COMMENT ON COLUMN POSITION."code" IS '직급 코드 (POS-XXXXX, 자동 생성)';
COMMENT ON COLUMN POSITION."level" IS '직급 레벨 (1-10, 낮을수록 높은 직급)';
COMMENT ON COLUMN POSITION."deleted_at" IS 'Soft delete 시각';

-- ============================================================
-- 4. EMPLOYEE (직원) 테이블
-- ============================================================

CREATE TABLE EMPLOYEE (
  "id"              NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  "name"            VARCHAR2(100) NOT NULL,
  "email"           VARCHAR2(255) NOT NULL,
  "phone"           VARCHAR2(20),
  "job_title"       VARCHAR2(100),
  DEPARTMENT_ID     NUMBER NOT NULL,
  POSITION_ID       NUMBER NOT NULL,
  MANAGER_ID        NUMBER,
  "hired_at"        TIMESTAMP NOT NULL,
  "birthday"        TIMESTAMP,
  "created_at"      TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
  CREATED_BY_ID     NUMBER,
  "updated_at"      TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
  UPDATED_BY_ID     NUMBER,
  "deleted_at"      TIMESTAMP,

  -- 부서 FK (ON DELETE RESTRICT)
  CONSTRAINT FK_EMPLOYEE_DEPARTMENT
    FOREIGN KEY (DEPARTMENT_ID)
    REFERENCES DEPARTMENT("id"),

  -- 직급 FK (ON DELETE RESTRICT)
  CONSTRAINT FK_EMPLOYEE_POSITION
    FOREIGN KEY (POSITION_ID)
    REFERENCES POSITION("id"),

  -- 매니저 FK (ON DELETE SET NULL)
  CONSTRAINT FK_EMPLOYEE_MANAGER
    FOREIGN KEY (MANAGER_ID)
    REFERENCES EMPLOYEE("id")
    ON DELETE SET NULL,

  -- 생성자 FK (ON DELETE RESTRICT) - Decision #3
  CONSTRAINT FK_EMPLOYEE_CREATED_BY
    FOREIGN KEY (CREATED_BY_ID)
    REFERENCES EMPLOYEE("id"),

  -- 수정자 FK (ON DELETE RESTRICT) - Decision #3
  CONSTRAINT FK_EMPLOYEE_UPDATED_BY
    FOREIGN KEY (UPDATED_BY_ID)
    REFERENCES EMPLOYEE("id")
);

-- 이메일 유니크 (soft delete 제외)
CREATE UNIQUE INDEX UQ_EMPLOYEE_EMAIL ON EMPLOYEE("email")
  WHERE "deleted_at" IS NULL;

-- 인덱스
CREATE INDEX IDX_EMPLOYEE_DEPARTMENT ON EMPLOYEE(DEPARTMENT_ID);
CREATE INDEX IDX_EMPLOYEE_POSITION ON EMPLOYEE(POSITION_ID);
CREATE INDEX IDX_EMPLOYEE_MANAGER ON EMPLOYEE(MANAGER_ID);
CREATE INDEX IDX_EMPLOYEE_HIRED ON EMPLOYEE("hired_at");
CREATE INDEX IDX_EMPLOYEE_DELETED ON EMPLOYEE("deleted_at");
CREATE INDEX IDX_EMPLOYEE_NAME ON EMPLOYEE("name");

COMMENT ON TABLE EMPLOYEE IS '직원 정보';
COMMENT ON COLUMN EMPLOYEE."id" IS '직원 ID (PK)';
COMMENT ON COLUMN EMPLOYEE."name" IS '직원 이름';
COMMENT ON COLUMN EMPLOYEE."email" IS '이메일 (로그인 ID로 사용 가능)';
COMMENT ON COLUMN EMPLOYEE."phone" IS '전화번호';
COMMENT ON COLUMN EMPLOYEE."job_title" IS '직무/직책';
COMMENT ON COLUMN EMPLOYEE.DEPARTMENT_ID IS '소속 부서 ID';
COMMENT ON COLUMN EMPLOYEE.POSITION_ID IS '직급 ID';
COMMENT ON COLUMN EMPLOYEE.MANAGER_ID IS '상위 매니저 ID';
COMMENT ON COLUMN EMPLOYEE."hired_at" IS '입사일';
COMMENT ON COLUMN EMPLOYEE."birthday" IS '생일';
COMMENT ON COLUMN EMPLOYEE.CREATED_BY_ID IS '생성자 ID (ON DELETE RESTRICT)';
COMMENT ON COLUMN EMPLOYEE.UPDATED_BY_ID IS '수정자 ID (ON DELETE RESTRICT)';
COMMENT ON COLUMN EMPLOYEE."deleted_at" IS 'Soft delete 시각';

-- ============================================================
-- 5. ACCOUNT (계정) 테이블
-- ============================================================

CREATE TABLE ACCOUNT (
  "id"                  NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  EMPLOYEE_ID           NUMBER NOT NULL,
  "username"            VARCHAR2(50) NOT NULL,
  PASSWORD_HASH         VARCHAR2(255) NOT NULL,
  "role"                VARCHAR2(20) DEFAULT 'USER' NOT NULL,
  IS_ACTIVE             NUMBER(1) DEFAULT 1 NOT NULL,
  LAST_LOGIN_AT         TIMESTAMP,
  PASSWORD_CHANGED_AT   TIMESTAMP,
  "created_at"          TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
  "updated_at"          TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
  "deleted_at"          TIMESTAMP,

  -- 직원 FK (ON DELETE RESTRICT)
  CONSTRAINT FK_ACCOUNT_EMPLOYEE
    FOREIGN KEY (EMPLOYEE_ID)
    REFERENCES EMPLOYEE("id"),

  -- 역할 제약
  CONSTRAINT CHK_ACCOUNT_ROLE CHECK ("role" IN ('ADMIN', 'MANAGER', 'USER')),

  -- 활성 상태 제약
  CONSTRAINT CHK_ACCOUNT_ACTIVE CHECK (IS_ACTIVE IN (0, 1))
);

-- 직원당 하나의 계정만 허용
CREATE UNIQUE INDEX UQ_ACCOUNT_EMPLOYEE ON ACCOUNT(EMPLOYEE_ID);

-- username 유니크 (soft delete 제외)
CREATE UNIQUE INDEX UQ_ACCOUNT_USERNAME ON ACCOUNT("username")
  WHERE "deleted_at" IS NULL;

-- 인덱스
CREATE INDEX IDX_ACCOUNT_ACTIVE ON ACCOUNT(IS_ACTIVE);
CREATE INDEX IDX_ACCOUNT_ROLE ON ACCOUNT("role");
CREATE INDEX IDX_ACCOUNT_DELETED ON ACCOUNT("deleted_at");

COMMENT ON TABLE ACCOUNT IS '직원 계정 정보';
COMMENT ON COLUMN ACCOUNT."id" IS '계정 ID (PK)';
COMMENT ON COLUMN ACCOUNT.EMPLOYEE_ID IS '직원 ID (1:1 관계)';
COMMENT ON COLUMN ACCOUNT."username" IS '로그인 ID';
COMMENT ON COLUMN ACCOUNT.PASSWORD_HASH IS '비밀번호 해시 (bcrypt)';
COMMENT ON COLUMN ACCOUNT."role" IS '시스템 역할 (ADMIN/MANAGER/USER)';
COMMENT ON COLUMN ACCOUNT.IS_ACTIVE IS '활성 상태 (1=활성, 0=비활성, 로그인 차단용)';
COMMENT ON COLUMN ACCOUNT.LAST_LOGIN_AT IS '마지막 로그인 시각';
COMMENT ON COLUMN ACCOUNT.PASSWORD_CHANGED_AT IS '비밀번호 변경 시각';
COMMENT ON COLUMN ACCOUNT."deleted_at" IS 'Soft delete 시각';

-- ============================================================
-- 6. EMPLOYEE_HISTORY (감사 로그) 테이블
-- ============================================================

CREATE TABLE EMPLOYEE_HISTORY (
  "id"              NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  EMPLOYEE_ID       NUMBER NOT NULL,
  CHANGE_TYPE       VARCHAR2(20) NOT NULL,
  CHANGED_FIELDS    CLOB,
  CHANGED_BY_ID     NUMBER NOT NULL,
  CHANGED_AT        TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,

  -- 직원 FK (ON DELETE RESTRICT)
  CONSTRAINT FK_HISTORY_EMPLOYEE
    FOREIGN KEY (EMPLOYEE_ID)
    REFERENCES EMPLOYEE("id"),

  -- 변경자 FK (ON DELETE RESTRICT) - Decision #3
  CONSTRAINT FK_HISTORY_CHANGED_BY
    FOREIGN KEY (CHANGED_BY_ID)
    REFERENCES EMPLOYEE("id"),

  -- 변경 유형 제약
  CONSTRAINT CHK_HISTORY_TYPE CHECK (CHANGE_TYPE IN ('CREATE', 'UPDATE', 'DELETE', 'DEACTIVATE'))
);

-- 인덱스
CREATE INDEX IDX_HISTORY_EMPLOYEE ON EMPLOYEE_HISTORY(EMPLOYEE_ID);
CREATE INDEX IDX_HISTORY_CHANGED_AT ON EMPLOYEE_HISTORY(CHANGED_AT);
CREATE INDEX IDX_HISTORY_TYPE ON EMPLOYEE_HISTORY(CHANGE_TYPE);
CREATE INDEX IDX_HISTORY_EMPLOYEE_DATE ON EMPLOYEE_HISTORY(EMPLOYEE_ID, CHANGED_AT);

-- CHANGED_FIELDS JSON 검색용 (Oracle 21c)
-- CREATE SEARCH INDEX IDX_HISTORY_FIELDS ON EMPLOYEE_HISTORY(CHANGED_FIELDS) FOR JSON;

COMMENT ON TABLE EMPLOYEE_HISTORY IS '직원 변경 이력 (감사 로그)';
COMMENT ON COLUMN EMPLOYEE_HISTORY."id" IS '이력 ID (PK)';
COMMENT ON COLUMN EMPLOYEE_HISTORY.EMPLOYEE_ID IS '대상 직원 ID';
COMMENT ON COLUMN EMPLOYEE_HISTORY.CHANGE_TYPE IS '변경 유형 (CREATE/UPDATE/DELETE/DEACTIVATE)';
COMMENT ON COLUMN EMPLOYEE_HISTORY.CHANGED_FIELDS IS '변경 필드 정보 (JSON: {fieldName: {before, after}})';
COMMENT ON COLUMN EMPLOYEE_HISTORY.CHANGED_BY_ID IS '변경자 ID (ON DELETE RESTRICT)';
COMMENT ON COLUMN EMPLOYEE_HISTORY.CHANGED_AT IS '변경 시각';
