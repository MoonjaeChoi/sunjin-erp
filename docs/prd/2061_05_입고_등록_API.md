### 구현 상세: POST Handler

```typescript
import { NextRequest, NextResponse } from 'next/server';
import { getServerSession } from 'next-auth/next';
import { authOptions } from '@/lib/auth';
import { getRepository } from '@/lib/database';
import { Inventory } from '@/entities/Inventory';
import { InventoryHistory } from '@/entities/InventoryHistory';
import { validate } from 'class-validator';
import { CreateInventoryDto } from '@/dtos/inventory.dto';

export async function POST(request: NextRequest): Promise<NextResponse> {
  try {
    // 1. 인증 및 권한 검증
    const session = await getServerSession(authOptions);
    if (!session?.user) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }

    if (!['MANAGER', 'ADMIN'].includes(session.user.role)) {
      return NextResponse.json(
        { error: 'Forbidden: Only MANAGER and ADMIN can create inventory' },
        { status: 403 }
      );
    }

    // 2. Request Body 파싱
    const body = await request.json();

    // 3. DTO 검증
    const dto = Object.assign(new CreateInventoryDto(), body);
    const errors = await validate(dto);
    if (errors.length > 0) {
      return NextResponse.json(
        {
          error: 'Validation failed',
          details: errors.map((e) => ({
            field: e.property,
            messages: Object.values(e.constraints || {}),
          })),
        },
        { status: 400 }
      );
    }

    // 4. 시리얼 번호 중복 검증
    const inventoryRepository = getRepository(Inventory);
    const existing = await inventoryRepository.findOne({
      where: { serial_number: dto.serial_number, deleted_at: IsNull() },
    });

    if (existing) {
      return NextResponse.json(
        { error: 'Serial number already exists' },
        { status: 409 }
      );
    }

    // 5. Inventory 생성
    const inventory = new Inventory();
    inventory.category = dto.category;
    inventory.model = dto.model;
    inventory.serial_number = dto.serial_number;
    inventory.purchase_date = new Date(dto.purchase_date);
    inventory.purchase_from = dto.purchase_from;
    inventory.current_location = dto.current_location;
    inventory.current_status = '재고';
    inventory.notes = dto.notes || null;
    inventory.created_by_id = session.user.id;
    inventory.updated_by_id = session.user.id;

    const savedInventory = await inventoryRepository.save(inventory);

    // 6. 입고 이력 자동 생성
    const historyRepository = getRepository(InventoryHistory);
    const history = new InventoryHistory();
    history.inventory_id = savedInventory.id;
    history.change_type = '입고';
    history.previous_location = null;
    history.new_location = dto.current_location;
    history.previous_status = null;
    history.new_status = '재고';
    history.checkout_location = null;
    history.expected_checkin_date = null;
    history.reason = null;
    history.changed_by_id = session.user.id;

    const savedHistory = await historyRepository.save(history);

    // 7. 응답 (저장된 데이터 다시 조회)
    const result = await inventoryRepository.findOne({
      where: { id: savedInventory.id },
      relations: ['created_by', 'updated_by', 'histories'],
    });

    if (!result) {
      return NextResponse.json(
        { error: 'Failed to retrieve created inventory' },
        { status: 500 }
      );
    }

    return NextResponse.json(
      {
        id: result.id,
        category: result.category,
        model: result.model,
        serial_number: result.serial_number,
        purchase_date: result.purchase_date.toISOString().split('T')[0],
        purchase_from: result.purchase_from,
        current_location: result.current_location,
        current_status: result.current_status,
        notes: result.notes,
        created_by: {
          id: result.created_by.id,
          name: result.created_by.name,
        },
        updated_by: {
          id: result.updated_by.id,
          name: result.updated_by.name,
        },
        created_at: result.created_at.toISOString(),
        updated_at: result.updated_at.toISOString(),
        histories: result.histories.map((h) => ({
          id: h.id,
          change_type: h.change_type,
          previous_location: h.previous_location,
          new_location: h.new_location,
          previous_status: h.previous_status,
          new_status: h.new_status,
          checkout_location: h.checkout_location,
          expected_checkin_date: null,
          reason: null,
          changed_by: {
            id: h.changed_by.id,
            name: h.changed_by.name,
          },
          changed_at: h.changed_at.toISOString(),
        })),
      },
      { status: 201 }
    );
  } catch (error) {
    console.error('POST /api/inventory error:', error);
    return NextResponse.json(
      { error: 'Internal Server Error' },
      { status: 500 }
    );
  }
}
```

### DTO Validation: `src/dtos/inventory.dto.ts`

```typescript
import {
  IsString,
  IsEnum,
  MaxLength,
  MinLength,
  IsISO8601,
  IsNotEmpty,
  IsOptional,
  Validate,
} from 'class-validator';
import { ValidateSerialUnique } from '@/validators/validate-serial-unique.validator';

export class CreateInventoryDto {
  @IsEnum(['모니터', '노트북', '라우터', '프린터', '기타'])
  @IsNotEmpty()
  category!: string;

  @IsString()
  @MinLength(1)
  @MaxLength(255)
  @IsNotEmpty()
  model!: string;

  @IsString()
  @MinLength(1)
  @MaxLength(100)
  @Validate(ValidateSerialUnique)
  @IsNotEmpty()
  serial_number!: string;

  @IsISO8601()
  @IsNotEmpty()
  purchase_date!: string;

  @IsString()
  @MinLength(1)
  @MaxLength(255)
  @IsNotEmpty()
  purchase_from!: string;

  @IsString()
  @MinLength(1)
  @MaxLength(255)
  @IsNotEmpty()
  current_location!: string;

  @IsString()
  @MaxLength(4000)
  @IsOptional()
  notes?: string;
}
```

### Custom Validator: `src/validators/validate-serial-unique.validator.ts`

```typescript
import {
  ValidatorConstraint,
  ValidatorConstraintInterface,
  ValidationArguments,
} from 'class-validator';
import { getRepository } from '@/lib/database';
import { Inventory } from '@/entities/Inventory';
import { IsNull } from 'typeorm';

@ValidatorConstraint({ name: 'ValidateSerialUnique', async: true })
export class ValidateSerialUnique implements ValidatorConstraintInterface {
  async validate(serialNumber: string): Promise<boolean> {
    const repository = getRepository(Inventory);
    const existing = await repository.findOne({
      where: { serial_number: serialNumber, deleted_at: IsNull() },
    });
    return !existing; // true if unique, false if duplicate
  }

  defaultMessage(): string {
    return 'Serial number already exists';
  }
}
```
