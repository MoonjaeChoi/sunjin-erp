## 구현 목표

재고 상태인 장비의 물리적 보관 위치를 변경하는 API를 구현한다. 재고 상태에서만 가능하며, 위치 변경 이력을 기록한다.

**Endpoint**: `POST /api/inventory/[id]/relocate`

---

```typescript
{
  new_location: string;         // 필수, 새 위치
  reason?: string;              // 선택, 변경 사유
}
```

```typescript
{
  id: number;
  current_location: string;     // 새 위치로 업데이트
  updated_at: string;           // ISO 8601 timestamp
```

## 구현 상세

```typescript
import { NextRequest, NextResponse } from 'next/server';
import { getServerSession } from 'next-auth/next';
import { authOptions } from '@/lib/auth';
import { getRepository } from '@/lib/database';
import { Inventory } from '@/entities/Inventory';
import { InventoryHistory } from '@/entities/InventoryHistory';
import { IsNull } from 'typeorm';

export async function POST(
  request: NextRequest,
  { params }: { params: { id: string } }
): Promise<NextResponse> {
  try {
    const session = await getServerSession(authOptions);
    if (!session?.user) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }

    if (!['MANAGER', 'ADMIN'].includes(session.user.role)) {
      return NextResponse.json(
        { error: 'Forbidden: Only MANAGER and ADMIN can relocate inventory' },
        { status: 403 }
      );
    }

    const inventoryId = parseInt(params.id);
    if (isNaN(inventoryId)) {
      return NextResponse.json({ error: 'Invalid ID' }, { status: 400 });
    }

    const body = await request.json();
    const { new_location, reason } = body;

    if (!new_location) {
      return NextResponse.json(
        { error: 'new_location is required' },
        { status: 400 }
      );
    }

    const inventoryRepository = getRepository(Inventory);
    const inventory = await inventoryRepository.findOne({
      where: { id: inventoryId, deleted_at: IsNull() },
    });

    if (!inventory) {
      return NextResponse.json(
        { error: 'Inventory not found' },
        { status: 404 }
      );
    }

    // 재고 상태에서만 위치 변경 가능
    if (inventory.current_status !== '재고') {
      return NextResponse.json(
        { error: 'Location can only be changed for inventory in stock' },
        { status: 422 }
      );
    }

    // Inventory 업데이트
    const previousLocation = inventory.current_location;
    inventory.current_location = new_location;
    inventory.updated_by_id = session.user.id;
    inventory.updated_at = new Date();

    const updatedInventory = await inventoryRepository.save(inventory);

    // 위치 변경 이력 생성
    const historyRepository = getRepository(InventoryHistory);
    const history = new InventoryHistory();
    history.inventory_id = inventoryId;
    history.change_type = '위치변경';
    history.previous_location = previousLocation;
    history.new_location = new_location;
    history.previous_status = null;
    history.new_status = null;
    history.checkout_location = null;
    history.expected_checkin_date = null;
    history.reason = reason || null;
    history.changed_by_id = session.user.id;

    const savedHistory = await historyRepository.save(history);

    return NextResponse.json(
      {
        id: updatedInventory.id,
        current_location: updatedInventory.current_location,
        updated_at: updatedInventory.updated_at.toISOString(),
        history: {
          id: savedHistory.id,
          change_type: savedHistory.change_type,
          previous_location: savedHistory.previous_location,
          new_location: savedHistory.new_location,
          reason: savedHistory.reason,
          changed_by_id: savedHistory.changed_by_id,
          changed_at: savedHistory.changed_at.toISOString(),
        },
      },
      { status: 200 }
    );
  } catch (error) {
    console.error('POST /api/inventory/[id]/relocate error:', error);
    return NextResponse.json(
      { error: 'Internal Server Error' },
      { status: 500 }
    );
  }
}
```
