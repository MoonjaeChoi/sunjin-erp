## 구현 상세

```typescript
import { NextRequest, NextResponse } from 'next/server';
import { getServerSession } from 'next-auth/next';
import { authOptions } from '@/lib/auth';
import { getRepository } from '@/lib/database';
import { Inventory } from '@/entities/Inventory';
import { IsNull } from 'typeorm';

export async function GET(request: NextRequest): Promise<NextResponse> {
  try {
    const session = await getServerSession(authOptions);
    if (!session?.user) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }

    // MANAGER, ADMIN만 통계 조회 가능 (v2: USER 제외)
    if (!['MANAGER', 'ADMIN'].includes(session.user.role)) {
      return NextResponse.json(
        { error: 'Forbidden: Only MANAGER and ADMIN can view statistics' },
        { status: 403 }
      );
    }

    const repository = getRepository(Inventory);

    // 1. 전체 수량
    const total = await repository.count({
      where: { deleted_at: IsNull() },
    });

    // 2. 상태별 집계
    const byStatusData = await repository
      .createQueryBuilder('i')
      .select('i.current_status', 'status')
      .addSelect('COUNT(*)', 'count')
      .where('i.deleted_at IS NULL')
      .groupBy('i.current_status')
      .getRawMany();

    const byStatus: Record<string, number> = {
      '재고': 0,
      '출고': 0,
      '고장': 0,
      '폐기': 0,
    };

    byStatusData.forEach(
      (row: { status: string; count: string }) => {
        byStatus[row.status] = parseInt(row.count);
      }
    );

    // 3. 카테고리별 집계
    const byCategoryData = await repository
      .createQueryBuilder('i')
      .select('i.category', 'category')
      .addSelect('i.current_status', 'status')
      .addSelect('COUNT(*)', 'count')
      .where('i.deleted_at IS NULL')
      .groupBy('i.category')
      .addGroupBy('i.current_status')
      .orderBy('i.category', 'ASC')
      .addOrderBy('i.current_status', 'ASC')
      .getRawMany();

    // 카테고리별로 그룹핑
    const byCategoryMap = new Map<
      string,
      Record<string, number>
    >();

    byCategoryData.forEach(
      (row: { category: string; status: string; count: string }) => {
        if (!byCategoryMap.has(row.category)) {
          byCategoryMap.set(row.category, {
            '재고': 0,
            '출고': 0,
            '고장': 0,
            '폐기': 0,
          });
        }
        const map = byCategoryMap.get(row.category)!;
        map[row.status] = parseInt(row.count);
      }
    );

    const byCategory = Array.from(byCategoryMap.entries())
      .map(([category, statuses]) => ({
        category,
        total: Object.values(statuses).reduce((a, b) => a + b, 0),
        statuses,
      }))
      .sort((a, b) => a.category.localeCompare(b.category));

    // 4. 응답
    return NextResponse.json(
      {
        total,
        byStatus,
        byCategory,
        updated_at: new Date().toISOString(),
      },
      {
        status: 200,
        headers: {
          'Cache-Control': 'public, max-age=60', // 1분 캐시
        },
      }
    );
  } catch (error) {
    console.error('GET /api/inventory/stats error:', error);
    return NextResponse.json(
      { error: 'Internal Server Error' },
      { status: 500 }
    );
  }
}
```
