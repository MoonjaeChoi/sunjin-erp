```typescript
// Generated: 2026-01-28 15:30:00 KST

import { NextRequest, NextResponse } from 'next/server';
import { getServerSession } from 'next-auth';
import { authOptions } from '@/lib/auth';
import { executeQuerySingle, executeUpdate } from '@/lib/db-direct';

export const dynamic = 'force-dynamic';

interface RouteParams {
  params: { id: string; attachmentId: string };
}

// ============================================================
// DELETE /api/notices/[id]/attachments/[attachmentId]
// ============================================================

export async function DELETE(req: NextRequest, { params }: RouteParams): Promise<NextResponse> {
  try {
    // 1. 인증 검증
    const session = await getServerSession(authOptions);
    if (!session?.user) {
      return NextResponse.json({ message: 'Unauthorized' }, { status: 401 });
    }

    const noticeId = parseInt(params.id, 10);
    const attachmentId = parseInt(params.attachmentId, 10);

    if (isNaN(noticeId) || isNaN(attachmentId)) {
      return NextResponse.json({ message: 'Invalid ID' }, { status: 400 });
    }

    const user = session.user as any;
    const userId = user.id;
    const userRole = user.role;

    // 2. 첨부파일 및 게시물 조회
    const attachment = await executeQuerySingle<any>(
      `SELECT a.ID, a.NOTICE_ID, a.FILENAME, n.AUTHOR_ID AS NOTICE_AUTHOR_ID
       FROM NOTICE_ATTACHMENT a
       JOIN NOTICE n ON n.ID = a.NOTICE_ID
       WHERE a.ID = :attachmentId AND a.NOTICE_ID = :noticeId
         AND a."deleted_at" IS NULL AND n."deleted_at" IS NULL`,
      { attachmentId, noticeId }
    );

    if (!attachment) {
      return NextResponse.json({ message: 'Attachment not found' }, { status: 404 });
    }

    // 3. 권한 확인 (ADMIN 또는 게시물 작성자)
    if (userRole !== 'ADMIN' && attachment.NOTICE_AUTHOR_ID !== userId) {
      return NextResponse.json({ message: 'Forbidden' }, { status: 403 });
    }

    // 4. Soft delete
    await executeUpdate(
      `UPDATE NOTICE_ATTACHMENT SET "deleted_at" = SYSTIMESTAMP WHERE ID = :attachmentId`,
      { attachmentId }
    );

    return NextResponse.json({ message: '첨부파일이 삭제되었습니다' });
  } catch (error) {
    console.error('DELETE /api/notices/[id]/attachments/[attachmentId] error:', error);
    return NextResponse.json({ message: 'Internal server error' }, { status: 500 });
  }
}
```
