```typescript
// e2e/issue-workflow.spec.ts

test('Complete issue workflow: create → assign → process → complete → rollback', async ({ page }) => {
  // 1. 로그인 (USER)
  await page.goto('/login');
  await page.fill('input[name="email"]', 'user@test.com');
  await page.fill('input[name="password"]', 'password');
  await page.click('button:has-text("로그인")');

  // 2. 목록 페이지 확인
  await page.goto('/issues');
  const initialCount = await page.locator('table tbody tr').count();

  // 3. 신규 등록 (USER)
  await page.click('button:has-text("신규 등록")');
  await page.selectOption('select[name="customer_id"]', '1');
  await page.fill('input[name="title"]', '테스트 장애');
  await page.fill('textarea[name="description"]', '테스트 장애 설명입니다. 최소 10자 이상.');
  await page.click('input[value="HIGH"]');
  await page.click('button:has-text("등록")');

  await page.waitForSelector('[data-testid="success-toast"]');
  expect(await page.locator('table tbody tr').count()).toBe(initialCount + 1);

  // 4. 상세 페이지 조회
  const firstRow = page.locator('table tbody tr').first();
  await firstRow.click();
  await page.waitForURL('**/issues/**');

  // 5. 담당자 할당 (MANAGER로 변경)
  await page.context().clearCookies();
  await page.goto('/login');
  await page.fill('input[name="email"]', 'manager@test.com');
  await page.fill('input[name="password"]', 'password');
  await page.click('button:has-text("로그인")');

  // 목록에서 테스트 이슈 찾기
  await page.goto('/issues');
  const issueCell = page.locator('text=테스트 장애').first();
  await issueCell.click();

  // 담당자 할당
  await page.selectOption('select[name="assigned_to_id"]', 'user1');
  await page.click('button:has-text("수정")');
  await page.waitForSelector('[data-testid="success-toast"]');

  // 6. 처리 정보 입력
  await page.fill('input[name="treatment_time_minutes"]', '150');
  await page.click('input[value="REMOTE"]');
  await page.fill('textarea[name="treatment_result"]', '원격 지원으로 해결되었습니다.');
  await page.click('button:has-text("수정")');

  // 7. 상태를 완료로 변경
  await page.selectOption('select[name="status"]', 'COMPLETED');
  await page.click('button:has-text("수정")');

  const completedStatus = await page.locator('[data-testid="status-badge"]').textContent();
  expect(completedStatus).toContain('완료');

  // 8. 상태 롤백 (ADMIN로 변경)
  await page.context().clearCookies();
  await page.goto('/login');
  await page.fill('input[name="email"]', 'admin@test.com');
  await page.fill('input[name="password"]', 'password');
  await page.click('button:has-text("로그인")');

  // 상세 페이지로 이동
  await page.goto('/issues');
  await page.locator('text=테스트 장애').first().click();

  // 롤백 버튼 확인
  const rollbackBtn = page.locator('button:has-text("상태 되돌리기")');
  expect(rollbackBtn).toBeVisible();

  // 롤백 실행
  await rollbackBtn.click();
  await page.click('text=확인');
  await page.waitForSelector('[data-testid="success-toast"]');

  const rolledBackStatus = await page.locator('[data-testid="status-badge"]').textContent();
  expect(rolledBackStatus).toContain('진행중');
});
```

```typescript
// e2e/issue-filters.spec.ts

test('Filter and search functionality', async ({ page }) => {
  await page.goto('/login');
  await page.fill('input[name="email"]', 'admin@test.com');
  await page.fill('input[name="password"]', 'password');
  await page.click('button:has-text("로그인")');

  await page.goto('/issues');

  // 1. 고객사 필터
  await page.selectOption('select[name="customer_id"]', '1');
  await page.click('button:has-text("검색")');

  // 모든 행이 해당 고객사만 표시
  const rows = await page.locator('table tbody tr');
  for (let i = 0; i < await rows.count(); i++) {
    const customerCell = rows.nth(i).locator('td:nth-child(2)');
    expect(await customerCell.textContent()).toContain('고객사명');
  }

  // 2. 다중 상태 필터
  await page.check('input[id="status_INTAKE"]');
  await page.check('input[id="status_IN_PROGRESS"]');
  await page.click('button:has-text("검색")');

  // 3. 키워드 검색
  await page.fill('input[name="keyword"]', 'DB');
  await page.click('button:has-text("검색")');

  // 제목에 "DB" 포함된 것만 표시
  const titleCells = await page.locator('table tbody td:first-child');
  for (let i = 0; i < await titleCells.count(); i++) {
    expect(await titleCells.nth(i).textContent()).toContain('DB');
  }

  // 4. 초기화
  await page.click('button:has-text("초기화")');
  expect(await page.inputValue('input[name="keyword"]')).toBe('');
});
```

```typescript
test('Upload and delete attachments', async ({ page }) => {
  await page.goto('/login');
  await page.fill('input[name="email"]', 'admin@test.com');
  await page.fill('input[name="password"]', 'password');
  await page.click('button:has-text("로그인")');

  // 상세 페이지로 이동
  await page.goto('/issues/1');

  // 파일 업로드
  const fileInput = page.locator('input[type="file"]');
  await fileInput.setInputFiles('test.pdf');
  await page.click('button:has-text("업로드")');

  await page.waitForSelector('[data-testid="success-toast"]');

  // 파일 확인
  const fileName = page.locator('[data-testid="attachment-name"]');
  expect(fileName).toContainText('test.pdf');

  // 파일 삭제
  await page.click('button:has-text("삭제")');
  await page.click('text=확인');

  await page.waitForSelector('[data-testid="success-toast"]');
  expect(fileName).not.toBeVisible();
});
```
