### 3. Client Component: `src/components/features/inventory/InventoryPage.tsx`

```typescript
'use client';

import { useState, useCallback } from 'react';
import { useRouter, useSearchParams } from 'next/navigation';
import { useInventoryList, useInventoryStats } from '@/hooks/useInventory*';
import {
  InventorySearchParams,
  InventoryStatus,
  InventoryFilters,
} from '@/types/inventory';
import { InventoryStats } from './InventoryStats';
import { InventoryFilters as FiltersComponent } from './InventoryFilters';
import { InventoryTable } from './InventoryTable';
import { Button } from '@/components/ui/button';
import { Plus } from 'lucide-react';

export function InventoryPage({ userRole }: { userRole: string }) {
  const router = useRouter();
  const searchParams = useSearchParams();

  // 필터 상태 초기화
  const [filters, setFilters] = useState<InventoryFilters>({
    categories: searchParams.getAll('categories'),
    statuses: (searchParams.getAll('status') as InventoryStatus[]) || [],
    location: searchParams.get('location') || '',
    search: searchParams.get('search') || '',
    sortBy: searchParams.get('sortBy') || 'category',
    order: (searchParams.get('order') as 'asc' | 'desc') || 'asc',
    page: parseInt(searchParams.get('page') || '1'),
    pageSize: parseInt(searchParams.get('pageSize') || '20'),
  });

  // 쿼리 실행
  const inventoryQuery = useInventoryList(filters);
  const statsQuery = useInventoryStats();

  // 필터 변경
  const handleFilterChange = useCallback(
    (newFilters: Partial<InventoryFilters>) => {
      const updated = { ...filters, ...newFilters, page: 1 };
      setFilters(updated);
      updateURL(updated);
    },
    [filters]
  );

  // URL 업데이트
  const updateURL = (filters: InventoryFilters) => {
    const params = new URLSearchParams();
    if (filters.categories.length) {
      filters.categories.forEach((c) => params.append('categories', c));
    }
    if (filters.statuses.length) {
      filters.statuses.forEach((s) => params.append('status', s));
    }
    if (filters.location) params.set('location', filters.location);
    if (filters.search) params.set('search', filters.search);
    params.set('page', String(filters.page));
    params.set('pageSize', String(filters.pageSize));
    if (filters.sortBy) params.set('sortBy', filters.sortBy);
    if (filters.order) params.set('order', filters.order);

    router.push(`/inventory?${params.toString()}`);
  };

  return (
    <div className="space-y-6 p-6">
      {/* 페이지 제목 */}
      <div className="flex items-center justify-between">
        <h1 className="text-3xl font-bold">재고 관리</h1>
        {['MANAGER', 'ADMIN'].includes(userRole) && (
          <Button onClick={() => router.push('/inventory/new')}>
            <Plus className="mr-2 h-4 w-4" />
            입고 등록
          </Button>
        )}
      </div>

      {/* 통계 패널 */}
      {statsQuery.data && <InventoryStats data={statsQuery.data} />}

      {/* 필터 패널 */}
      <FiltersComponent
        filters={filters}
        onFilterChange={handleFilterChange}
      />

      {/* 목록 테이블 */}
      <InventoryTable
        data={inventoryQuery.data?.data || []}
        loading={inventoryQuery.isLoading}
        pagination={inventoryQuery.data?.pagination}
        onPageChange={(page) => handleFilterChange({ page })}
        onPageSizeChange={(pageSize) =>
          handleFilterChange({ pageSize, page: 1 })
        }
        userRole={userRole}
      />
    </div>
  );
}
```
