```typescript
export async function POST(request: Request, { params }: { params: { id: string } }) {
  // 1. 세션 검증
  const session = await getServerSession(authOptions);
  if (!session) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });

  // 2. 프로젝트 존재/권한 확인
  const project = await findProjectWithAccess(params.id, session.user, 'edit');
  if (!project) return error response;

  // 3. FormData 파싱
  const formData = await request.formData();
  const file = formData.get('file') as File;
  const category = formData.get('category') as string;

  // 4. 파일 유효성 검증
  validateFile(file);       // 확장자, 크기, MIME
  validateCategory(category); // enum 검증

  // 5. 파일 저장
  const uuid = crypto.randomUUID();
  const ext = path.extname(file.name).toLowerCase();
  const savePath = path.join(UPLOAD_DIR, 'projects', params.id, `${uuid}${ext}`);

  // 디렉토리 생성 (recursive)
  await fs.mkdir(path.dirname(savePath), { recursive: true });

  // 파일 쓰기
  const buffer = Buffer.from(await file.arrayBuffer());
  await fs.writeFile(savePath, buffer);

  // 6. DB 레코드 삽입
  const attachment = await attachmentRepository.save({
    project_id: project.id,
    file_path: savePath,
    file_name: file.name,
    file_size: buffer.length,
    category: category,
  });

  // 7. 응답
  return NextResponse.json({
    id: attachment.id,
    file_name: attachment.file_name,
    category: attachment.category,
    file_size: attachment.file_size,
    created_at: attachment.created_at,
  }, { status: 201 });
}
```

#### 2. 파일 유효성 검증

```typescript
const ALLOWED_EXTENSIONS = ['.pdf', '.docx', '.xlsx', '.jpg', '.jpeg', '.png'];
const MAX_FILE_SIZE = 10 * 1024 * 1024; // 10MB

const MIME_TYPE_MAP: Record<string, string[]> = {
  '.pdf': ['application/pdf'],
  '.docx': ['application/vnd.openxmlformats-officedocument.wordprocessingml.document'],
  '.xlsx': ['application/vnd.openxmlformats-officedocument.spreadsheetml.sheet'],
  '.jpg': ['image/jpeg'],
  '.jpeg': ['image/jpeg'],
  '.png': ['image/png'],
};

function validateFile(file: File): void {
  if (!file) throw new ValidationError('File is required');

  // 확장자 검증
  const ext = path.extname(file.name).toLowerCase();
  if (!ALLOWED_EXTENSIONS.includes(ext)) {
    throw new ValidationError(`허용되지 않는 파일 형식입니다. 허용: ${ALLOWED_EXTENSIONS.join(', ')}`);
  }

  // 크기 검증
  if (file.size > MAX_FILE_SIZE) {
    throw new ValidationError('파일 크기는 10MB를 초과할 수 없습니다.');
  }

  // MIME type 검증
  const allowedMimes = MIME_TYPE_MAP[ext];
  if (allowedMimes && !allowedMimes.includes(file.type)) {
    throw new ValidationError('파일 형식과 MIME type이 일치하지 않습니다.');
  }
}
```

#### 4. GET /api/projects/[id]/attachments/[attachmentId] - 파일 다운로드

```typescript
export async function GET(
  request: Request,
  { params }: { params: { id: string; attachmentId: string } }
) {
  // 1. 세션 검증
  const session = await getServerSession(authOptions);
  if (!session) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });

  // 2. 프로젝트 + 첨부파일 조회 및 권한 확인
  const project = await findProjectWithAccess(params.id, session.user, 'view');
  if (!project) return error response;

  const attachment = await attachmentRepository.findOne({
    where: { id: Number(params.attachmentId), project_id: project.id },
  });
  if (!attachment) return NextResponse.json({ error: 'Attachment not found' }, { status: 404 });

  // 3. 파일 읽기
  const fileBuffer = await fs.readFile(attachment.file_path);

  // 4. Content-Disposition 헤더 설정 (원본 파일명)
  const encodedFileName = encodeURIComponent(attachment.file_name);

  return new Response(fileBuffer, {
    headers: {
      'Content-Type': 'application/octet-stream',
      'Content-Disposition': `attachment; filename*=UTF-8''${encodedFileName}`,
      'Content-Length': String(fileBuffer.length),
    },
  });
}
```

#### 5. DELETE /api/projects/[id]/attachments/[attachmentId] - 파일 삭제

```typescript
export async function DELETE(
  request: Request,
  { params }: { params: { id: string; attachmentId: string } }
) {
  // 1. 세션 검증
  const session = await getServerSession(authOptions);
  if (!session) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });

  // 2. 프로젝트 권한 확인 (edit 권한 필요)
  const project = await findProjectWithAccess(params.id, session.user, 'edit');
  if (!project) return error response;

  // 3. 첨부파일 조회
  const attachment = await attachmentRepository.findOne({
    where: { id: Number(params.attachmentId), project_id: project.id },
  });
  if (!attachment) return NextResponse.json({ error: 'Attachment not found' }, { status: 404 });

  // 4. 파일 시스템에서 삭제
  try {
    await fs.unlink(attachment.file_path);
  } catch (err) {
    // 파일이 이미 없는 경우에도 DB 레코드는 삭제 진행
    console.error('File deletion failed:', err);
  }

  // 5. DB 레코드 물리 삭제 (첨부파일은 soft delete 아닌 물리 삭제)
  await attachmentRepository.delete(attachment.id);

  // 6. 응답
  return NextResponse.json({ success: true }, { status: 200 });
}
```

#### 6. 보안 고려사항

- **Path Traversal 방지:** 파일명에 UUID 사용, 원본 파일명은 DB에만 저장
- **저장 경로:** `UPLOAD_DIR/projects/{projectId}/{uuid}.{ext}` 구조
- **MIME type 이중 검증:** 확장자와 MIME type 모두 확인
- **파일 크기 제한:** 10MB 초과 시 즉시 거부
- **디렉토리 구조:** 프로젝트별 하위 디렉토리로 격리

## Acceptance Criteria

- [ ] POST /api/projects/[id]/attachments가 FormData(file + category)를 받아 파일을 저장한다
- [ ] 파일이 UPLOAD_DIR/projects/{id}/{uuid}.{ext} 경로에 저장된다
- [ ] PROJECT_ATTACHMENT 레코드가 생성되며 file_name, file_path, file_size, category가 기록된다
- [ ] 확장자 whitelist(pdf, docx, xlsx, jpg, jpeg, png)만 허용된다
- [ ] 10MB 초과 파일은 400 에러를 반환한다
- [ ] MIME type과 확장자 불일치 시 400 에러를 반환한다
- [ ] 유효하지 않은 category 값은 400 에러를 반환한다
- [ ] GET /api/projects/[id]/attachments/[attachmentId]가 파일을 스트리밍 다운로드한다
- [ ] Content-Disposition 헤더에 원본 파일명이 UTF-8로 인코딩되어 포함된다
- [ ] DELETE /api/projects/[id]/attachments/[attachmentId]가 파일 + DB 레코드를 물리 삭제한다
- [ ] RBAC: 업로드/삭제는 프로젝트 edit 권한 필요 (USER=본인, MANAGER=부서, ADMIN=전체)
- [ ] RBAC: 다운로드는 프로젝트 view 권한 필요
- [ ] 인증되지 않은 요청은 401을 반환한다
- [ ] 존재하지 않는 프로젝트/첨부파일 ID는 404를 반환한다
