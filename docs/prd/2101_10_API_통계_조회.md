// Generated: 2026-01-28 15:30:00 KST

import { NextRequest, NextResponse } from 'next/server';
import { getServerSession } from 'next-auth';
import { authOptions } from '@/lib/auth';
import { executeQuery } from '@/lib/db-direct';
import type { NoticeStatistics, NoticeType } from '@/types/notice';

export const dynamic = 'force-dynamic';

// ============================================================
// GET /api/notices/statistics - 통계 조회 (ADMIN 전용)
// ============================================================

export async function GET(req: NextRequest): Promise<NextResponse> {
  try {
    // 1. 인증 검증
    const session = await getServerSession(authOptions);
    if (!session?.user) {
      return NextResponse.json({ message: 'Unauthorized' }, { status: 401 });
    }

    const user = session.user as any;

    // 2. ADMIN 권한 확인
    if (user.role !== 'ADMIN') {
      return NextResponse.json({ message: 'Forbidden' }, { status: 403 });
    }

    // 3. 전체 게시물 수
    const totalResult = await executeQuery<any>(
      `SELECT COUNT(*) AS TOTAL FROM NOTICE WHERE "deleted_at" IS NULL`,
      {}
    );
    const totalCount = parseInt(totalResult.rows[0]?.TOTAL || '0', 10);

    // 4. 유형별 게시물 수
    const typeCountResult = await executeQuery<any>(
      `SELECT TYPE, COUNT(*) AS CNT
       FROM NOTICE
       WHERE "deleted_at" IS NULL
       GROUP BY TYPE`,
      {}
    );

    const typeCount: Record<NoticeType, number> = {
      '공지': 0,
      '공유': 0,
      '지시': 0,
      '건의': 0,
      '자유': 0,
    };

    typeCountResult.rows.forEach((row: any) => {
      if (row.TYPE in typeCount) {
        typeCount[row.TYPE as NoticeType] = row.CNT;
      }
    });

    // 5. 상위 조회수 게시물 TOP 5
    const topViewedResult = await executeQuery<any>(
      `SELECT n.ID, n.TITLE, n.TYPE, n.AUTHOR_ID, e.NAME AS AUTHOR_NAME,
              n.VIEW_COUNT, n."created_at" AS CREATED_AT,
              (SELECT COUNT(*) FROM COMMENT c WHERE c.NOTICE_ID = n.ID AND c."deleted_at" IS NULL) AS COMMENT_COUNT
       FROM NOTICE n
       LEFT JOIN EMPLOYEE e ON e.ID = n.AUTHOR_ID
       WHERE n."deleted_at" IS NULL
       ORDER BY n.VIEW_COUNT DESC
       FETCH FIRST 5 ROWS ONLY`,
      {}
    );

    const topViewed = topViewedResult.rows.map((row: any) => ({
      id: row.ID,
      title: row.TITLE,
      type: row.TYPE as NoticeType,
      authorId: row.AUTHOR_ID,
      authorName: row.AUTHOR_NAME || '',
      viewCount: row.VIEW_COUNT || 0,
      commentCount: row.COMMENT_COUNT || 0,
      attachmentCount: 0,
      createdAt: row.CREATED_AT?.toISOString() || '',
      updatedAt: '',
    }));

    // 6. 상위 댓글 수 게시물 TOP 5
    const topCommentedResult = await executeQuery<any>(
      `SELECT n.ID, n.TITLE, n.TYPE, n.AUTHOR_ID, e.NAME AS AUTHOR_NAME,
              n.VIEW_COUNT, n."created_at" AS CREATED_AT,
              (SELECT COUNT(*) FROM COMMENT c WHERE c.NOTICE_ID = n.ID AND c."deleted_at" IS NULL) AS COMMENT_COUNT
       FROM NOTICE n
       LEFT JOIN EMPLOYEE e ON e.ID = n.AUTHOR_ID
       WHERE n."deleted_at" IS NULL
       ORDER BY COMMENT_COUNT DESC
       FETCH FIRST 5 ROWS ONLY`,
      {}
    );

    const topCommented = topCommentedResult.rows.map((row: any) => ({
      id: row.ID,
      title: row.TITLE,
      type: row.TYPE as NoticeType,
      authorId: row.AUTHOR_ID,
      authorName: row.AUTHOR_NAME || '',
      viewCount: row.VIEW_COUNT || 0,
      commentCount: row.COMMENT_COUNT || 0,
      attachmentCount: 0,
      createdAt: row.CREATED_AT?.toISOString() || '',
      updatedAt: '',
    }));

    // 7. 최근 7일 활동 추이
    const recentActivityResult = await executeQuery<any>(
      `SELECT
         TO_CHAR(TRUNC(dt.activity_date), 'YYYY-MM-DD') AS ACTIVITY_DATE,
         NVL(nc.notice_count, 0) AS NOTICE_COUNT,
         NVL(cc.comment_count, 0) AS COMMENT_COUNT
       FROM (
         SELECT TRUNC(SYSDATE) - LEVEL + 1 AS activity_date
         FROM DUAL
         CONNECT BY LEVEL <= 7
       ) dt
       LEFT JOIN (
         SELECT TRUNC("created_at") AS created_date, COUNT(*) AS notice_count
         FROM NOTICE
         WHERE "deleted_at" IS NULL AND "created_at" >= TRUNC(SYSDATE) - 6
         GROUP BY TRUNC("created_at")
       ) nc ON nc.created_date = dt.activity_date
       LEFT JOIN (
         SELECT TRUNC("created_at") AS created_date, COUNT(*) AS comment_count
         FROM COMMENT
         WHERE "deleted_at" IS NULL AND "created_at" >= TRUNC(SYSDATE) - 6
         GROUP BY TRUNC("created_at")
       ) cc ON cc.created_date = dt.activity_date
       ORDER BY dt.activity_date ASC`,
      {}
    );

    const recentActivity = recentActivityResult.rows.map((row: any) => ({
      date: row.ACTIVITY_DATE,
      noticeCount: row.NOTICE_COUNT || 0,
      commentCount: row.COMMENT_COUNT || 0,
    }));

    // 8. 응답
    const statistics: NoticeStatistics = {
      totalCount,
      typeCount,
      topViewed,
      topCommented,
      recentActivity,
    };

    return NextResponse.json({ data: statistics });
  } catch (error) {
    console.error('GET /api/notices/statistics error:', error);
    return NextResponse.json({ message: 'Internal server error' }, { status: 500 });
  }
}
