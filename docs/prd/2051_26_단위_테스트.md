  it('should allow 재고 → 출고', () => {
    const result = InventoryService.validateStateTransition('재고', '출고');
    expect(result.valid).toBe(true);
  });

  it('should allow 출고 → 재고', () => {
    const result = InventoryService.validateStateTransition('출고', '재고');
    expect(result.valid).toBe(true);
  });

  it('should deny 출고 → 폐기', () => {
    const result = InventoryService.validateStateTransition('출고', '폐기');
    expect(result.valid).toBe(false);
    expect(result.error).toContain('반납');
  });

  it('should deny 폐기 state changes', () => {
    const result = InventoryService.validateStateTransition('폐기', '고장');
    expect(result.valid).toBe(false);
  });
});

describe('InventoryService.calculateOverdueStatus', () => {
  it('should return is_overdue=true for past expected_checkin_date', () => {
    const yesterday = new Date();
    yesterday.setDate(yesterday.getDate() - 5);
    
    const result = InventoryService.calculateOverdueStatus('출고', yesterday);
    expect(result.is_overdue).toBe(true);
    expect(result.overdue_days).toBe(5);
  });

  it('should return false for 재고 status', () => {
    const yesterday = new Date();
    yesterday.setDate(yesterday.getDate() - 5);
    
    const result = InventoryService.calculateOverdueStatus('재고', yesterday);
    expect(result.is_overdue).toBe(false);
  });

  it('should return false for future date', () => {
    const tomorrow = new Date();
    tomorrow.setDate(tomorrow.getDate() + 1);
    
    const result = InventoryService.calculateOverdueStatus('출고', tomorrow);
    expect(result.is_overdue).toBe(false);
  });

    const { result } = renderHook(() => useInventoryList());
    
    await waitFor(() => {
      expect(result.current.data).toBeDefined();
    });
    
    expect(result.current.data?.pagination).toBeDefined();
  });

  it('should update cache key on filter change', async () => {
    const { result, rerender } = renderHook(
      ({ params }) => useInventoryList(params),
      { initialProps: { params: { page: 1 } } }
    );

    const initialKey = (result.current as any).queryKey;
    
    rerender({ params: { page: 2 } });
    
    const updatedKey = (result.current as any).queryKey;
    expect(initialKey).not.toEqual(updatedKey);
  });
});
