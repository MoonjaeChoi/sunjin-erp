```typescript
        pagination={data?.pagination}

    </div>
  );
}
```

## 2081_11: 폼 컴포넌트 (CustomerForm)

### 핵심 구현

**파일**: `src/components/features/customers/CustomerFormClient.tsx`

```typescript
'use client';

import { useForm } from 'react-hook-form';
import { useCreateCustomer, useUpdateCustomer } from '@/hooks/useCustomerMutations';
import { useCustomer } from '@/hooks/useCustomer';
import { CreateCustomerRequest, UpdateCustomerRequest, CustomerClassification } from '@/types/customer';
import { useRouter } from 'next/navigation';
import { toast } from 'sonner';

interface CustomerFormClientProps {
  isNew?: boolean;
  customerId?: number;
}

export default function CustomerFormClient({ isNew, customerId }: CustomerFormClientProps) {
  const router = useRouter();
  const { register, handleSubmit, formState: { errors }, reset } = useForm({
    defaultValues: isNew ? {} : undefined
  });

  const { data: customerData, isLoading: isLoadingDetail } = useCustomer(customerId);
  const createMutation = useCreateCustomer();
  const updateMutation = useUpdateCustomer(customerId!);

  const onSubmit = async (formData: any) => {
    try {
      if (isNew) {
        await createMutation.mutateAsync(formData);
        toast.success('고객이 등록되었습니다.');
        router.push('/customers');
      } else {
        await updateMutation.mutateAsync(formData);
        toast.success('고객이 수정되었습니다.');
        router.refresh();
      }
    } catch (error) {
      toast.error('오류가 발생했습니다.');
    }
  };

  return (
    <form onSubmit={handleSubmit(onSubmit)} className="space-y-6 max-w-2xl">
      {/* 고객명 */}
      <div>
        <label>고객명 *</label>
        <input {...register('name', { required: true })} />
        {errors.name && <span>필수 필드입니다</span>}
      </div>

      {/* 고객 분류 */}
      <div>
        <label>고객 분류 *</label>
        <select {...register('classification', { required: true })}>
          <option value="">선택</option>
          <option value="RESELLER">리셀러</option>
          <option value="END_USER">최종고객</option>
          <option value="MAINTENANCE">유지보수</option>
          <option value="GENERAL">일반</option>
        </select>
        {errors.classification && <span>필수 필드입니다</span>}
      </div>

      {/* 주소 */}
      <div>
        <label>주소</label>
        <input {...register('address')} />
      </div>

      {/* 전화 */}
      <div>
        <label>전화</label>
        <input
          {...register('phone', {
            pattern: /^[\d\-]+$/ // 기본 형식 검증
          })}
        />
        {errors.phone && <span>올바른 형식이 아닙니다</span>}
      </div>

      {/* 이메일 */}
      <div>
        <label>이메일</label>
        <input
          {...register('email', {
            pattern: /^[^\s@]+@[^\s@]+\.[^\s@]+$/ // 이메일 형식
          })}
        />
        {errors.email && <span>올바른 이메일 형식이 아닙니다</span>}
      </div>

      {/* 메모 */}
      <div>
        <label>메모</label>
        <textarea
          {...register('memo', { maxLength: 1000 })}
          rows={4}
        />
      </div>

      {/* 버튼 */}
      <div className="flex gap-2">
        <button type="submit" disabled={createMutation.isPending || updateMutation.isPending}>
          {isNew ? '등록' : '수정'}
        </button>
        <button type="button" onClick={() => router.back()}>취소</button>
      </div>
    </form>
  );
}
```

```typescript
'use client';

import { useCustomer } from '@/hooks/useCustomer';
import { useSession } from 'next-auth/react';
import { useState } from 'react';
import CustomerTabs from './CustomerTabs';
import { Button } from '@/components/ui/button';
import { useRouter } from 'next/navigation';

interface CustomerDetailClientProps {
  customerId: number;
}

export default function CustomerDetailClient({ customerId }: CustomerDetailClientProps) {
  const { data: customer, isLoading } = useCustomer(customerId);
  const { data: session } = useSession();
  const router = useRouter();

  if (isLoading) return <div>로딩 중...</div>;
  if (!customer) return <div>고객을 찾을 수 없습니다.</div>;

  const isAdmin = session?.user.role === 'ADMIN';

  return (
    <div className="space-y-6">
      {/* 헤더 */}
      <div className="flex justify-between items-center">
        <div>
          <h1 className="text-3xl font-bold">{customer.data.name}</h1>
          <p className="text-gray-500">{customer.data.code}</p>
        </div>
        <div className="flex gap-2">
          <Button onClick={() => router.push(`/customers/${customerId}/edit`)}>
            수정
          </Button>
          {isAdmin && (
            <Button variant="destructive" onClick={() => {/* 삭제 로직 */}}>
              삭제
            </Button>
          )}
        </div>
      </div>

      {/* 탭 */}
      <CustomerTabs customerId={customerId} customer={customer.data} />
    </div>

}
```

## 2081_13: 자동완성 검색 (CustomerSearch)

### 핵심 구현

**파일**: `src/components/features/customers/CustomerSearch.tsx`

```typescript
'use client';

import { useState, useCallback } from 'react';
import { useCustomers } from '@/hooks/useCustomers';
import { debounce } from 'lodash';
import { Combobox } from '@/components/ui/combobox';
import { Customer } from '@/types/customer';

interface CustomerSearchProps {
  onSelectCustomer: (customer: Customer) => void;
  onCreateCustomer?: (name: string) => void;
}

export default function CustomerSearch({
  onSelectCustomer,
  onCreateCustomer
}: CustomerSearchProps) {
  const [search, setSearch] = useState('');

  // Debounce 300ms
  const debouncedSearch = useCallback(
    debounce((value: string) => {
      // TanStack Query hook이 자동으로 search 파라미터 감지하여 API 호출
    }, 300),
    []
  );

  const { data } = useCustomers({ search, limit: 10 });

  return (
    <Combobox
      placeholder="고객사명으로 검색..."
      items={data?.data || []}
      getItemLabel={(item) => item.name}
      onValueChange={(value) => {
        setSearch(value);
        debouncedSearch(value);
      }}
      onSelectItem={(item) => onSelectCustomer(item)}
      onCreateItem={onCreateCustomer ? (name) => onCreateCustomer(name) : undefined}
    />
  );
}
```

```typescript
describe('Customer API', () => {
  describe('POST /api/customers', () => {
    it('should create customer with MANAGER role', async () => {
      const res = await fetch('/api/customers', {
        method: 'POST',
        body: JSON.stringify({
          name: '테스트회사',
          classification: 'END_USER'
        }),
        headers: { 'Authorization': 'Bearer manager-token' }
      });
      expect(res.status).toBe(201);
    });

    it('should reject USER role', async () => {
      const res = await fetch('/api/customers', {
        method: 'POST',
        body: JSON.stringify({
          name: '테스트회사',
          classification: 'END_USER'
        }),
        headers: { 'Authorization': 'Bearer user-token' }
      });
      expect(res.status).toBe(403);
    });
  });

  describe('DELETE /api/customers/[id]', () => {
    it('should delete customer as ADMIN only', async () => {
      const res = await fetch('/api/customers/1', {
        method: 'DELETE',
        headers: { 'Authorization': 'Bearer admin-token' }
      });
      expect(res.status).toBe(200);
    });

    it('should reject MANAGER role (Decision 1, 8)', async () => {
      const res = await fetch('/api/customers/1', {
        method: 'DELETE',
        headers: { 'Authorization': 'Bearer manager-token' }
      });
      expect(res.status).toBe(403);
    });
  });
});
```
