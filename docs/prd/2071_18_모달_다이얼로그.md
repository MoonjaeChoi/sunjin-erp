```typescript
// Generated: 2026-01-26 HH:MM:SS KST

'use client';

import { useState } from 'react';
import { ContractStatus, ContractStatusLabel, useCanTransitionStatus } from '@/hooks/useContractStatusBadge';
import { useStatusChange } from '@/hooks/useStatusChange';
import {
  Dialog,
  DialogContent,
  DialogDescription,
  DialogHeader,
  DialogTitle,
  DialogFooter
} from '@/components/ui/dialog';
import {
  Select,
  SelectContent,
  SelectItem,
  SelectTrigger,
  SelectValue
} from '@/components/ui/select';
import { Input } from '@/components/ui/input';
import { Button } from '@/components/ui/button';
import { Loader2 } from 'lucide-react';

interface StatusChangeDialogProps {
  open: boolean;
  onOpenChange: (open: boolean) => void;
  contractId: number;
  currentStatus: ContractStatus;
}

/**
 * 상태 변경 다이얼로그
 */
export default function StatusChangeDialog({
  open,
  onOpenChange,
  contractId,
  currentStatus
}: StatusChangeDialogProps) {
  const [newStatus, setNewStatus] = useState<ContractStatus | ''>('');
  const [reason, setReason] = useState('');
  const mutation = useStatusChange(contractId);

  const availableStatuses = Object.values(ContractStatus).filter(
    (status) => useCanTransitionStatus(currentStatus, status)
  );

  const handleSubmit = async () => {
    if (!newStatus || !reason) return;

    try {
      await mutation.mutateAsync({
        status: newStatus as ContractStatus,
        reason
      });
      onOpenChange(false);
      setNewStatus('');
      setReason('');
    } catch (error) {
      console.error('Failed to change status:', error);
    }
  };

  return (
    <Dialog open={open} onOpenChange={onOpenChange}>
      <DialogContent>
        <DialogHeader>
          <DialogTitle>상태 변경</DialogTitle>
          <DialogDescription>
            현재 상태: {ContractStatusLabel[currentStatus]}
          </DialogDescription>
        </DialogHeader>

        <div className="space-y-4">
          <div>
            <label className="text-sm font-medium mb-2 block">변경할 상태 *</label>
            <Select value={newStatus} onValueChange={setNewStatus}>
              <SelectTrigger>
                <SelectValue placeholder="상태 선택" />
              </SelectTrigger>
              <SelectContent>
                {availableStatuses.map((status) => (
                  <SelectItem key={status} value={status}>
                    {ContractStatusLabel[status]}
                  </SelectItem>
                ))}
              </SelectContent>
            </Select>
          </div>

          <div>
            <label className="text-sm font-medium mb-2 block">사유 *</label>
            <Input
              placeholder="변경 사유를 입력하세요"
              value={reason}
              onChange={(e) => setReason(e.target.value)}
            />
          </div>
        </div>

        <DialogFooter>
          <Button
            variant="outline"
            onClick={() => onOpenChange(false)}
            disabled={mutation.isPending}
          >
            취소
          </Button>
          <Button
            onClick={handleSubmit}
            disabled={!newStatus || !reason || mutation.isPending}
          >
            {mutation.isPending && <Loader2 className="w-4 h-4 mr-2 animate-spin" />}
            변경
          </Button>
        </DialogFooter>
      </DialogContent>
    </Dialog>
  );
}
```

```typescript
// Generated: 2026-01-26 HH:MM:SS KST

'use client';

import { useDeleteMaintenanceContract } from '@/hooks/useMaintenanceContractMutation';
import { useRouter } from 'next/navigation';
import {
  Dialog,
  DialogContent,
  DialogDescription,
  DialogHeader,
  DialogTitle,
  DialogFooter
} from '@/components/ui/dialog';
import { Button } from '@/components/ui/button';
import { AlertTriangle, Loader2 } from 'lucide-react';

interface DeleteConfirmDialogProps {
  open: boolean;
  onOpenChange: (open: boolean) => void;
  contractId: number;
  contractName: string;
}

/**
 * 삭제 확인 다이얼로그
 */
export default function DeleteConfirmDialog({
  open,
  onOpenChange,
  contractId,
  contractName
}: DeleteConfirmDialogProps) {
  const router = useRouter();
  const mutation = useDeleteMaintenanceContract();

  const handleDelete = async () => {
    try {
      await mutation.mutateAsync(contractId);
      onOpenChange(false);
      router.push('/maintenance');
    } catch (error) {
      console.error('Failed to delete contract:', error);
    }
  };

  return (
    <Dialog open={open} onOpenChange={onOpenChange}>
      <DialogContent>
        <DialogHeader>
          <DialogTitle className="flex items-center gap-2">
            <AlertTriangle className="w-5 h-5 text-red-500" />
            계약 삭제
          </DialogTitle>
          <DialogDescription>
            이 작업은 되돌릴 수 없습니다
          </DialogDescription>
        </DialogHeader>

        <div className="space-y-4">
          <p className="text-sm">
            "<strong>{contractName}</strong>" 계약을 정말 삭제하시겠습니까?
          </p>
          <p className="text-xs text-muted-foreground">
            · 관련된 모든 첨부 파일이 함께 삭제됩니다
            <br />· 변경 이력도 함께 삭제됩니다
            <br />· 이 작업은 취소할 수 없습니다
          </p>
        </div>

        <DialogFooter>
          <Button
            variant="outline"
            onClick={() => onOpenChange(false)}
            disabled={mutation.isPending}
          >
            취소
          </Button>
          <Button
            variant="destructive"
            onClick={handleDelete}
            disabled={mutation.isPending}
          >
            {mutation.isPending && <Loader2 className="w-4 h-4 mr-2 animate-spin" />}
            삭제
          </Button>
        </DialogFooter>
      </DialogContent>
    </Dialog>
  );
}
```

```typescript
// Generated: 2026-01-26 HH:MM:SS KST

'use client';

import { useRef, useState } from 'react';
import { useUploadAttachment } from '@/hooks/useAttachmentMutation';
import {
  Dialog,
  DialogContent,
  DialogDescription,
  DialogHeader,
  DialogTitle,
  DialogFooter
} from '@/components/ui/dialog';
import { Button } from '@/components/ui/button';
import { Upload, Loader2 } from 'lucide-react';

interface FileUploadDialogProps {
  open: boolean;
  onOpenChange: (open: boolean) => void;
  contractId: number;
}

/**
 * 파일 업로드 다이얼로그
 */
export default function FileUploadDialog({
  open,
  onOpenChange,
  contractId
}: FileUploadDialogProps) {
  const fileInputRef = useRef<HTMLInputElement>(null);
  const [selectedFile, setSelectedFile] = useState<File | null>(null);
  const mutation = useUploadAttachment(contractId);

  const handleFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (file) {
      // 파일 검증
      const validTypes = ['application/pdf', 'application/msword', 'application/vnd.openxmlformats-officedocument.wordprocessingml.document'];
      const maxSize = 10 * 1024 * 1024; // 10MB

      if (!validTypes.includes(file.type)) {
        alert('PDF, DOCX, DOC 파일만 업로드 가능합니다');
        return;
      }

      if (file.size > maxSize) {
        alert('파일 크기는 10MB 이하여야 합니다');
        return;
      }

      setSelectedFile(file);
    }
  };

  const handleUpload = async () => {
    if (!selectedFile) return;

    try {
      await mutation.mutateAsync(selectedFile);
      onOpenChange(false);
      setSelectedFile(null);
      if (fileInputRef.current) fileInputRef.current.value = '';
    } catch (error) {
      console.error('Failed to upload file:', error);
    }
  };

  return (
    <Dialog open={open} onOpenChange={onOpenChange}>
      <DialogContent>
        <DialogHeader>
          <DialogTitle>파일 업로드</DialogTitle>
          <DialogDescription>
            PDF, DOCX, DOC 파일 (최대 10MB)
          </DialogDescription>
        </DialogHeader>

        <div className="space-y-4">
          <div
            className="border-2 border-dashed rounded-lg p-6 text-center cursor-pointer hover:bg-muted"
            onClick={() => fileInputRef.current?.click()}
          >
            <input
              ref={fileInputRef}
              type="file"
              className="hidden"
              accept=".pdf,.doc,.docx"
              onChange={handleFileChange}
            />
            <Upload className="w-8 h-8 mx-auto mb-2 text-muted-foreground" />
            <p className="text-sm font-medium">파일을 클릭하여 선택</p>
            <p className="text-xs text-muted-foreground">
              또는 드래그해서 드롭
            </p>
          </div>

          {selectedFile && (
            <div className="p-3 border rounded-lg bg-muted">
              <p className="text-sm font-medium truncate">
                {selectedFile.name}
              </p>
              <p className="text-xs text-muted-foreground">
                {(selectedFile.size / 1024).toFixed(1)} KB
              </p>
            </div>
          )}
        </div>

        <DialogFooter>
          <Button
            variant="outline"
            onClick={() => {
              onOpenChange(false);
              setSelectedFile(null);
            }}
            disabled={mutation.isPending}
          >
            취소
          </Button>
          <Button
            onClick={handleUpload}
            disabled={!selectedFile || mutation.isPending}
          >
            {mutation.isPending && <Loader2 className="w-4 h-4 mr-2 animate-spin" />}
            업로드
          </Button>
        </DialogFooter>
      </DialogContent>
    </Dialog>
  );
}
```
