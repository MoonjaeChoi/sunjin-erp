// e2e/maintenance.spec.ts

import { test, expect } from './fixtures/auth';

test.describe('유지보수 계약 관리 전체 플로우', () => {
  test('should complete full contract lifecycle', async ({ authenticatedPage: page }) => {
    // 1. 목록 페이지로 이동
    await page.goto('/maintenance');
    await expect(page.locator('h1')).toContainText('유지보수 계약 관리');

    // 2. 신규 등록 버튼 클릭
    await page.click('button:has-text("신규 등록")');
    await page.waitForURL('/maintenance/new');

    // 3. 폼 작성
    await page.selectOption('select[name="customer_id"]', '1');
    await page.fill('input[name="contract_name"]', '테스트계약');
    await page.fill('input[name="contract_type"]', 'Maintenance');
    await page.fill('input[name="start_date"]', '2026-01-01');
    await page.fill('input[name="end_date"]', '2026-12-31');
    await page.fill('input[name="contract_amount"]', '5000000');
    await page.selectOption('select[name="assigned_employee_id"]', '1');

    // 4. 등록 클릭
    await page.click('button:has-text("등록")');
    await page.waitForURL('/maintenance/*');

    // 상세 페이지 확인
    const contractId = page.url().split('/').pop();
    await expect(page.locator('h1')).toContainText('테스트계약');

    // 5. 첨부 파일 업로드
    const fileInput = page.locator('input[type="file"]');
    await fileInput.setInputFiles('./e2e/fixtures/sample.pdf');
    await page.click('button:has-text("업로드")');
    await expect(page.locator('text=sample.pdf')).toBeVisible();

    // 6. 상태 변경
    await page.click('button:has-text("상태 변경")');
    await page.selectOption('select', '갱신예정');
    await page.fill('input[placeholder="변경 사유"]', '갱신 준비 중');
    await page.click('button:has-text("변경")');
    await expect(page.locator('text=갱신예정')).toBeVisible();

    // 7. 이력 확인
    const histories = page.locator('[data-testid="history-item"]');
    await expect(histories).toHaveCount(2); // 생성 + 상태변경

    // 8. 수정
    await page.click('button:has-text("수정")');
    await page.waitForURL('/maintenance/*/edit');
    await page.fill('input[name="notes"]', '수정된 노트');
    await page.click('button:has-text("수정")');
    await expect(page.locator('text=수정된 노트')).toBeVisible();

    // 9. 목록으로 돌아가기
    await page.goto('/maintenance');
    await expect(page.locator('text=테스트계약')).toBeVisible();
  });
});

// e2e/maintenance-list.spec.ts

import { test, expect } from './fixtures/auth';

test.describe('계약 목록 페이지', () => {
  test('should display contracts list', async ({ authenticatedPage: page }) => {
    await page.goto('/maintenance');

    // 헤더 확인
    await expect(page.locator('h1')).toContainText('유지보수 계약 관리');

    // 통계 카드 확인
    await expect(page.locator('text=활성 계약')).toBeVisible();
    await expect(page.locator('text=갱신 예정')).toBeVisible();

    // 테이블 확인
    const table = page.locator('table');
    await expect(table).toBeVisible();
  });

  test('should filter by status', async ({ authenticatedPage: page }) => {
    await page.goto('/maintenance');

    // 상태 필터 변경
    await page.selectOption('select[aria-label="상태"]', '활성');
    await page.waitForTimeout(500); // 필터 적용 대기

    // URL 변경 확인
    await expect(page).toHaveURL(/\?status=활성/);

    // 테이블 업데이트 확인
    const badges = page.locator('[aria-label="상태"]');
    const badgeTexts = await badges.allTextContents();
    badgeTexts.forEach(text => {
      expect(text).toContain('활성');
    });
  });

  test('should navigate pages', async ({ authenticatedPage: page }) => {
    await page.goto('/maintenance');

    // 다음 버튼 확인
    const nextBtn = page.locator('button:has-text("다음")');
    if (await nextBtn.isEnabled()) {
      await nextBtn.click();
      await page.waitForURL(/\?page=2/);
    }

    // 이전 버튼 클릭
    const prevBtn = page.locator('button:has-text("이전")');
    await prevBtn.click();
    await page.waitForURL(/\?page=1|^\?$/);
  });

  test('should search by contract name', async ({ authenticatedPage: page }) => {
    await page.goto('/maintenance');

    // 검색 입력
    await page.fill('input[placeholder="계약명으로 검색..."]', '테스트');
    await page.waitForTimeout(500);

    // URL 변경 확인
    await expect(page).toHaveURL(/\?search=테스트/);
  });
});

// e2e/maintenance-detail.spec.ts

import { test, expect } from './fixtures/auth';

test.describe('계약 상세 페이지', () => {
  test('should display contract details', async ({ authenticatedPage: page }) => {
    // 테스트용 계약 ID (미리 생성되어야 함)
    await page.goto('/maintenance/1');

    // 기본 정보 확인
    await expect(page.locator('text=고객')).toBeVisible();
    await expect(page.locator('text=기간')).toBeVisible();
    await expect(page.locator('text=상태')).toBeVisible();

    // 액션 버튼 확인
    await expect(page.locator('button:has-text("수정")')).toBeVisible();
    await expect(page.locator('button:has-text("삭제")')).toBeVisible();
  });

  test('should upload and download files', async ({ authenticatedPage: page }) => {
    await page.goto('/maintenance/1');

    // 파일 업로드
    const fileInput = page.locator('input[type="file"]');
    await fileInput.setInputFiles('./e2e/fixtures/sample.pdf');

    // 파일 목록 확인
    await expect(page.locator('text=sample.pdf')).toBeVisible();

    // 다운로드 버튼 클릭
    const downloadPromise = page.waitForEvent('download');
    await page.click('[aria-label="다운로드"]');
    const download = await downloadPromise;
    expect(download.suggestedFilename()).toContain('pdf');
  });

  test('should change status', async ({ authenticatedPage: page }) => {
    await page.goto('/maintenance/1');

    const currentStatus = await page.locator('[aria-label="상태"]').textContent();

    // 상태 변경 버튼 클릭
    await page.click('button:has-text("상태 변경")');

    // 다이얼로그 확인
    await expect(page.locator('text=상태 변경')).toBeVisible();

    // 새로운 상태 선택
    await page.selectOption('select', '갱신예정');
    await page.fill('textarea', '갱신 준비');
    await page.click('button:has-text("변경")');

    // 상태 업데이트 확인
    await page.waitForTimeout(1000);
    const newStatus = await page.locator('[aria-label="상태"]').textContent();
    expect(newStatus).not.toBe(currentStatus);
  });

  test('should show history', async ({ authenticatedPage: page }) => {
    await page.goto('/maintenance/1');

    // 이력 섹션 확인
    const historySection = page.locator(':has-text("변경 이력")');
    await expect(historySection).toBeVisible();

    // 이력 아이템 확인
    const historyItems = page.locator('[data-testid="history-item"]');
    const count = await historyItems.count();
    expect(count).toBeGreaterThan(0);
  });
});
