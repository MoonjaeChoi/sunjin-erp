### 테스트 범위

**1. API Route Tests (Integration-like)**

```typescript
// Mock: NextAuth session, TypeORM repository
// 검증: HTTP status, response body, RBAC, validation

describe('GET /api/tasks', () => {
  it('401: 미인증 요청');
  it('400: date_from/date_to 누락');
  it('200: 정상 조회 + 필터');
  it('RBAC: USER는 본인 업무만');
  it('RBAC: ADMIN은 전체 조회');
  it('soft delete 제외');
  it('정렬: task_date ASC, start_time ASC');
});

describe('POST /api/tasks', () => {
  it('401: 미인증');
  it('400: 필수 필드 누락 (title)');
  it('400: 잘못된 Enum 값');
  it('400: start_time > end_time');
  it('400: start_time 범위 초과 (>1439)');
  it('201: 정상 생성');
  it('employee_id = session.user.id 자동 설정');
  it('status DONE → completed_at 자동 설정');
  it('XSS sanitize (title)');
});

describe('PUT /api/tasks/[id]', () => {
  it('403: 타인 업무 수정 (USER)');
  it('200: ADMIN이 타인 업무 수정');
  it('completed_at: DONE 전환 시 자동 설정');
  it('completed_at: DONE 해제 시 null');
  it('completed_at: 클라이언트 값 무시');
});

describe('DELETE /api/tasks/[id]', () => {
  it('soft delete (deleted_at 설정)');
  it('physical delete 금지 확인');
  it('403: 타인 업무 삭제 (USER)');
});
```
