```typescript
import { ContractStatus, contractStatusBadgeVariant, ContractStatusLabel } from '@/types/maintenance';

/**
 * 계약 상태를 배지 형식으로 표시할 정보 제공
 *
 * 반환값:
 * - variant: shadcn/ui Badge 컴포넌트의 variant
 * - label: 한글 레이블
 * - icon: 상태별 아이콘 (필요시)
 */
export function useContractStatusBadge(status: ContractStatus) {
  const variant = contractStatusBadgeVariant[status];
  const label = ContractStatusLabel[status];

  // 상태별 추가 정보
  const details = {
    [ContractStatus.ACTIVE]: {
      variant: 'default' as const,
      label: '활성',
      color: 'bg-green-100 text-green-800',
      description: '현재 활성 상태의 계약입니다'
    },
    [ContractStatus.RENEWAL_PENDING]: {
      variant: 'secondary' as const,
      label: '갱신예정',
      color: 'bg-yellow-100 text-yellow-800',
      description: '갱신 준비 중인 계약입니다'
    },
    [ContractStatus.ENDED]: {
      variant: 'destructive' as const,
      label: '종료',
      color: 'bg-red-100 text-red-800',
      description: '종료된 계약입니다'
    }
  };

  return details[status];
}

/**
 * 상태 변경 가능 여부 확인
 */
export function useCanTransitionStatus(
  currentStatus: ContractStatus,
  targetStatus: ContractStatus
): boolean {
  const validTransitions: Record<ContractStatus, ContractStatus[]> = {
    [ContractStatus.ACTIVE]: [
      ContractStatus.ENDED,
      ContractStatus.RENEWAL_PENDING
    ],
    [ContractStatus.RENEWAL_PENDING]: [ContractStatus.ACTIVE],
    [ContractStatus.ENDED]: [ContractStatus.ACTIVE]
  };

  return validTransitions[currentStatus]?.includes(targetStatus) ?? false;
}
```

### 2. useContractExpiry.ts - 만료 계산

```typescript
// Generated: 2026-01-26 HH:MM:SS KST

import { useMemo } from 'react';

interface ExpiryInfo {
  isExpired: boolean;
  isExpiring30Days: boolean;
  isExpiring60Days: boolean;
  daysUntilExpiry: number;
  expiryStatus: 'expired' | 'critical' | 'warning' | 'normal';
}

/**
 * 계약 만료 여부 및 예정 정보 계산
 */
export function useContractExpiry(endDate: string): ExpiryInfo {
  return useMemo(() => {
    const today = new Date();
    const end = new Date(endDate);

    // 시간 부분 무시하고 비교 (자정 기준)
    today.setHours(0, 0, 0, 0);
    end.setHours(0, 0, 0, 0);

    const daysUntilExpiry = Math.ceil(
      (end.getTime() - today.getTime()) / (1000 * 60 * 60 * 24)
    );

    const isExpired = daysUntilExpiry < 0;
    const isExpiring30Days = daysUntilExpiry >= 0 && daysUntilExpiry <= 30;
    const isExpiring60Days = daysUntilExpiry >= 0 && daysUntilExpiry <= 60;

    let expiryStatus: ExpiryInfo['expiryStatus'] = 'normal';
    if (isExpired) expiryStatus = 'expired';
    else if (isExpiring30Days) expiryStatus = 'critical';
    else if (isExpiring60Days) expiryStatus = 'warning';

    return {
      isExpired,
      isExpiring30Days,
      isExpiring60Days,
      daysUntilExpiry,
      expiryStatus
    };
  }, [endDate]);
}

/**
 * 만료 상태 배지 렌더링 정보
 */
export function useExpiryBadge(endDate: string) {
  const expiry = useContractExpiry(endDate);

  const badgeInfo = {
    expired: { color: 'bg-red-100 text-red-800', label: '만료됨' },
    critical: { color: 'bg-orange-100 text-orange-800', label: '30일 내 만료' },
    warning: { color: 'bg-yellow-100 text-yellow-800', label: '60일 내 만료' },
    normal: { color: 'bg-gray-100 text-gray-800', label: '정상' }
  };

  return {
    ...expiry,
    badgeColor: badgeInfo[expiry.expiryStatus].color,
    badgeLabel: badgeInfo[expiry.expiryStatus].label,
    displayText: expiry.isExpired
      ? '만료됨'
      : `${expiry.daysUntilExpiry}일 남음`
  };
}
```

### 3. useMaintenanceFilter.ts - 필터 상태 + URL 동기화

```typescript
// Generated: 2026-01-26 HH:MM:SS KST

import { useCallback, useEffect, useState } from 'react';
import { useRouter, useSearchParams } from 'next/navigation';
import { MaintenanceListFilters, ContractStatus } from '@/types/maintenance';

/**
 * 필터 상태를 URL 검색 파라미터와 동기화
 *
 * 사용자가 필터를 변경하면 URL 업데이트 (뒤로 가기 지원)
 * URL이 변경되면 필터 상태 자동 업데이트
 */
export function useMaintenanceFilter() {
  const router = useRouter();
  const searchParams = useSearchParams();
  const [filters, setFilters] = useState<MaintenanceListFilters>({
    status: undefined,
    customerId: undefined,
    assignedEmployeeId: undefined,
    contractNameSearch: '',
    sortBy: 'created_at',
    sortOrder: 'desc',
    page: 1,
    limit: 20
  });

  // URL → 필터 상태 (초기 로드 시)
  useEffect(() => {
    const status = searchParams.get('status') as ContractStatus | null;
    const customerId = searchParams.get('customerId');
    const search = searchParams.get('search');
    const page = searchParams.get('page');

    setFilters((prev) => ({
      ...prev,
      status: status || undefined,
      customerId: customerId ? parseInt(customerId) : undefined,
      contractNameSearch: search || '',
      page: page ? parseInt(page) : 1
    }));
  }, [searchParams]);

  // 필터 변경 → URL 업데이트
  const updateFilters = useCallback(
    (newFilters: Partial<MaintenanceListFilters>) => {
      setFilters((prev) => ({ ...prev, ...newFilters }));

      const params = new URLSearchParams();
      const merged = { ...filters, ...newFilters };

      if (merged.status) params.append('status', merged.status);
      if (merged.customerId) params.append('customerId', merged.customerId.toString());
      if (merged.contractNameSearch) params.append('search', merged.contractNameSearch);
      if (merged.page && merged.page > 1) params.append('page', merged.page.toString());

      router.push(`?${params.toString()}`);
    },
    [filters, router]
  );

  // 필터 초기화
  const resetFilters = useCallback(() => {
    setFilters({
      status: undefined,
      customerId: undefined,
      assignedEmployeeId: undefined,
      contractNameSearch: '',
      sortBy: 'created_at',
      sortOrder: 'desc',
      page: 1,
      limit: 20
    });
    router.push('');
  }, [router]);

  return { filters, updateFilters, resetFilters };
}

/**
 * 단순 필터 (URL 동기화 없이)
 */
export function useLocalFilter(
  initialFilters?: Partial<MaintenanceListFilters>
) {
  const [filters, setFilters] = useState<MaintenanceListFilters>({
    ...initialFilters,
    page: 1,
    limit: 20
  } as MaintenanceListFilters);

  const updateFilters = useCallback(
    (newFilters: Partial<MaintenanceListFilters>) => {
      setFilters((prev) => ({ ...prev, ...newFilters }));
    },
    []
  );

  const resetFilters = useCallback(() => {
    setFilters({
      ...initialFilters,
      page: 1,
      limit: 20
    } as MaintenanceListFilters);
  }, [initialFilters]);

  return { filters, updateFilters, resetFilters };
}
```

```typescript
// Generated: 2026-01-26 HH:MM:SS KST

import { useCallback, useState } from 'react';

interface PaginationState {
  page: number;
  limit: number;
  total: number;
  hasNextPage: boolean;
}

/**
 * 페이지네이션 상태 관리
 */
export function usePagination(initialLimit: number = 20) {
  const [pagination, setPagination] = useState<PaginationState>({
    page: 1,
    limit: initialLimit,
    total: 0,
    hasNextPage: false
  });

  // 서버 응답으로부터 상태 업데이트
  const updatePagination = useCallback(
    (serverPagination: { page: number; limit: number; total: number }) => {
      setPagination((prev) => ({
        ...prev,
        ...serverPagination,
        hasNextPage:
          serverPagination.page * serverPagination.limit < serverPagination.total
      }));
    },
    []
  );

  // 다음 페이지로 이동
  const nextPage = useCallback(() => {
    setPagination((prev) =>
      prev.hasNextPage ? { ...prev, page: prev.page + 1 } : prev
    );
  }, []);

  // 이전 페이지로 이동
  const prevPage = useCallback(() => {
    setPagination((prev) =>
      prev.page > 1 ? { ...prev, page: prev.page - 1 } : prev
    );
  }, []);

  // 특정 페이지로 이동
  const goToPage = useCallback((page: number) => {
    setPagination((prev) => ({
      ...prev,
      page: Math.max(1, Math.min(page, Math.ceil(prev.total / prev.limit)))
    }));
  }, []);

  // 페이지 크기 변경
  const setPageSize = useCallback((limit: number) => {
    setPagination((prev) => ({
      ...prev,
      limit,
      page: 1 // 페이지 크기 변경 시 첫 페이지로
    }));
  }, []);

  // 계산된 정보
  const pageCount = Math.ceil(pagination.total / pagination.limit);
  const startItem = (pagination.page - 1) * pagination.limit + 1;
  const endItem = Math.min(pagination.page * pagination.limit, pagination.total);

  return {
    ...pagination,
    pageCount,
    startItem,
    endItem,
    updatePagination,
    nextPage,
    prevPage,
    goToPage,
    setPageSize
  };
}
```

```typescript
import { useMemo } from 'react';
import { MaintenanceFormInput, maintenanceFormSchema } from '@/types/maintenance';

/**
 * 폼 입력 검증 로직
 */
export function useFormValidation(
  data: Partial<MaintenanceFormInput>
) {
  const errors = useMemo(() => {
    const result = maintenanceFormSchema.safeParse(data);
    return result.success ? {} : result.error.flatten().fieldErrors;
  }, [data]);

  const isValid = Object.keys(errors).length === 0;

  return { errors, isValid };
}

/**
 * 날짜 범위 검증
 */
export function useDateRangeValidation(
  startDate?: string,
  endDate?: string
): { isValid: boolean; error?: string } {
  return useMemo(() => {
    if (!startDate || !endDate) return { isValid: true };

    const start = new Date(startDate);
    const end = new Date(endDate);

    if (start > end) {
      return {
        isValid: false,
        error: '시작일이 종료일보다 클 수 없습니다'
      };
    }

    return { isValid: true };
  }, [startDate, endDate]);
}

/**
 * 계약 금액 검증
 */
export function useAmountValidation(amount?: number): {
  isValid: boolean;
  error?: string;
} {
  return useMemo(() => {
    if (amount === undefined || amount === null) return { isValid: true };

    if (amount < 0) return { isValid: false, error: '금액은 0 이상이어야 합니다' };
    if (amount > 999999999) return { isValid: false, error: '너무 큰 금액입니다' };

    return { isValid: true };
  }, [amount]);
}
```
