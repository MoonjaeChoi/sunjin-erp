import { NextRequest } from 'next/server';

// 세션 모킹
export const mockSession = {
  admin: {
    user: { id: 1, name: '관리자', email: 'admin@test.com', role: 'ADMIN', department: 1 },
  },
  manager: {
    user: { id: 2, name: '매니저', email: 'manager@test.com', role: 'MANAGER', department: 1 },
  },
  user: {
    user: { id: 3, name: '일반사용자', email: 'user@test.com', role: 'USER', department: 2 },
  },
};

// NextRequest 생성 헬퍼
export function createRequest(
  method: string,
  url: string,
  body?: object,
  searchParams?: Record<string, string>
): NextRequest {
  const baseUrl = 'http://localhost:3000';
  const urlObj = new URL(url, baseUrl);

  if (searchParams) {
    Object.entries(searchParams).forEach(([key, value]) => {
      urlObj.searchParams.set(key, value);
    });
  }

  return new NextRequest(urlObj, {
    method,
    body: body ? JSON.stringify(body) : undefined,
    headers: body ? { 'Content-Type': 'application/json' } : {},
  });
}

// 응답 파싱 헬퍼
export async function parseResponse<T>(response: Response): Promise<T> {
  return response.json() as Promise<T>;
}

    vi.clearAllMocks();

  it('인증되지 않은 사용자는 401 반환', async () => {
    vi.mocked(getServerSession).mockResolvedValue(null);

    const response = await GET(request);

describe('GET /api/notices/[id]', () => {

  });

      CONTENT: '테스트 내용',

    });

    const response = await GET(request, { params: mockParams });

    expect(response.status).toBe(200);
    expect(executeUpdate).toHaveBeenCalled(); // 조회수 증가 확인
  });
});

describe('PUT /api/notices/[id]', () => {
  beforeEach(() => {
    vi.clearAllMocks();
  });

  it('작성자가 아닌 USER는 403 반환', async () => {
    vi.mocked(getServerSession).mockResolvedValue(mockSession.user);
    vi.mocked(executeQuerySingle).mockResolvedValue({
      ID: 1,
      AUTHOR_ID: 999, // 다른 사용자

    const request = createRequest('PUT', '/api/notices/1', {
      title: '수정된 제목입니다',
      content: '수정된 내용입니다. 최소 10자.',

    expect(response.status).toBe(403);

      title: '수정된 제목입니다',

      title: '수정된 제목입니다',
      content: '수정된 내용입니다. 최소 10자.',

  it('트리 구조로 댓글 반환', async () => {

    vi.mocked(executeQuery).mockResolvedValue([
      { ID: 1, CONTENT: '댓글1', PARENT_COMMENT_ID: null },
      { ID: 2, CONTENT: '답글1', PARENT_COMMENT_ID: 1 },
      { ID: 3, CONTENT: '댓글2', PARENT_COMMENT_ID: null },
    ]);

    const request = createRequest('GET', '/api/notices/1/comments');
    const response = await GET(request, { params: mockParams });

    expect(response.status).toBe(200);
    const data = await parseResponse(response);
    // 최상위 댓글만 data.data에 포함, 답글은 replies 배열에
    expect(data.data.length).toBe(2);
    expect(data.data[0].replies.length).toBe(1);
  });
});

describe('POST /api/notices/[id]/comments', () => {
  beforeEach(() => {
    vi.clearAllMocks();
  });

  it('댓글 작성 성공', async () => {
    vi.mocked(getServerSession).mockResolvedValue(mockSession.user);
    vi.mocked(executeQuerySingle).mockResolvedValue({ ID: 1 }); // notice exists
    vi.mocked(executeUpdate).mockResolvedValue({ rowsAffected: 1, lastInsertId: 1 });

    const request = createRequest('POST', '/api/notices/1/comments', {
      content: '테스트 댓글입니다',
    });
    const response = await POST(request, { params: mockParams });

    expect(response.status).toBe(201);
  });

  it('답글 작성 성공', async () => {
    vi.mocked(getServerSession).mockResolvedValue(mockSession.user);
    vi.mocked(executeQuerySingle)
      .mockResolvedValueOnce({ ID: 1 }) // notice exists
      .mockResolvedValueOnce({ ID: 1, PARENT_COMMENT_ID: null }); // parent comment (depth 0)
    vi.mocked(executeUpdate).mockResolvedValue({ rowsAffected: 1, lastInsertId: 2 });

    const request = createRequest('POST', '/api/notices/1/comments', {
      content: '테스트 답글입니다',
      parentCommentId: 1,
    });
    const response = await POST(request, { params: mockParams });

    expect(response.status).toBe(201);
  });

  it('깊이 2단계 초과 답글 시 400 반환 (Decision #3)', async () => {
    vi.mocked(getServerSession).mockResolvedValue(mockSession.user);
    vi.mocked(executeQuerySingle)
      .mockResolvedValueOnce({ ID: 1 }) // notice exists
      .mockResolvedValueOnce({ ID: 2, PARENT_COMMENT_ID: 1 }); // parent is already a reply (depth 1)

    const request = createRequest('POST', '/api/notices/1/comments', {
      content: '깊은 답글',
      parentCommentId: 2, // 이미 답글인 댓글에 답글 시도
    });
    const response = await POST(request, { params: mockParams });

    expect(response.status).toBe(400);
  });

  it('댓글 내용 1000자 초과 시 400 반환', async () => {
    vi.mocked(getServerSession).mockResolvedValue(mockSession.user);
    vi.mocked(executeQuerySingle).mockResolvedValue({ ID: 1 });

    const request = createRequest('POST', '/api/notices/1/comments', {
      content: 'a'.repeat(1001),
    });
    const response = await POST(request, { params: mockParams });

    expect(response.status).toBe(400);
  });
});
