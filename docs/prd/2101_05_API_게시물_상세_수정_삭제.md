import { NextRequest, NextResponse } from 'next/server';
import { getServerSession } from 'next-auth';
import { authOptions } from '@/lib/auth';
import { executeQuery, executeUpdate, executeTransaction } from '@/lib/db-direct';
import type { NoticeDetail, NoticeType, NoticeAttachment } from '@/types/notice';

export const dynamic = 'force-dynamic';

const VALID_TYPES: NoticeType[] = ['공지', '공유', '지시', '건의', '자유'];

interface RouteParams {
  params: { id: string };
}

// ============================================================
// GET /api/notices/[id] - 상세 조회 + 조회수 증가
// ============================================================

export async function GET(req: NextRequest, { params }: RouteParams): Promise<NextResponse> {
  try {
    // 1. 인증 검증
    const session = await getServerSession(authOptions);
    if (!session?.user) {
      return NextResponse.json({ message: 'Unauthorized' }, { status: 401 });
    }

    const noticeId = parseInt(params.id, 10);
    if (isNaN(noticeId)) {
      return NextResponse.json({ message: 'Invalid notice ID' }, { status: 400 });
    }

    const user = session.user as any;

    // 2. 게시물 조회
    const noticeResult = await executeQuery<any>(
      `SELECT
        n.ID,
        n.TITLE,
        n.CONTENT,
        n.TYPE,
        n.AUTHOR_ID,
        e_author.NAME AS AUTHOR_NAME,
        n.VIEW_COUNT,
        n.UPDATED_BY_ID,
        e_updated.NAME AS UPDATED_BY_NAME,
        n.DELETION_REASON,
        n."created_at" AS CREATED_AT,
        n."updated_at" AS UPDATED_AT,
        n."deleted_at" AS DELETED_AT
      FROM NOTICE n
      LEFT JOIN EMPLOYEE e_author ON e_author.ID = n.AUTHOR_ID
      LEFT JOIN EMPLOYEE e_updated ON e_updated.ID = n.UPDATED_BY_ID
      WHERE n.ID = :noticeId AND n."deleted_at" IS NULL`,
      { noticeId }
    );

    if (noticeResult.rows.length === 0) {
      return NextResponse.json({ message: 'Notice not found' }, { status: 404 });
    }

    const notice = noticeResult.rows[0];

    // 3. 첨부파일 조회
    const attachmentResult = await executeQuery<any>(
      `SELECT ID, NOTICE_ID, FILENAME, FILE_PATH, FILE_SIZE, MIME_TYPE, UPLOADED_AT
       FROM NOTICE_ATTACHMENT
       WHERE NOTICE_ID = :noticeId AND "deleted_at" IS NULL
       ORDER BY UPLOADED_AT ASC`,
      { noticeId }
    );

    // 4. 댓글 수 조회
    const commentCountResult = await executeQuery<any>(
      `SELECT COUNT(*) AS CNT FROM COMMENT WHERE NOTICE_ID = :noticeId AND "deleted_at" IS NULL`,
      { noticeId }
    );

    // 5. 조회수 증가 (세션 기반 중복 제외)
    // 세션 스토리지에서 viewed_notices 확인 (서버사이드에서는 쿠키 사용)
    const viewedCookie = req.cookies.get(`viewed_notice_${noticeId}`);

    if (!viewedCookie) {
      // 비동기로 조회수 증가 (응답 지연 없음)
      executeUpdate(
        `UPDATE NOTICE SET VIEW_COUNT = VIEW_COUNT + 1 WHERE ID = :noticeId`,
        { noticeId }
      ).catch((err) => console.error('View count update failed:', err));
    }

    // 6. 응답 구성
    const attachments: NoticeAttachment[] = attachmentResult.rows.map((row: any) => ({
      id: row.ID,
      noticeId: row.NOTICE_ID,
      filename: row.FILENAME,
      filePath: row.FILE_PATH,
      fileSize: row.FILE_SIZE,
      mimeType: row.MIME_TYPE,
      uploadedAt: row.UPLOADED_AT?.toISOString() || '',
    }));

    const response: NoticeDetail = {
      id: notice.ID,
      title: notice.TITLE,
      content: notice.CONTENT,
      type: notice.TYPE as NoticeType,
      authorId: notice.AUTHOR_ID,
      authorName: notice.AUTHOR_NAME || '',
      viewCount: (notice.VIEW_COUNT || 0) + (viewedCookie ? 0 : 1), // 증가된 값 반영
      updatedById: notice.UPDATED_BY_ID || undefined,
      updatedByName: notice.UPDATED_BY_NAME || undefined,
      createdAt: notice.CREATED_AT?.toISOString() || '',
      updatedAt: notice.UPDATED_AT?.toISOString() || '',
      attachments,
      commentCount: parseInt(commentCountResult.rows[0]?.CNT || '0', 10),
    };

    // 7. 조회수 쿠키 설정 (응답 헤더)
    const res = NextResponse.json({ data: response });
    if (!viewedCookie) {
      res.cookies.set(`viewed_notice_${noticeId}`, '1', {
        maxAge: 60 * 60, // 1시간
        httpOnly: true,
        sameSite: 'lax',
      });
    }

    return res;
  } catch (error) {
    console.error('GET /api/notices/[id] error:', error);
    return NextResponse.json({ message: 'Internal server error' }, { status: 500 });
  }
}

// PUT /api/notices/[id] - 게시물 수정
// ============================================================

export async function PUT(req: NextRequest, { params }: RouteParams): Promise<NextResponse> {
  try {
    // 1. 인증 검증
    const session = await getServerSession(authOptions);
    if (!session?.user) {
      return NextResponse.json({ message: 'Unauthorized' }, { status: 401 });
    }

    const noticeId = parseInt(params.id, 10);
    if (isNaN(noticeId)) {
      return NextResponse.json({ message: 'Invalid notice ID' }, { status: 400 });
    }

    const user = session.user as any;
    const userId = user.id;
    const userRole = user.role;

    // 2. 게시물 조회 및 권한 확인
    const noticeResult = await executeQuery<any>(
      `SELECT ID, AUTHOR_ID FROM NOTICE WHERE ID = :noticeId AND "deleted_at" IS NULL`,
      { noticeId }
    );

    if (noticeResult.rows.length === 0) {
      return NextResponse.json({ message: 'Notice not found' }, { status: 404 });
    }

    const notice = noticeResult.rows[0];

    // 자신의 게시물 또는 ADMIN만 수정 가능
    if (notice.AUTHOR_ID !== userId && userRole !== 'ADMIN') {
      return NextResponse.json({ message: 'Forbidden' }, { status: 403 });
    }

    // 3. 요청 본문 파싱
    const body = await req.json();
    const { title, content, type } = body;

    // 4. 필드 검증
    const errors: Record<string, string> = {};

    if (title !== undefined) {
      if (title.length < 5 || title.length > 200) {
        errors.title = '제목은 5~200자 사이여야 합니다';
      }
    }

    if (content !== undefined) {
      if (content.length < 10 || content.length > 10000) {
        errors.content = '내용은 10~10,000자 사이여야 합니다';
      }
    }

    if (type !== undefined && !VALID_TYPES.includes(type)) {
      errors.type = '유효하지 않은 게시판 유형입니다';
    }

    if (Object.keys(errors).length > 0) {
      return NextResponse.json({ message: 'Validation failed', errors }, { status: 400 });
    }

    // 5. 업데이트 쿼리 구성
    const updateFields: string[] = [];
    const updateParams: Record<string, any> = { noticeId, updatedById: userId };

    if (title !== undefined) {
      updateFields.push('TITLE = :title');
      updateParams.title = title;
    }

    if (content !== undefined) {
      updateFields.push('CONTENT = :content');
      updateParams.content = content;
    }

    if (type !== undefined) {
      updateFields.push('TYPE = :type');
      updateParams.type = type;
    }

    updateFields.push('UPDATED_BY_ID = :updatedById');
    updateFields.push('"updated_at" = SYSTIMESTAMP');

    // 6. 업데이트 실행
    await executeUpdate(
      `UPDATE NOTICE SET ${updateFields.join(', ')} WHERE ID = :noticeId`,
      updateParams
    );

    // 7. 업데이트된 게시물 조회
    const updatedResult = await executeQuery<any>(
      `SELECT ID, TITLE, CONTENT, TYPE, AUTHOR_ID, VIEW_COUNT, UPDATED_BY_ID,
              "created_at" AS CREATED_AT, "updated_at" AS UPDATED_AT
       FROM NOTICE WHERE ID = :noticeId`,
      { noticeId }
    );

    const updated = updatedResult.rows[0];

    return NextResponse.json({
      message: '게시물이 수정되었습니다',
      data: {
        id: updated.ID,
        title: updated.TITLE,
        content: updated.CONTENT,
        type: updated.TYPE,
        authorId: updated.AUTHOR_ID,
        viewCount: updated.VIEW_COUNT,
        updatedById: updated.UPDATED_BY_ID,
        createdAt: updated.CREATED_AT?.toISOString() || '',
        updatedAt: updated.UPDATED_AT?.toISOString() || '',
      },
    });
  } catch (error) {
    console.error('PUT /api/notices/[id] error:', error);
    return NextResponse.json({ message: 'Internal server error' }, { status: 500 });
  }
}

// DELETE /api/notices/[id] - 게시물 삭제 (Cascade Soft Delete)
// ============================================================

export async function DELETE(req: NextRequest, { params }: RouteParams): Promise<NextResponse> {
  try {
    // 1. 인증 검증
    const session = await getServerSession(authOptions);
    if (!session?.user) {
      return NextResponse.json({ message: 'Unauthorized' }, { status: 401 });
    }

    const noticeId = parseInt(params.id, 10);
    if (isNaN(noticeId)) {
      return NextResponse.json({ message: 'Invalid notice ID' }, { status: 400 });
    }

    const user = session.user as any;
    const userId = user.id;
    const userRole = user.role;

    // 2. 게시물 조회 및 권한 확인
    const noticeResult = await executeQuery<any>(
      `SELECT ID, AUTHOR_ID FROM NOTICE WHERE ID = :noticeId AND "deleted_at" IS NULL`,
      { noticeId }
    );

    if (noticeResult.rows.length === 0) {
      return NextResponse.json({ message: 'Notice not found' }, { status: 404 });
    }

    const notice = noticeResult.rows[0];

    // MANAGER는 자신의 게시물만, ADMIN은 모든 게시물 삭제 가능
    if (userRole !== 'ADMIN' && notice.AUTHOR_ID !== userId) {
      return NextResponse.json({ message: 'Forbidden' }, { status: 403 });
    }

    // 3. 삭제 사유 (선택)
    let deletionReason: string | null = null;
    try {
      const body = await req.json();
      deletionReason = body.reason || null;
    } catch {
      // body가 없어도 OK
    }

    // 4. Cascade Soft Delete (트랜잭션)
    // PRD Decision #1: Notice 삭제 시 Comment, Attachment 연쇄 soft delete
    await executeTransaction([
      // 4.1 댓글 soft delete
      {
        query: `UPDATE COMMENT SET "deleted_at" = SYSTIMESTAMP
                WHERE NOTICE_ID = :noticeId AND "deleted_at" IS NULL`,
        params: { noticeId },
      },
      // 4.2 첨부파일 soft delete
      {
        query: `UPDATE NOTICE_ATTACHMENT SET "deleted_at" = SYSTIMESTAMP
                WHERE NOTICE_ID = :noticeId AND "deleted_at" IS NULL`,
        params: { noticeId },
      },
      // 4.3 게시물 soft delete
      {
        query: `UPDATE NOTICE SET "deleted_at" = SYSTIMESTAMP,
                DELETION_REASON = :deletionReason
                WHERE ID = :noticeId`,
        params: { noticeId, deletionReason },
      },
    ]);

    return NextResponse.json({ message: '게시물이 삭제되었습니다' });
  } catch (error) {
    console.error('DELETE /api/notices/[id] error:', error);
    return NextResponse.json({ message: 'Internal server error' }, { status: 500 });
  }
}
