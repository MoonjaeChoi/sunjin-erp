```typescript
// Generated: 2026-01-28 15:30:00 KST

import { NextRequest, NextResponse } from 'next/server';
import { getServerSession } from 'next-auth';
import { authOptions } from '@/lib/auth';
import { executeQuerySingle } from '@/lib/db-direct';
import { readFile } from 'fs/promises';
import { existsSync } from 'fs';
import path from 'path';

export const dynamic = 'force-dynamic';

interface RouteParams {
  params: { id: string; attachmentId: string };
}

// ============================================================
// GET /api/notices/[id]/attachments/[attachmentId]/download
// ============================================================

export async function GET(req: NextRequest, { params }: RouteParams): Promise<NextResponse> {
  try {
    // 1. 인증 검증
    const session = await getServerSession(authOptions);
    if (!session?.user) {
      return NextResponse.json({ message: 'Unauthorized' }, { status: 401 });
    }

    const noticeId = parseInt(params.id, 10);
    const attachmentId = parseInt(params.attachmentId, 10);

    if (isNaN(noticeId) || isNaN(attachmentId)) {
      return NextResponse.json({ message: 'Invalid ID' }, { status: 400 });
    }

    // 2. 첨부파일 조회 (게시물도 함께 확인)
    const attachment = await executeQuerySingle<any>(
      `SELECT a.ID, a.NOTICE_ID, a.FILENAME, a.FILE_PATH, a.FILE_SIZE, a.MIME_TYPE,
              a."deleted_at" AS ATTACHMENT_DELETED,
              n."deleted_at" AS NOTICE_DELETED
       FROM NOTICE_ATTACHMENT a
       JOIN NOTICE n ON n.ID = a.NOTICE_ID
       WHERE a.ID = :attachmentId AND a.NOTICE_ID = :noticeId`,
      { attachmentId, noticeId }
    );

    if (!attachment) {
      return NextResponse.json({ message: 'Attachment not found' }, { status: 404 });
    }

    // 3. 삭제 여부 확인
    if (attachment.ATTACHMENT_DELETED || attachment.NOTICE_DELETED) {
      return NextResponse.json({ message: 'Attachment not found' }, { status: 404 });
    }

    // 4. 파일 존재 확인
    const filePath = path.join(process.cwd(), 'public', attachment.FILE_PATH);

    if (!existsSync(filePath)) {
      console.error(`File not found on disk: ${filePath}`);
      return NextResponse.json({ message: 'File not found on disk' }, { status: 404 });
    }

    // 5. 파일 읽기
    const fileBuffer = await readFile(filePath);

    // 6. 응답 헤더 설정
    // 파일명 인코딩 (한글 지원)
    const encodedFilename = encodeURIComponent(attachment.FILENAME);

    const headers = new Headers();
    headers.set('Content-Type', attachment.MIME_TYPE);
    headers.set('Content-Length', String(attachment.FILE_SIZE));
    headers.set(
      'Content-Disposition',
      `attachment; filename="${encodedFilename}"; filename*=UTF-8''${encodedFilename}`
    );
    headers.set('Cache-Control', 'private, max-age=3600');

    return new NextResponse(fileBuffer, {
      status: 200,
      headers,
    });
  } catch (error) {
    console.error('GET /api/notices/[id]/attachments/[attachmentId]/download error:', error);
    return NextResponse.json({ message: 'Internal server error' }, { status: 500 });
  }
}
```

## API 스펙

### GET /api/notices/[id]/attachments/[attachmentId]/download

**응답**:
- 200: 파일 바이너리 (Content-Disposition: attachment)
- 401: 인증 필요
- 404: 첨부파일 없음 또는 삭제됨

**응답 헤더**:
| 헤더 | 값 |
|------|-----|
| Content-Type | 파일의 MIME 타입 |
| Content-Length | 파일 크기 |
| Content-Disposition | `attachment; filename="..."` |
| Cache-Control | `private, max-age=3600` |
