**1. calendar-views.spec.ts — 캘린더 뷰 전환**

```typescript
test.describe('Calendar Views', () => {
  test.beforeEach(async ({ page }) => {
    // 인증된 상태로 /dashboard 접근
    await loginAs(page, 'user');
    await page.goto('/dashboard');
  });

  test('should display month view by default', async ({ page }) => {
    await expect(page.getByRole('tab', { name: '월' })).toHaveAttribute('data-state', 'active');
    await expect(page.locator('.grid-cols-7')).toBeVisible();
  });

  test('should switch to week view', async ({ page }) => {
    await page.getByRole('tab', { name: '주' }).click();
    await expect(page).toHaveURL(/view=week/);
  });

  test('should switch to day view', async ({ page }) => {
    await page.getByRole('tab', { name: '일' }).click();
    await expect(page).toHaveURL(/view=day/);
  });

  test('should navigate to next month', async ({ page }) => {
    const currentHeader = await page.locator('h2').textContent();
    await page.getByRole('button', { name: /다음|>/ }).click();
    await expect(page.locator('h2')).not.toHaveText(currentHeader!);
  });

  test('should return to today', async ({ page }) => {
    await page.getByRole('button', { name: /다음|>/ }).click();
    await page.getByRole('button', { name: '오늘' }).click();
    const today = new Date().toISOString().split('T')[0];
    await expect(page).toHaveURL(new RegExp(`date=${today}`));
  });

  test('should maintain view state in URL', async ({ page }) => {
    await page.getByRole('tab', { name: '주' }).click();
    await page.reload();
    await expect(page.getByRole('tab', { name: '주' })).toHaveAttribute('data-state', 'active');
  });
});
```

**2. task-crud.spec.ts — 업무 CRUD 전체 플로우**

test.describe('Task CRUD', () => {
  test.beforeEach(async ({ page }) => {
    await loginAs(page, 'user');
    await page.goto('/dashboard');
  });

  test('should create a new task', async ({ page }) => {
    // 날짜 셀 더블클릭 또는 + 버튼
    await page.locator('[data-date="2026-01-24"]').dblclick();

    // 폼 채우기
    await page.getByLabel('제목').fill('테스트 업무');
    await page.getByLabel('업무 유형').click();
    await page.getByRole('option', { name: '문서작성' }).click();
    await page.getByLabel('근무 형태').click();
    await page.getByRole('option', { name: '내근' }).click();

    // 저장
    await page.getByRole('button', { name: '등록' }).click();

    // 결과 확인
    await expect(page.getByText('테스트 업무')).toBeVisible();
  });

  test('should edit an existing task', async ({ page }) => {
    // 기존 업무 클릭
    await page.getByText('테스트 업무').click();

    // 제목 수정
    await page.getByLabel('제목').clear();
    await page.getByLabel('제목').fill('수정된 업무');

    // 저장
    await page.getByRole('button', { name: '수정' }).click();

    // 결과 확인
    await expect(page.getByText('수정된 업무')).toBeVisible();
  });

  test('should change task status', async ({ page }) => {
    // 날짜 클릭하여 DayDetailPanel 표시
    await page.locator('[data-date="2026-01-24"]').click();

    // 상태 배지 클릭
    await page.getByText('준비').click();
    await page.getByRole('option', { name: '진행' }).click();

    // 결과 확인
    await expect(page.getByText('진행')).toBeVisible();
  });

  test('should delete a task with confirmation', async ({ page }) => {
    // 업무 클릭하여 수정 폼 열기
    await page.getByText('수정된 업무').click();

    // 삭제 버튼
    await page.getByRole('button', { name: '삭제' }).click();

    // 확인 다이얼로그
    await page.getByRole('button', { name: '확인' }).click();

    // 결과 확인 (목록에서 사라짐)
    await expect(page.getByText('수정된 업무')).not.toBeVisible();
  });

// e2e/helpers/auth.ts
async function loginAs(page: Page, role: 'user' | 'manager' | 'admin') {
  // NextAuth 세션 쿠키 설정 또는 로그인 페이지 경유
  await page.goto('/login');
  await page.getByLabel('아이디').fill(testUsers[role].username);
  await page.getByLabel('비밀번호').fill(testUsers[role].password);
  await page.getByRole('button', { name: '로그인' }).click();
  await page.waitForURL('/dashboard');
}
