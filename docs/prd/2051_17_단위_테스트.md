// src/__tests__/api/issues.test.ts

import { GET } from '@/app/api/issues/route';
import { NextRequest } from 'next/server';
import { getServerSession } from 'next-auth/next';

jest.mock('next-auth/next');

describe('GET /api/issues', () => {
  beforeEach(() => {
    jest.clearAllMocks();
  });

  it('should return 401 when not authenticated', async () => {
    (getServerSession as jest.Mock).mockResolvedValue(null);

    const req = new NextRequest('http://localhost:3000/api/issues');
    const res = await GET(req);

    expect(res.status).toBe(401);
  });

  it('ADMIN should see all issues', async () => {
    (getServerSession as jest.Mock).mockResolvedValue({
      user: { id: 1, role: 'ADMIN', department_id: 1 },
    });

    const req = new NextRequest('http://localhost:3000/api/issues');
    const res = await GET(req);

    expect(res.status).toBe(200);
    const data = await res.json();
    expect(data.data).toBeDefined();
  });

  it('MANAGER should see only same department issues', async () => {
    (getServerSession as jest.Mock).mockResolvedValue({
      user: { id: 1, role: 'MANAGER', department_id: 2 },
    });

    const req = new NextRequest('http://localhost:3000/api/issues');
    const res = await GET(req);

    expect(res.status).toBe(200);
  });

  it('should filter by customer_id', async () => {
    (getServerSession as jest.Mock).mockResolvedValue({
      user: { id: 1, role: 'ADMIN', department_id: 1 },
    });

    const req = new NextRequest('http://localhost:3000/api/issues?customer_id=5');
    const res = await GET(req);

    const data = await res.json();
    expect(data.data.every((issue: any) => issue.customer_id === 5)).toBe(true);
  });

  it('should handle pagination', async () => {
    (getServerSession as jest.Mock).mockResolvedValue({
      user: { id: 1, role: 'ADMIN', department_id: 1 },
    });

    const req = new NextRequest('http://localhost:3000/api/issues?page=2&page_size=10');
    const res = await GET(req);

    const data = await res.json();
    expect(data.pagination.page).toBe(2);
    expect(data.pagination.page_size).toBe(10);
  });
});

describe('POST /api/issues', () => {
  it('should create issue with is_public=false', async () => {
    (getServerSession as jest.Mock).mockResolvedValue({
      user: { id: 1, role: 'USER', department_id: 1 },
    });

    const req = new NextRequest('http://localhost:3000/api/issues', {
      method: 'POST',
      body: JSON.stringify({
        customer_id: 1,
        title: 'Test Issue',
        severity: 'CRITICAL',
        description: 'Test description with enough characters',
      }),
    });

    const res = await POST(req);
    expect(res.status).toBe(201);

    const data = await res.json();
    expect(data.data.is_public).toBe(0);
    expect(data.data.status).toBe('INTAKE');
  });

  it('should validate required fields', async () => {
    (getServerSession as jest.Mock).mockResolvedValue({
      user: { id: 1, role: 'USER', department_id: 1 },
    });

    const req = new NextRequest('http://localhost:3000/api/issues', {
      method: 'POST',
      body: JSON.stringify({
        title: 'Test',
      }),
    });

    const res = await POST(req);
    expect(res.status).toBe(400);
  });

  it('MANAGER cannot assign to other department', async () => {
    (getServerSession as jest.Mock).mockResolvedValue({
      user: { id: 1, role: 'MANAGER', department_id: 2 },
    });

    // 다른 부서 직원 할당 시도
    const req = new NextRequest('http://localhost:3000/api/issues', {
      method: 'POST',
      body: JSON.stringify({
        customer_id: 1,
        title: 'Test',
        severity: 'HIGH',
        description: 'Long enough description',
        assigned_to_id: 99, // 다른 부서
      }),
    });

    const res = await POST(req);
    expect(res.status).toBe(400);
  });
});

### Hooks 테스트 예시

```typescript
// src/__tests__/hooks/useIssueListQuery.test.ts

import { renderHook, waitFor } from '@testing-library/react';
import { useIssueListQuery } from '@/hooks/issues';
import { QueryClient, QueryClientProvider } from '@tanstack/react-query';

const createWrapper = () => {
  const queryClient = new QueryClient({
    defaultOptions: {
      queries: { retry: false },
    },
  });
  return ({ children }: any) => (
    <QueryClientProvider client={queryClient}>{children}</QueryClientProvider>
  );
};

describe('useIssueListQuery', () => {
  it('should fetch issue list', async () => {
    const { result } = renderHook(
      () => useIssueListQuery({ page: 1, page_size: 20 }),
      { wrapper: createWrapper() }
    );

    await waitFor(() => {
      expect(result.current.isSuccess).toBe(true);
    });

    expect(result.current.data?.data).toBeDefined();
    expect(result.current.data?.pagination).toBeDefined();
  });

  it('should apply filters', async () => {
    const { result } = renderHook(
      () => useIssueListQuery({ page: 1, page_size: 20, customer_id: 5 }),
      { wrapper: createWrapper() }
    );

    await waitFor(() => {
      expect(result.current.isSuccess).toBe(true);
    });

    const issues = result.current.data?.data || [];
    expect(issues.every((i: any) => i.customer_id === 5)).toBe(true);
  });
});
```
