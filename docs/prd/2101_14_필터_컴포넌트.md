```typescript
// Generated: 2026-01-28 15:30:00 KST

'use client';

import { useState, useCallback } from 'react';
import { Input } from '@/components/ui/input';
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select';
import { Button } from '@/components/ui/button';
import { Tabs, TabsList, TabsTrigger } from '@/components/ui/tabs';
import { Search, X } from 'lucide-react';
import { useDebounce } from '@/hooks/useDebounce';
import { NOTICE_TYPES, type NoticeType, type NoticeSortBy } from '@/types/notice';

interface NoticeFiltersProps {
  type: NoticeType | 'all';
  search: string;
  sortBy: NoticeSortBy;
  onTypeChange: (type: NoticeType | 'all') => void;
  onSearchChange: (search: string) => void;
  onSortByChange: (sortBy: NoticeSortBy) => void;
  onReset: () => void;
}

export default function NoticeFilters({
  type,
  search,
  sortBy,
  onTypeChange,
  onSearchChange,
  onSortByChange,
  onReset,
}: NoticeFiltersProps) {
  // 검색 입력 로컬 상태 (디바운스용)
  const [searchInput, setSearchInput] = useState(search);

  // 디바운스 처리
  const debouncedSearch = useDebounce(
    useCallback((value: string) => {
      onSearchChange(value);
    }, [onSearchChange]),
    300
  );

  const handleSearchChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    const value = e.target.value;
    setSearchInput(value);
    debouncedSearch(value);
  };

  const handleClearSearch = () => {
    setSearchInput('');
    onSearchChange('');
  };

  const handleReset = () => {
    setSearchInput('');
    onReset();
  };

  // 필터가 적용되었는지 확인
  const hasActiveFilters = type !== 'all' || search !== '' || sortBy !== 'latest';

  return (
    <div className="space-y-4">
      {/* 유형 탭 */}
      <Tabs value={type} onValueChange={(value) => onTypeChange(value as NoticeType | 'all')}>
        <TabsList>
          <TabsTrigger value="all">전체</TabsTrigger>
          {NOTICE_TYPES.map((t) => (
            <TabsTrigger key={t} value={t}>
              {t}
            </TabsTrigger>
          ))}
        </TabsList>
      </Tabs>

      {/* 검색 및 정렬 */}
      <div className="flex flex-wrap gap-4">
        {/* 검색 */}
        <div className="relative flex-1 min-w-[200px] max-w-md">
          <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
          <Input
            placeholder="제목 또는 내용 검색..."
            value={searchInput}
            onChange={handleSearchChange}
            className="pl-9 pr-9"
          />
          {searchInput && (
            <button
              onClick={handleClearSearch}
              className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
            >
              <X className="h-4 w-4" />
            </button>
          )}
        </div>

        {/* 정렬 */}
        <Select value={sortBy} onValueChange={(value) => onSortByChange(value as NoticeSortBy)}>
          <SelectTrigger className="w-[140px]">
            <SelectValue placeholder="정렬" />
          </SelectTrigger>
          <SelectContent>
            <SelectItem value="latest">최신순</SelectItem>
            <SelectItem value="viewCount">조회순</SelectItem>
            <SelectItem value="commentCount">댓글순</SelectItem>
          </SelectContent>
        </Select>

        {/* 초기화 버튼 */}
        {hasActiveFilters && (
          <Button variant="ghost" onClick={handleReset}>
            <X className="h-4 w-4 mr-1" />
            필터 초기화
          </Button>
        )}
      </div>
    </div>
  );
}
```
