// Generated: 2026-01-28 15:30:00 KST

import { NextRequest, NextResponse } from 'next/server';
import { getServerSession } from 'next-auth';
import { authOptions } from '@/lib/auth';
import { executeQuery, executeUpdate, executeQuerySingle } from '@/lib/db-direct';
import type { Comment, CommentResponse } from '@/types/notice';

export const dynamic = 'force-dynamic';

interface RouteParams {
  params: { id: string; commentId: string };
}

// ============================================================
// PUT /api/notices/[id]/comments/[commentId] - 댓글 수정
// ============================================================

export async function PUT(req: NextRequest, { params }: RouteParams): Promise<NextResponse> {
  try {
    // 1. 인증 검증
    const session = await getServerSession(authOptions);
    if (!session?.user) {
      return NextResponse.json({ message: 'Unauthorized' }, { status: 401 });
    }

    const noticeId = parseInt(params.id, 10);
    const commentId = parseInt(params.commentId, 10);

    if (isNaN(noticeId) || isNaN(commentId)) {
      return NextResponse.json({ message: 'Invalid ID' }, { status: 400 });
    }

    const user = session.user as any;
    const userId = user.id;
    const userRole = user.role;

    // 2. 댓글 조회 및 권한 확인
    const commentResult = await executeQuerySingle<any>(
      `SELECT c.ID, c.NOTICE_ID, c.CONTENT, c.AUTHOR_ID, c.PARENT_COMMENT_ID,
              c."created_at" AS CREATED_AT, c."updated_at" AS UPDATED_AT,
              e.NAME AS AUTHOR_NAME
       FROM COMMENT c
       LEFT JOIN EMPLOYEE e ON e.ID = c.AUTHOR_ID
       WHERE c.ID = :commentId AND c.NOTICE_ID = :noticeId AND c."deleted_at" IS NULL`,
      { commentId, noticeId }
    );

    if (!commentResult) {
      return NextResponse.json({ message: 'Comment not found' }, { status: 404 });
    }

    // 자신의 댓글 또는 ADMIN만 수정 가능
    if (commentResult.AUTHOR_ID !== userId && userRole !== 'ADMIN') {
      return NextResponse.json({ message: 'Forbidden' }, { status: 403 });
    }

    // 3. 요청 본문 파싱
    const body = await req.json();
    const { content } = body;

    // 4. 필드 검증
    const errors: Record<string, string> = {};

    if (!content || content.length < 1 || content.length > 1000) {
      errors.content = '댓글은 1~1000자 사이여야 합니다';
    }

    if (Object.keys(errors).length > 0) {
      return NextResponse.json({ message: 'Validation failed', errors }, { status: 400 });
    }

    // 5. 댓글 수정
    await executeUpdate(
      `UPDATE COMMENT SET CONTENT = :content, "updated_at" = SYSTIMESTAMP
       WHERE ID = :commentId`,
      { content, commentId }
    );

    // 6. 수정된 댓글 조회
    const updatedResult = await executeQuerySingle<any>(
      `SELECT c.ID, c.NOTICE_ID, c.CONTENT, c.AUTHOR_ID, c.PARENT_COMMENT_ID,
              c."created_at" AS CREATED_AT, c."updated_at" AS UPDATED_AT,
              e.NAME AS AUTHOR_NAME,
              p.AUTHOR_ID AS PARENT_AUTHOR_ID, pe.NAME AS PARENT_AUTHOR_NAME
       FROM COMMENT c
       LEFT JOIN EMPLOYEE e ON e.ID = c.AUTHOR_ID
       LEFT JOIN COMMENT p ON p.ID = c.PARENT_COMMENT_ID
       LEFT JOIN EMPLOYEE pe ON pe.ID = p.AUTHOR_ID
       WHERE c.ID = :commentId`,
      { commentId }
    );

    // 7. 응답 반환
    const comment: Comment = {
      id: updatedResult.ID,
      noticeId: updatedResult.NOTICE_ID,
      content: updatedResult.CONTENT,
      authorId: updatedResult.AUTHOR_ID,
      authorName: updatedResult.AUTHOR_NAME || '',
      parentCommentId: updatedResult.PARENT_COMMENT_ID || undefined,
      parentAuthorName: updatedResult.PARENT_AUTHOR_NAME || undefined,
      isEdited: true, // 수정되었으므로 true
      createdAt: updatedResult.CREATED_AT?.toISOString() || '',
      updatedAt: updatedResult.UPDATED_AT?.toISOString() || '',
    };

    return NextResponse.json<CommentResponse>({
      message: '댓글이 수정되었습니다',
      data: comment,
    });
  } catch (error) {
    console.error('PUT /api/notices/[id]/comments/[commentId] error:', error);
    return NextResponse.json({ message: 'Internal server error' }, { status: 500 });
  }
}

// DELETE /api/notices/[id]/comments/[commentId] - 댓글 삭제
// ============================================================

export async function DELETE(req: NextRequest, { params }: RouteParams): Promise<NextResponse> {
  try {
    // 1. 인증 검증
    const session = await getServerSession(authOptions);
    if (!session?.user) {
      return NextResponse.json({ message: 'Unauthorized' }, { status: 401 });
    }

    const noticeId = parseInt(params.id, 10);
    const commentId = parseInt(params.commentId, 10);

    if (isNaN(noticeId) || isNaN(commentId)) {
      return NextResponse.json({ message: 'Invalid ID' }, { status: 400 });
    }

    const user = session.user as any;
    const userId = user.id;
    const userRole = user.role;

    // 2. 댓글 조회 및 권한 확인
    const commentResult = await executeQuerySingle<any>(
      `SELECT ID, AUTHOR_ID, PARENT_COMMENT_ID
       FROM COMMENT
       WHERE ID = :commentId AND NOTICE_ID = :noticeId AND "deleted_at" IS NULL`,
      { commentId, noticeId }
    );

    if (!commentResult) {
      return NextResponse.json({ message: 'Comment not found' }, { status: 404 });
    }

    // 자신의 댓글 또는 ADMIN만 삭제 가능
    if (commentResult.AUTHOR_ID !== userId && userRole !== 'ADMIN') {
      return NextResponse.json({ message: 'Forbidden' }, { status: 403 });
    }

    // 3. 댓글 삭제 (soft delete)
    // 만약 최상위 댓글이면 답글도 함께 삭제
    if (commentResult.PARENT_COMMENT_ID === null) {
      // 답글도 soft delete
      await executeUpdate(
        `UPDATE COMMENT SET "deleted_at" = SYSTIMESTAMP
         WHERE PARENT_COMMENT_ID = :commentId AND "deleted_at" IS NULL`,
        { commentId }
      );
    }

    // 댓글 soft delete
    await executeUpdate(
      `UPDATE COMMENT SET "deleted_at" = SYSTIMESTAMP WHERE ID = :commentId`,
      { commentId }
    );

    return NextResponse.json({ message: '댓글이 삭제되었습니다' });
  } catch (error) {
    console.error('DELETE /api/notices/[id]/comments/[commentId] error:', error);
    return NextResponse.json({ message: 'Internal server error' }, { status: 500 });
  }
}
