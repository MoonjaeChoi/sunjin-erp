## 구현 목표

장비의 상태를 고장 또는 폐기로 변경하는 API를 구현한다. 상태 전이 규칙을 검증하며, 상태 변경 사유(필수)를 기록한다.

**Endpoint**: `POST /api/inventory/[id]/status`

```typescript
{
  new_status: '고장' | '폐기';  // 필수
  reason: string;               // 필수, 상태 변경 사유 (1-500자)
}
```

## 구현 상세

```typescript
import { NextRequest, NextResponse } from 'next/server';
import { getServerSession } from 'next-auth/next';
import { authOptions } from '@/lib/auth';
import { getRepository } from '@/lib/database';
import { Inventory } from '@/entities/Inventory';
import { InventoryHistory } from '@/entities/InventoryHistory';
import { InventoryService } from '@/lib/inventory-service';
import { IsNull } from 'typeorm';

export async function POST(
  request: NextRequest,
  { params }: { params: { id: string } }
): Promise<NextResponse> {
  try {
    const session = await getServerSession(authOptions);
    if (!session?.user) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }

    if (!['MANAGER', 'ADMIN'].includes(session.user.role)) {
      return NextResponse.json(
        { error: 'Forbidden: Only MANAGER and ADMIN can change status' },
        { status: 403 }
      );
    }

    const inventoryId = parseInt(params.id);
    if (isNaN(inventoryId)) {
      return NextResponse.json({ error: 'Invalid ID' }, { status: 400 });
    }

    const body = await request.json();
    const { new_status, reason } = body;

    if (!new_status || !reason) {
      return NextResponse.json(
        { error: 'new_status and reason are required' },
        { status: 400 }
      );
    }

    if (!['고장', '폐기'].includes(new_status)) {
      return NextResponse.json(
        { error: 'new_status must be "고장" or "폐기"' },
        { status: 400 }
      );
    }

    if (typeof reason !== 'string' || reason.length < 1 || reason.length > 500) {
      return NextResponse.json(
        { error: 'reason must be 1-500 characters' },
        { status: 400 }
      );
    }

    const inventoryRepository = getRepository(Inventory);
    const inventory = await inventoryRepository.findOne({
      where: { id: inventoryId, deleted_at: IsNull() },
    });

    if (!inventory) {
      return NextResponse.json(
        { error: 'Inventory not found' },
        { status: 404 }
      );
    }

    // 상태 전이 규칙 검증
    const validation = InventoryService.validateStateTransition(
      inventory.current_status,
      new_status
    );

    if (!validation.valid) {
      return NextResponse.json(
        { error: validation.error },
        { status: 422 }
      );
    }

    // Inventory 상태 업데이트
    const previousStatus = inventory.current_status;
    inventory.current_status = new_status;
    inventory.updated_by_id = session.user.id;
    inventory.updated_at = new Date();

    const updatedInventory = await inventoryRepository.save(inventory);

    // 상태 변경 이력 생성
    const historyRepository = getRepository(InventoryHistory);
    const history = new InventoryHistory();
    history.inventory_id = inventoryId;
    history.change_type = '상태변경';
    history.previous_status = previousStatus;
    history.new_status = new_status;
    history.reason = reason;
    history.previous_location = null;
    history.new_location = null;
    history.checkout_location = null;
    history.expected_checkin_date = null;
    history.changed_by_id = session.user.id;

    const savedHistory = await historyRepository.save(history);

    return NextResponse.json(
      {
        id: updatedInventory.id,
        current_status: updatedInventory.current_status,
        updated_at: updatedInventory.updated_at.toISOString(),
        history: {
          id: savedHistory.id,
          change_type: savedHistory.change_type,
          previous_status: savedHistory.previous_status,
          new_status: savedHistory.new_status,
          reason: savedHistory.reason,
          changed_by_id: savedHistory.changed_by_id,
          changed_at: savedHistory.changed_at.toISOString(),
        },
      },
      { status: 200 }
    );
  } catch (error) {
    console.error('POST /api/inventory/[id]/status error:', error);
    return NextResponse.json(
      { error: 'Internal Server Error' },
      { status: 500 }
    );
  }
}
```
