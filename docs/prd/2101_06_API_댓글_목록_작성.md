// Generated: 2026-01-28 15:30:00 KST

import { NextRequest, NextResponse } from 'next/server';
import { getServerSession } from 'next-auth';
import { authOptions } from '@/lib/auth';
import { executeQuery, executeUpdate, executeQuerySingle } from '@/lib/db-direct';
import type { Comment, CommentListResponse } from '@/types/notice';

export const dynamic = 'force-dynamic';

interface RouteParams {
  params: { id: string };
}

// ============================================================
// GET /api/notices/[id]/comments - 댓글 목록 조회
// ============================================================

export async function GET(req: NextRequest, { params }: RouteParams): Promise<NextResponse> {
  try {
    // 1. 인증 검증
    const session = await getServerSession(authOptions);
    if (!session?.user) {
      return NextResponse.json({ message: 'Unauthorized' }, { status: 401 });
    }

    const noticeId = parseInt(params.id, 10);
    if (isNaN(noticeId)) {
      return NextResponse.json({ message: 'Invalid notice ID' }, { status: 400 });
    }

    // 2. 쿼리 파라미터
    const searchParams = req.nextUrl.searchParams;
    const page = Math.max(1, parseInt(searchParams.get('page') || '1', 10));
    const pageSize = Math.min(100, Math.max(1, parseInt(searchParams.get('pageSize') || '50', 10)));

    // 3. 게시물 존재 확인
    const noticeExists = await executeQuerySingle<any>(
      `SELECT ID FROM NOTICE WHERE ID = :noticeId AND "deleted_at" IS NULL`,
      { noticeId }
    );

    if (!noticeExists) {
      return NextResponse.json({ message: 'Notice not found' }, { status: 404 });
    }

    // 4. 댓글 조회 (oldest-first, PRD Decision #14)
    // 트리 구조로 반환 (부모 댓글 + 답글)
    const offset = (page - 1) * pageSize;

    // 4.1 최상위 댓글만 조회 (PARENT_COMMENT_ID IS NULL)
    const rootCommentsResult = await executeQuery<any>(
      `SELECT
        c.ID,
        c.NOTICE_ID,
        c.CONTENT,
        c.AUTHOR_ID,
        e.NAME AS AUTHOR_NAME,
        c.PARENT_COMMENT_ID,
        c."created_at" AS CREATED_AT,
        c."updated_at" AS UPDATED_AT
      FROM COMMENT c
      LEFT JOIN EMPLOYEE e ON e.ID = c.AUTHOR_ID
      WHERE c.NOTICE_ID = :noticeId
        AND c.PARENT_COMMENT_ID IS NULL
        AND c."deleted_at" IS NULL
      ORDER BY c."created_at" ASC
      OFFSET :offset ROWS FETCH NEXT :pageSize ROWS ONLY`,
      { noticeId, offset, pageSize }
    );

    // 4.2 전체 최상위 댓글 수
    const countResult = await executeQuery<any>(
      `SELECT COUNT(*) AS TOTAL FROM COMMENT
       WHERE NOTICE_ID = :noticeId AND PARENT_COMMENT_ID IS NULL AND "deleted_at" IS NULL`,
      { noticeId }
    );
    const total = parseInt(countResult.rows[0]?.TOTAL || '0', 10);

    // 4.3 각 최상위 댓글에 대한 답글 조회
    const rootCommentIds = rootCommentsResult.rows.map((c: any) => c.ID);

    let repliesMap: Map<number, any[]> = new Map();

    if (rootCommentIds.length > 0) {
      // IN 절을 위한 파라미터 구성
      const inParams: Record<string, any> = {};
      const placeholders = rootCommentIds.map((id: number, i: number) => {
        inParams[`id${i}`] = id;
        return `:id${i}`;
      }).join(',');

      const repliesResult = await executeQuery<any>(
        `SELECT
          c.ID,
          c.NOTICE_ID,
          c.CONTENT,
          c.AUTHOR_ID,
          e.NAME AS AUTHOR_NAME,
          c.PARENT_COMMENT_ID,
          p_author.NAME AS PARENT_AUTHOR_NAME,
          c."created_at" AS CREATED_AT,
          c."updated_at" AS UPDATED_AT
        FROM COMMENT c
        LEFT JOIN EMPLOYEE e ON e.ID = c.AUTHOR_ID
        LEFT JOIN COMMENT p ON p.ID = c.PARENT_COMMENT_ID
        LEFT JOIN EMPLOYEE p_author ON p_author.ID = p.AUTHOR_ID
        WHERE c.PARENT_COMMENT_ID IN (${placeholders})
          AND c."deleted_at" IS NULL
        ORDER BY c."created_at" ASC`,
        inParams
      );

      // 부모 댓글별 답글 그룹화
      repliesResult.rows.forEach((reply: any) => {
        const parentId = reply.PARENT_COMMENT_ID;
        if (!repliesMap.has(parentId)) {
          repliesMap.set(parentId, []);
        }
        repliesMap.get(parentId)!.push(reply);
      });
    }

    // 5. 응답 구성 (트리 구조)
    const comments: Comment[] = rootCommentsResult.rows.map((row: any) => {
      const replies = repliesMap.get(row.ID) || [];

      return {
        id: row.ID,
        noticeId: row.NOTICE_ID,
        content: row.CONTENT,
        authorId: row.AUTHOR_ID,
        authorName: row.AUTHOR_NAME || '',
        parentCommentId: undefined,
        isEdited: row.CREATED_AT?.getTime() !== row.UPDATED_AT?.getTime(),
        createdAt: row.CREATED_AT?.toISOString() || '',
        updatedAt: row.UPDATED_AT?.toISOString() || '',
        replies: replies.map((r: any) => ({
          id: r.ID,
          noticeId: r.NOTICE_ID,
          content: r.CONTENT,
          authorId: r.AUTHOR_ID,
          authorName: r.AUTHOR_NAME || '',
          parentCommentId: r.PARENT_COMMENT_ID,
          parentAuthorName: r.PARENT_AUTHOR_NAME || undefined,
          isEdited: r.CREATED_AT?.getTime() !== r.UPDATED_AT?.getTime(),
          createdAt: r.CREATED_AT?.toISOString() || '',
          updatedAt: r.UPDATED_AT?.toISOString() || '',
        })),
      };
    });

    return NextResponse.json<CommentListResponse>({
      data: comments,
      pagination: {
        page,
        pageSize,
        total,
        totalPages: Math.ceil(total / pageSize),
      },
    });
  } catch (error) {
    console.error('GET /api/notices/[id]/comments error:', error);
    return NextResponse.json({ message: 'Internal server error' }, { status: 500 });
  }
}

// ============================================================
// POST /api/notices/[id]/comments - 댓글/답글 작성
// ============================================================

export async function POST(req: NextRequest, { params }: RouteParams): Promise<NextResponse> {
  try {
    // 1. 인증 검증
    const session = await getServerSession(authOptions);
    if (!session?.user) {
      return NextResponse.json({ message: 'Unauthorized' }, { status: 401 });
    }

    const noticeId = parseInt(params.id, 10);
    if (isNaN(noticeId)) {
      return NextResponse.json({ message: 'Invalid notice ID' }, { status: 400 });
    }

    const user = session.user as any;
    const userId = user.id;

    // 2. 게시물 존재 확인
    const noticeExists = await executeQuerySingle<any>(
      `SELECT ID FROM NOTICE WHERE ID = :noticeId AND "deleted_at" IS NULL`,
      { noticeId }
    );

    if (!noticeExists) {
      return NextResponse.json({ message: 'Notice not found' }, { status: 404 });
    }

    // 3. 요청 본문 파싱
    const body = await req.json();
    const { content, parentCommentId } = body;

    // 4. 필드 검증
    const errors: Record<string, string> = {};

    if (!content || content.length < 1 || content.length > 1000) {
      errors.content = '댓글은 1~1000자 사이여야 합니다';
    }

    if (Object.keys(errors).length > 0) {
      return NextResponse.json({ message: 'Validation failed', errors }, { status: 400 });
    }

    // 5. 부모 댓글 검증 (답글인 경우)
    // PRD Decision #3: 최대 2단계 깊이
    let parentAuthorName: string | undefined;

    if (parentCommentId) {
      const parentComment = await executeQuerySingle<any>(
        `SELECT c.ID, c.PARENT_COMMENT_ID, c.AUTHOR_ID, e.NAME AS AUTHOR_NAME
         FROM COMMENT c
         LEFT JOIN EMPLOYEE e ON e.ID = c.AUTHOR_ID
         WHERE c.ID = :parentCommentId
           AND c.NOTICE_ID = :noticeId
           AND c."deleted_at" IS NULL`,
        { parentCommentId, noticeId }
      );

      if (!parentComment) {
        return NextResponse.json(
          { message: 'Parent comment not found', errors: { parentCommentId: '부모 댓글이 존재하지 않습니다' } },
          { status: 400 }
        );
      }

      // 깊이 검증: 부모 댓글이 이미 답글이면 (PARENT_COMMENT_ID가 있으면) 거부
      if (parentComment.PARENT_COMMENT_ID !== null) {
        return NextResponse.json(
          {
            message: 'Depth limit exceeded',
            errors: { parentCommentId: '답글에 대한 답글은 작성할 수 없습니다 (최대 2단계)' }
          },
          { status: 400 }
        );
      }

      parentAuthorName = parentComment.AUTHOR_NAME;
    }

    // 6. 댓글 INSERT
    const now = new Date();

    await executeUpdate(
      `INSERT INTO COMMENT (NOTICE_ID, CONTENT, AUTHOR_ID, PARENT_COMMENT_ID, "created_at", "updated_at")
       VALUES (:noticeId, :content, :authorId, :parentCommentId, :now, :now)`,
      {
        noticeId,
        content,
        authorId: userId,
        parentCommentId: parentCommentId || null,
        now,
      }
    );

    // 7. 생성된 댓글 조회
    const createdResult = await executeQuery<any>(
      `SELECT c.ID, c.NOTICE_ID, c.CONTENT, c.AUTHOR_ID, e.NAME AS AUTHOR_NAME,
              c.PARENT_COMMENT_ID, c."created_at" AS CREATED_AT, c."updated_at" AS UPDATED_AT
       FROM COMMENT c
       LEFT JOIN EMPLOYEE e ON e.ID = c.AUTHOR_ID
       WHERE c.AUTHOR_ID = :authorId AND c.NOTICE_ID = :noticeId
       ORDER BY c.ID DESC
       FETCH FIRST 1 ROWS ONLY`,
      { authorId: userId, noticeId }
    );

    const created = createdResult.rows[0];

    // 8. 응답 반환
    const comment: Comment = {
      id: created.ID,
      noticeId: created.NOTICE_ID,
      content: created.CONTENT,
      authorId: created.AUTHOR_ID,
      authorName: created.AUTHOR_NAME || '',
      parentCommentId: created.PARENT_COMMENT_ID || undefined,
      parentAuthorName,
      isEdited: false,
      createdAt: created.CREATED_AT?.toISOString() || '',
      updatedAt: created.UPDATED_AT?.toISOString() || '',
    };

    return NextResponse.json(
      {
        message: parentCommentId ? '답글이 등록되었습니다' : '댓글이 등록되었습니다',
        data: comment,
      },
      { status: 201 }
    );
  } catch (error) {
    console.error('POST /api/notices/[id]/comments error:', error);
    return NextResponse.json({ message: 'Internal server error' }, { status: 500 });
  }
}

## API 스펙

### GET /api/notices/[id]/comments

| 파라미터 | 타입 | 기본값 | 설명 |
|----------|------|--------|------|
| page | number | 1 | 페이지 번호 |
| pageSize | number | 50 | 페이지 크기 (최대 100) |

**응답**: 트리 구조 (최상위 댓글 + replies 배열)

### POST /api/notices/[id]/comments

| 필드 | 타입 | 필수 | 설명 |
|------|------|------|------|
| content | string | O | 댓글 내용 (1~1000자) |
| parentCommentId | number | X | 부모 댓글 ID (답글인 경우) |

**에러 케이스**:
- 400: 부모 댓글이 이미 답글인 경우 (깊이 제한)
- 404: 게시물 또는 부모 댓글 없음
