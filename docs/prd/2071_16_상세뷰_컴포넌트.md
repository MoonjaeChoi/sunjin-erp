```typescript
// Generated: 2026-01-26 HH:MM:SS KST

'use client';

import { Suspense } from 'react';
import { useMaintenanceContractDetail } from '@/hooks/useMaintenanceContractDetail';
import MaintenanceInfo from './MaintenanceInfo';
import MaintenanceAttachments from './MaintenanceAttachments';
import MaintenanceHistory from './MaintenanceHistory';
import MaintenanceDetailSkeleton from './MaintenanceDetailSkeleton';
import { Button } from '@/components/ui/button';
import { useRouter } from 'next/navigation';
import { ArrowLeft } from 'lucide-react';

interface MaintenanceDetailProps {
  contractId: number;
}

/**
 * 계약 상세뷰 페이지
 */
export default function MaintenanceDetail({ contractId }: MaintenanceDetailProps) {
  const router = useRouter();
  const { data, isLoading, error } = useMaintenanceContractDetail(contractId);

  if (isLoading) return <MaintenanceDetailSkeleton />;

  if (error || !data) {
    return (
      <div className="space-y-4">
        <Button
          variant="ghost"
          onClick={() => router.back()}
        >
          <ArrowLeft className="w-4 h-4 mr-2" />
          돌아가기
        </Button>
        <div className="text-center py-8 text-red-500">
          계약 정보를 불러올 수 없습니다
        </div>
      </div>
    );
  }

  return (
    <div className="space-y-6">
      {/* 헤더 */}
      <div className="flex items-center justify-between">
        <div>
          <Button
            variant="ghost"
            onClick={() => router.back()}
            className="mb-4"
          >
            <ArrowLeft className="w-4 h-4 mr-2" />
            돌아가기
          </Button>
          <h1 className="text-3xl font-bold">{data.contract_name}</h1>
          <p className="text-muted-foreground">
            {data.customer.name} · {data.assigned_employee.name}
          </p>
        </div>
      </div>

      {/* 기본 정보 */}
      <MaintenanceInfo contract={data} />

      {/* 첨부 파일 */}
      <Suspense fallback={<div className="h-20 bg-gray-200 rounded" />}>
        <MaintenanceAttachments contractId={contractId} />
      </Suspense>

      {/* 변경 이력 */}
      <Suspense fallback={<div className="h-20 bg-gray-200 rounded" />}>
        <MaintenanceHistory contractId={contractId} />
      </Suspense>
    </div>
  );
}
```

### 3. MaintenanceAttachments.tsx - 첨부 파일 섹션 (CC)

```typescript
// Generated: 2026-01-26 HH:MM:SS KST

'use client';

import { useQuery } from '@tanstack/react-query';
import { MaintenanceAttachment } from '@/types/maintenance';
import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
import { Button } from '@/components/ui/button';
import { FileText, Download, Trash2 } from 'lucide-react';
import { useState } from 'react';

interface MaintenanceAttachmentsProps {
  contractId: number;
}

/**
 * 첨부 파일 섹션
 */
export default function MaintenanceAttachments({
  contractId
}: MaintenanceAttachmentsProps) {
  const [selectedFile, setSelectedFile] = useState<File | null>(null);

  const { data: attachmentsData, isLoading } = useQuery({
    queryKey: ['maintenance', 'attachments', contractId],
    queryFn: async () => {
      const res = await fetch(`/api/maintenance/${contractId}/attachments`);
      return res.json();
    }
  });

  const attachments: MaintenanceAttachment[] = attachmentsData?.data || [];

  return (
    <Card>
      <CardHeader>
        <CardTitle>첨부 파일</CardTitle>
      </CardHeader>

      <CardContent className="space-y-4">
        {/* 파일 업로드 */}
        <div className="border-2 border-dashed rounded-lg p-6 text-center">
          <input
            type="file"
            id="file-upload"
            className="hidden"
            accept=".pdf,.doc,.docx"
            onChange={(e) => setSelectedFile(e.target.files?.[0] || null)}
            multiple
          />
          <label
            htmlFor="file-upload"
            className="cursor-pointer block"
          >
            <FileText className="w-8 h-8 mx-auto mb-2 text-muted-foreground" />
            <p className="text-sm font-medium">파일을 드래그하거나 클릭하여 업로드</p>
            <p className="text-xs text-muted-foreground">
              PDF, DOCX, DOC (최대 10MB, 최대 5개)
            </p>
          </label>
        </div>

        {/* 파일 목록 */}
        {isLoading ? (
          <p className="text-sm text-muted-foreground">로딩 중...</p>
        ) : attachments.length === 0 ? (
          <p className="text-sm text-muted-foreground">첨부 파일이 없습니다</p>
        ) : (
          <div className="space-y-2">
            {attachments.map((file) => (
              <div
                key={file.id}
                className="flex items-center justify-between p-3 border rounded-lg hover:bg-muted"
              >
                <div className="flex items-center gap-2 flex-1">
                  <FileText className="w-4 h-4 text-muted-foreground" />
                  <div className="flex-1 min-w-0">
                    <p className="text-sm font-medium truncate">
                      {file.file_name}
                    </p>
                    <p className="text-xs text-muted-foreground">
                      {(file.file_size / 1024).toFixed(1)} KB · {file.uploaded_by.name}
                    </p>
                  </div>
                </div>
                <div className="flex gap-2">
                  <Button
                    variant="ghost"
                    size="sm"
                    onClick={() => {
                      window.open(
                        `/api/maintenance/${contractId}/attachments/${file.id}/download`
                      );
                    }}
                  >
                    <Download className="w-4 h-4" />
                  </Button>
                  <Button
                    variant="ghost"
                    size="sm"
                    className="text-red-600 hover:text-red-700"
                    onClick={() => {
                      // TODO: delete mutation
                    }}
                  >
                    <Trash2 className="w-4 h-4" />
                  </Button>
                </div>
              </div>
            ))}
          </div>
        )}
      </CardContent>
    </Card>
  );
}
```

### 4. MaintenanceHistory.tsx - 변경 이력 타임라인 (CC)

```typescript
// Generated: 2026-01-26 HH:MM:SS KST

'use client';

import { useMaintenanceHistory } from '@/hooks/useMaintenanceStats';
import { MaintenanceContractHistory, ChangeTypeLabel } from '@/types/maintenance';
import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
import { Badge } from '@/components/ui/badge';

interface MaintenanceHistoryProps {
  contractId: number;
}

/**
 * 변경 이력 타임라인
 */
export default function MaintenanceHistory({
  contractId
}: MaintenanceHistoryProps) {
  const { data: historyData, isLoading } = useMaintenanceHistory(contractId);

  const histories: MaintenanceContractHistory[] = historyData?.data || [];

  return (
    <Card>
      <CardHeader>
        <CardTitle>변경 이력</CardTitle>
      </CardHeader>

      <CardContent>
        {isLoading ? (
          <p className="text-sm text-muted-foreground">로딩 중...</p>
        ) : histories.length === 0 ? (
          <p className="text-sm text-muted-foreground">변경 이력이 없습니다</p>
        ) : (
          <div className="space-y-4">
            {histories.map((history) => (
              <div key={history.id} className="flex gap-4">
                {/* 타임라인 선 */}
                <div className="flex flex-col items-center">
                  <div className="w-3 h-3 bg-blue-500 rounded-full mt-1.5" />
                  {histories[histories.length - 1].id !== history.id && (
                    <div className="w-0.5 h-16 bg-gray-200 mt-2" />
                  )}
                </div>

                {/* 이력 내용 */}
                <div className="flex-1 pb-4">
                  <div className="flex items-center gap-2 mb-1">
                    <Badge variant="secondary">
                      {ChangeTypeLabel[history.change_type]}
                    </Badge>
                    <span className="text-sm font-medium">
                      {history.changed_by.name}
                    </span>
                  </div>
                  <p className="text-sm text-muted-foreground">
                    {new Date(history.changed_at).toLocaleString('ko-KR')}
                  </p>
                  {history.reason && (
                    <p className="text-sm mt-2">{history.reason}</p>
                  )}
                </div>
              </div>
            ))}
          </div>
        )}
      </CardContent>
    </Card>
  );
}
```
