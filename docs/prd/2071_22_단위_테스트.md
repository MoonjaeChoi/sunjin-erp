### 4. API Route 테스트 예제

#### route.test.ts (GET /POST)

```typescript
import { GET, POST } from '@/app/api/maintenance/route';
import { getServerSession } from 'next-auth';
import { getDataSource } from '@/lib/db';

jest.mock('next-auth');
jest.mock('@/lib/db');

describe('GET /api/maintenance', () => {
  beforeEach(() => {
    jest.clearAllMocks();
    (getServerSession as jest.Mock).mockResolvedValue({
      user: { id: 1, name: 'test', role: 'USER' }
    });
  });

  it('should return list with pagination', async () => {
    const mockData = [
      {
        id: 1,
        contract_name: 'Test Contract',
        customer: { id: 1, name: 'ABC' }
      }
    ];

    (getDataSource as jest.Mock).mockResolvedValue({
      createQueryRunner: jest.fn(() => ({
        query: jest.fn()
          .mockResolvedValueOnce(mockData)
          .mockResolvedValueOnce([{ TOTAL: 1 }]),
        release: jest.fn()
      }))
    });

    const req = new Request('http://localhost/api/maintenance');
    const res = await GET(req);
    const data = await res.json();

    expect(res.status).toBe(200);
    expect(data.data).toHaveLength(1);
    expect(data.pagination.total).toBe(1);
  });

  it('should return 401 when not authenticated', async () => {
    (getServerSession as jest.Mock).mockResolvedValue(null);

    const req = new Request('http://localhost/api/maintenance');
    const res = await GET(req);

    expect(res.status).toBe(401);
  });
});

describe('POST /api/maintenance', () => {
  it('should create contract', async () => {
    (getServerSession as jest.Mock).mockResolvedValue({
      user: { id: 1, name: 'test', role: 'MANAGER' }
    });

    (getDataSource as jest.Mock).mockResolvedValue({
      createQueryRunner: jest.fn(() => ({
        startTransaction: jest.fn(),
        query: jest.fn()
          .mockResolvedValueOnce([{ id: 1 }])
          .mockResolvedValueOnce([]),
        commitTransaction: jest.fn(),
        release: jest.fn()
      }))
    });

    const body = {
      customer_id: 1,
      contract_name: 'New Contract',
      start_date: '2026-01-01',
      end_date: '2026-12-31',
      assigned_employee_id: 1
    };

    const req = new Request('http://localhost/api/maintenance', {
      method: 'POST',
      body: JSON.stringify(body)
    });

    const res = await POST(req);
    expect(res.status).toBe(201);
  });

  it('should reject invalid data', async () => {
    const body = { contract_name: 'Test' }; // Missing required fields

    const req = new Request('http://localhost/api/maintenance', {
      method: 'POST',
      body: JSON.stringify(body)
    });

    const res = await POST(req);
    expect(res.status).toBe(400);
  });
});
```

### 5. Component 테스트 예제

#### MaintenanceFilters.test.tsx

```typescript
import { render, screen, fireEvent } from '@testing-library/react';
import userEvent from '@testing-library/user-event';
import MaintenanceFilters from '@/components/features/maintenance/MaintenanceFilters';

describe('MaintenanceFilters', () => {
  const mockOnFilterChange = jest.fn();
  const mockOnReset = jest.fn();

  it('should render filter inputs', () => {
    render(
      <MaintenanceFilters
        filters={{}}
        onFilterChange={mockOnFilterChange}
        onReset={mockOnReset}
      />
    );

    expect(screen.getByPlaceholderText('계약명으로 검색...')).toBeInTheDocument();
    expect(screen.getByText('상태 선택')).toBeInTheDocument();
  });

  it('should call onFilterChange when search input changes', async () => {
    const user = userEvent.setup();
    render(
      <MaintenanceFilters
        filters={{}}
        onFilterChange={mockOnFilterChange}
        onReset={mockOnReset}
      />
    );

    const input = screen.getByPlaceholderText('계약명으로 검색...');
    await user.type(input, 'test');

    expect(mockOnFilterChange).toHaveBeenCalledWith({
      contractNameSearch: 'test',
      page: 1
    });
  });

  it('should call onReset when reset button is clicked', async () => {
    const user = userEvent.setup();
    render(
      <MaintenanceFilters
        filters={{ status: '활성' }}
        onFilterChange={mockOnFilterChange}
        onReset={mockOnReset}
      />
    );

    const resetBtn = screen.getByText('필터 초기화');
    await user.click(resetBtn);

    expect(mockOnReset).toHaveBeenCalled();
  });
});
```

### 6. Hooks 테스트 예제

#### useMaintenanceContracts.test.ts

```typescript
import { renderHook, waitFor } from '@testing-library/react';
import { QueryClient, QueryClientProvider } from '@tanstack/react-query';
import { useMaintenanceContracts } from '@/hooks/useMaintenanceContracts';

describe('useMaintenanceContracts', () => {
  const wrapper = ({ children }) => (
    <QueryClientProvider client={new QueryClient()}>
      {children}
    </QueryClientProvider>
  );

  it('should fetch contracts with default pagination', async () => {
    const { result } = renderHook(
      () => useMaintenanceContracts({}),
      { wrapper }
    );

    await waitFor(() => {
      expect(result.current.isSuccess).toBe(true);
    });

    expect(result.current.data?.data).toBeDefined();
    expect(result.current.data?.pagination).toBeDefined();
  });

  it('should apply filters to query', async () => {
    const { result } = renderHook(
      () => useMaintenanceContracts({ status: '활성' }),
      { wrapper }
    );

    await waitFor(() => {
      expect(result.current.isSuccess).toBe(true);
    });

    // 필터가 쿼리 키에 포함되는지 확인
    expect(result.current.queryKey).toEqual(['maintenance', 'list', { status: '활성' }]);
  });
});
```
