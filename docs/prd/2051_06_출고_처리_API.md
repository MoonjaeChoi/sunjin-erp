## 구현 목표

재고 상태인 장비를 출고 처리하는 API를 구현한다. 상태 전이 규칙 검증(재고→출고), 출고 위치 및 예상 반납일 기록, 이력 자동 생성을 포함한다.

**Endpoint**: `POST /api/inventory/[id]/checkout`

### 상태 전이 검증 (InventoryService)

```typescript
// src/lib/inventory-service.ts
export class InventoryService {
  // 상태 전이 규칙 검증
  static validateStateTransition(
    currentStatus: string,
    newStatus: string
  ): { valid: boolean; error?: string } {
    // 재고 → 출고 (O)
    if (currentStatus === '재고' && newStatus === '출고') {
      return { valid: true };
    }
    // 출고 → 재고 (반납)
    if (currentStatus === '출고' && newStatus === '재고') {
      return { valid: true };
    }
    // 재고 → 고장 (O)
    if (currentStatus === '재고' && newStatus === '고장') {
      return { valid: true };
    }
    // 출고 → 고장 (O)
    if (currentStatus === '출고' && newStatus === '고장') {
      return { valid: true };
    }
    // 고장 → 폐기 (O, 유일한 경로)
    if (currentStatus === '고장' && newStatus === '폐기') {
      return { valid: true };
    }
    // 폐기 → 변경 불가 (X)
    if (currentStatus === '폐기') {
      return { valid: false, error: '폐기된 장비는 상태 변경 불가' };
    }
    // 출고 → 폐기 (X, 반납 후에만 가능)
    if (currentStatus === '출고' && newStatus === '폐기') {
      return {
        valid: false,
        error: '출고 상태에서는 폐기로 변경 불가. 반납 후 진행하세요.',
      };
    }

    return { valid: false, error: '허용되지 않는 상태 전이입니다' };
  }
}
```

### POST Handler

```typescript
import { NextRequest, NextResponse } from 'next/server';
import { getServerSession } from 'next-auth/next';
import { authOptions } from '@/lib/auth';
import { getRepository } from '@/lib/database';
import { Inventory } from '@/entities/Inventory';
import { InventoryHistory } from '@/entities/InventoryHistory';
import { InventoryService } from '@/lib/inventory-service';
import { IsNull } from 'typeorm';

export async function POST(
  request: NextRequest,
  { params }: { params: { id: string } }
): Promise<NextResponse> {
  try {
    // 1. 인증 및 권한 검증
    const session = await getServerSession(authOptions);
    if (!session?.user) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }

    if (!['MANAGER', 'ADMIN'].includes(session.user.role)) {
      return NextResponse.json(
        { error: 'Forbidden: Only MANAGER and ADMIN can checkout inventory' },
        { status: 403 }
      );
    }

    const inventoryId = parseInt(params.id);
    if (isNaN(inventoryId)) {
      return NextResponse.json({ error: 'Invalid ID' }, { status: 400 });
    }

    // 2. Request Body 파싱
    const body = await request.json();
    const {
      checkout_location,
      checked_out_by,
      expected_checkin_date,
    } = body;

    // 필수 필드 검증
    if (!checkout_location || !checked_out_by) {
      return NextResponse.json(
        { error: 'checkout_location and checked_out_by are required' },
        { status: 400 }
      );
    }

    // expected_checkin_date 검증 (미래일만)
    if (expected_checkin_date) {
      const expectedDate = new Date(expected_checkin_date);
      const today = new Date();
      today.setHours(0, 0, 0, 0);
      if (expectedDate < today) {
        return NextResponse.json(
          { error: 'expected_checkin_date must be in the future' },
          { status: 400 }
        );
      }
    }

    // 3. Inventory 조회
    const inventoryRepository = getRepository(Inventory);
    const inventory = await inventoryRepository.findOne({
      where: { id: inventoryId, deleted_at: IsNull() },
    });

    if (!inventory) {
      return NextResponse.json(
        { error: 'Inventory not found' },
        { status: 404 }
      );
    }

    // 4. 상태 전이 검증 (재고 → 출고)
    const validation = InventoryService.validateStateTransition(
      inventory.current_status,
      '출고'
    );

    if (!validation.valid) {
      return NextResponse.json(
        { error: validation.error },
        { status: 422 } // Unprocessable Entity
      );
    }

    // 5. Inventory 상태 업데이트
    inventory.current_status = '출고';
    inventory.updated_by_id = session.user.id;
    inventory.updated_at = new Date();

    const updatedInventory = await inventoryRepository.save(inventory);

    // 6. 출고 이력 생성
    const historyRepository = getRepository(InventoryHistory);
    const history = new InventoryHistory();
    history.inventory_id = inventoryId;
    history.change_type = '출고';
    history.previous_status = '재고';
    history.new_status = '출고';
    history.checkout_location = checkout_location;
    history.expected_checkin_date = expected_checkin_date
      ? new Date(expected_checkin_date)
      : null;
    history.reason = null;
    history.previous_location = null;
    history.new_location = null;
    history.changed_by_id = session.user.id;

    const savedHistory = await historyRepository.save(history);

    // 7. 응답
    return NextResponse.json(
      {
        id: updatedInventory.id,
        current_status: updatedInventory.current_status,
        current_location: updatedInventory.current_location,
        updated_at: updatedInventory.updated_at.toISOString(),
        history: {
          id: savedHistory.id,
          change_type: savedHistory.change_type,
          previous_status: savedHistory.previous_status,
          new_status: savedHistory.new_status,
          checkout_location: savedHistory.checkout_location,
          expected_checkin_date: savedHistory.expected_checkin_date
            ? savedHistory.expected_checkin_date.toISOString().split('T')[0]
            : null,
          changed_by_id: savedHistory.changed_by_id,
          changed_at: savedHistory.changed_at.toISOString(),
        },
      },
      { status: 200 }
    );
  } catch (error) {
    console.error('POST /api/inventory/[id]/checkout error:', error);
    return NextResponse.json(
      { error: 'Internal Server Error' },
      { status: 500 }
    );
  }
}
```
