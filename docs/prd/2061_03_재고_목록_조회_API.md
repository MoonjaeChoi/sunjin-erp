### Query Parameters

```
- page: number (기본값 1, 최소값 1)
- pageSize: number (기본값 20, 최소값 1, 최대값 100)
- categories: string[] (쉼표 구분 또는 반복)
  예: ?categories=monitor,keyboard 또는 ?categories=monitor&categories=keyboard
- status: string[] (쉼표 구분, 값: 재고, 출고, 고장, 폐기)
  예: ?status=재고,출고
- location: string (자유 텍스트 검색, 선택)
- search: string (시리얼번호 또는 모델명 접두사 검색, 선택)
  → Phase 1: 접두사 기반만 지원 (LIKE 'term%')
- sortBy: 'category' | 'model' | 'serialNumber' | 'location' | 'status' | 'purchaseDate'
- order: 'asc' | 'desc' (기본값 'asc')
```

### 구현 상세: GET Handler

```typescript
import { NextRequest, NextResponse } from 'next/server';
import { getServerSession } from 'next-auth/next';
import { authOptions } from '@/lib/auth';
import { getRepository } from '@/lib/database';
import { Inventory } from '@/entities/Inventory';
import { IsNull } from 'typeorm';

export async function GET(request: NextRequest): Promise<NextResponse> {
  try {
    // 1. 인증 검증
    const session = await getServerSession(authOptions);
    if (!session?.user) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }

    // 2. Query parameters 파싱
    const searchParams = request.nextUrl.searchParams;
    const page = Math.max(1, parseInt(searchParams.get('page') || '1'));
    const pageSize = Math.min(
      100,
      Math.max(1, parseInt(searchParams.get('pageSize') || '20'))
    );

    // pageSize 검증
    if (pageSize > 100) {
      return NextResponse.json(
        { error: 'Page size cannot exceed 100' },
        { status: 400 }
      );
    }

    // categories 파싱 (쉼표 구분 또는 반복)
    const categoriesParam = searchParams.getAll('categories');
    const categories = categoriesParam.flatMap((c) => c.split(',').map((s) => s.trim()));

    // status 파싱 (쉼표 구분 또는 반복)
    const statusParam = searchParams.getAll('status');
    const statuses = statusParam.flatMap((s) => s.split(',').map((st) => st.trim()));

    // 자유 텍스트 검색 파라미터
    const location = searchParams.get('location') || '';
    const search = searchParams.get('search') || '';

    // 정렬 파라미터
    const sortBy = searchParams.get('sortBy') || 'category';
    const order = (searchParams.get('order') || 'asc').toUpperCase() as 'ASC' | 'DESC';

    // 3. 데이터베이스 쿼리 빌드
    const repository = getRepository(Inventory);
    let query = repository
      .createQueryBuilder('i')
      .where('i.deletedAt IS NULL'); // soft delete 필터

    // 카테고리 필터
    if (categories.length > 0) {
      query = query.andWhere('i.category IN (:categories)', { categories });
    }

    // 상태 필터
    if (statuses.length > 0) {
      query = query.andWhere('i.current_status IN (:statuses)', { statuses });
    }

    // 위치 검색
    if (location) {
      query = query.andWhere('LOWER(i.current_location) LIKE LOWER(:location)', {
        location: `%${location}%`,
      });
    }

    // 시리얼번호 또는 모델명 검색 (접두사 기반)
    if (search) {
      query = query.andWhere(
        '(LOWER(i.serial_number) LIKE LOWER(:search) OR LOWER(i.model) LIKE LOWER(:search))',
        { search: `${search}%` }
      );
    }

    // 정렬 매핑
    const sortMap: Record<string, string> = {
      category: 'i.category',
      model: 'i.model',
      serialNumber: 'i.serial_number',
      location: 'i.current_location',
      status: 'i.current_status',
      purchaseDate: 'i.purchase_date',
    };

    const sortColumn = sortMap[sortBy] || 'i.category';

    // 기본 정렬: category > model (asc)
    query = query
      .orderBy(sortColumn, order)
      .addOrderBy('i.model', 'ASC')
      .addOrderBy('i.id', 'ASC');

    // 4. 페이지네이션 계산
    const skip = (page - 1) * pageSize;
    const [data, total] = await query
      .skip(skip)
      .take(pageSize)
      .getManyAndCount();

    const totalPages = Math.ceil(total / pageSize);

    // 5. HATEOAS Links 생성
    const baseUrl = new URL(request.url);
    baseUrl.searchParams.set('page', String(page));
    baseUrl.searchParams.set('pageSize', String(pageSize));

    const links: Record<string, string> = {};

    if (page > 1) {
      const prevUrl = new URL(baseUrl);
      prevUrl.searchParams.set('page', String(page - 1));
      links.prev = prevUrl.toString();

      const firstUrl = new URL(baseUrl);
      firstUrl.searchParams.set('page', '1');
      links.first = firstUrl.toString();
    }

    if (page < totalPages) {
      const nextUrl = new URL(baseUrl);
      nextUrl.searchParams.set('page', String(page + 1));
      links.next = nextUrl.toString();

      const lastUrl = new URL(baseUrl);
      lastUrl.searchParams.set('page', String(totalPages));
      links.last = lastUrl.toString();
    }

    // 6. 응답 포맷
    return NextResponse.json(
      {
        data: data.map((item) => ({
          id: item.id,
          category: item.category,
          model: item.model,
          serial_number: item.serial_number,
          purchase_date: item.purchase_date.toISOString().split('T')[0],
          purchase_from: item.purchase_from,
          current_location: item.current_location,
          current_status: item.current_status,
          created_at: item.created_at.toISOString(),
          updated_at: item.updated_at.toISOString(),
        })),
        pagination: {
          page,
          pageSize,
          total,
          totalPages,
        },
        _links: links,
      },
      { status: 200 }
    );
  } catch (error) {
    console.error('GET /api/inventory error:', error);
    return NextResponse.json(
      { error: 'Internal Server Error' },
      { status: 500 }
    );
  }
}
```
