        await queryRunner.release();

      }
    });

    it('should hide soft-deleted contracts in queries', async () => {
      const queryRunner = dataSource.createQueryRunner();
      try {
        // Create and delete
        await queryRunner.query(
          `INSERT INTO MAINTENANCE_CONTRACT (customer_id, contract_name, ...)
          VALUES (1, 'Test', ...)`
        );
        await queryRunner.query(
          `UPDATE MAINTENANCE_CONTRACT SET deleted_at = CURRENT_TIMESTAMP WHERE id = 1`
        );

        // Query without soft delete filter should not return
        const result = await queryRunner.query(
          `SELECT * FROM MAINTENANCE_CONTRACT WHERE deleted_at IS NULL AND id = 1`
        );
        expect(result).toHaveLength(0);
      } finally {
        await queryRunner.release();
      }
    });
  });
