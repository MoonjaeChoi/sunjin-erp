  it('should hide soft-deleted contracts in queries', async () => {
    const dataSource = await getDataSource();

    // Create test contract
    const contract = await dataSource.query(`
      INSERT INTO MAINTENANCE_CONTRACT
      (customer_id, contract_name, contract_type, start_date, end_date, assigned_employee_id, contract_status, created_by_id, updated_by_id)
      VALUES (1, 'Test Contract', 'Maintenance', TRUNC(SYSDATE), TRUNC(SYSDATE)+365, 1, '활성', 1, 1)
      RETURNING id INTO :id
    `);

    const contractId = contract[0];

    // Soft delete
    await dataSource.query(`
      UPDATE MAINTENANCE_CONTRACT SET deleted_at = CURRENT_TIMESTAMP WHERE id = :id
    `, [contractId]);

    // Query should not return deleted contract
    const result = await dataSource.query(`
      SELECT * FROM MAINTENANCE_CONTRACT WHERE deleted_at IS NULL AND id = :id
