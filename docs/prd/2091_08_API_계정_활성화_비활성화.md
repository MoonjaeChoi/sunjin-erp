#### 1. POST /api/employees/[id]/deactivate - 계정 비활성화

```typescript
// src/app/api/employees/[id]/deactivate/route.ts

import { NextRequest, NextResponse } from 'next/server';
import { getServerSession } from 'next-auth';
import { authOptions } from '@/lib/auth';
import { executeQuery, executeUpdate } from '@/lib/db-direct';

export async function POST(
  req: NextRequest,
  { params }: { params: Promise<{ id: string }> }
): Promise<NextResponse> {
  try {
    const { id } = await params;
    const employeeId = parseInt(id, 10);

    if (isNaN(employeeId)) {
      return NextResponse.json({ message: 'Invalid employee ID' }, { status: 400 });
    }

    // 1. 인증/권한 검증
    const session = await getServerSession(authOptions);
    if (!session?.user) {
      return NextResponse.json({ message: 'Unauthorized' }, { status: 401 });
    }

    const user = session.user as any;
    if (user.role !== 'ADMIN') {
      return NextResponse.json({ message: 'Forbidden' }, { status: 403 });
    }

    // 2. 자기 자신 비활성화 방지
    if (user.id === employeeId) {
      return NextResponse.json(
        { message: '자기 자신의 계정을 비활성화할 수 없습니다.' },
        { status: 400 }
      );
    }

    // 3. 직원 및 계정 확인
    const existing = await executeQuery<{
      EMPID: number;
      EMPNAME: string;
      ACCID: number;
      ISACTIVE: number;
    }>(
      `SELECT
        e."id" as EMPID, e."name" as EMPNAME,
        a."id" as ACCID, a.IS_ACTIVE as ISACTIVE
       FROM EMPLOYEE e
       LEFT JOIN ACCOUNT a ON a.EMPLOYEE_ID = e."id" AND a."deleted_at" IS NULL
       WHERE e."id" = :employeeId AND e."deleted_at" IS NULL`,
      { employeeId }
    );

    if ((existing.rows || []).length === 0) {
      return NextResponse.json({ message: '직원을 찾을 수 없습니다.' }, { status: 404 });
    }

    const emp = existing.rows[0];

    if (!emp.ACCID) {
      return NextResponse.json({ message: '계정이 존재하지 않습니다.' }, { status: 400 });
    }

    if (emp.ISACTIVE === 0) {
      return NextResponse.json({ message: '이미 비활성화된 계정입니다.' }, { status: 400 });
    }

    // 4. 계정 비활성화
    const now = new Date();
    await executeUpdate(
      `UPDATE ACCOUNT SET IS_ACTIVE = 0, "updated_at" = :now WHERE "id" = :accountId`,
      { now, accountId: emp.ACCID }
    );

    // 5. 이력 기록
    await executeUpdate(
      `INSERT INTO EMPLOYEE_HISTORY (
        EMPLOYEE_ID, CHANGE_TYPE, CHANGED_FIELDS, CHANGED_BY_ID, CHANGED_AT
      ) VALUES (
        :employeeId, :changeType, :changedFields, :changedById, :changedAt
      )`,
      {
        employeeId,
        changeType: 'DEACTIVATE',
        changedFields: JSON.stringify({
          isActive: { before: true, after: false },
        }),
        changedById: user.id,
        changedAt: now,
      }
    );

    return NextResponse.json({
      message: `'${emp.EMPNAME}' 직원의 계정이 비활성화되었습니다. 더 이상 로그인할 수 없습니다.`,
      data: {
        id: emp.ACCID,
        isActive: false,
      },
    });

  } catch (error) {
    console.error('POST /api/employees/[id]/deactivate error:', error);
    return NextResponse.json(
      { message: 'Internal server error' },
      { status: 500 }
    );
  }
}
```
