### 구현 상세: GET Handler

```typescript
import { NextRequest, NextResponse } from 'next/server';
import { getServerSession } from 'next-auth/next';
import { authOptions } from '@/lib/auth';
import { getRepository } from '@/lib/database';
import { Inventory } from '@/entities/Inventory';
import { InventoryHistory } from '@/entities/InventoryHistory';
import { IsNull, LessThan } from 'typeorm';

export async function GET(
  request: NextRequest,
  { params }: { params: { id: string } }
): Promise<NextResponse> {
  try {
    // 1. 인증 검증
    const session = await getServerSession(authOptions);
    if (!session?.user) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }

    const inventoryId = parseInt(params.id);
    if (isNaN(inventoryId)) {
      return NextResponse.json({ error: 'Invalid ID' }, { status: 400 });
    }

    // 2. 재고 조회 (soft delete 포함 또는 미포함)
    const inventoryRepository = getRepository(Inventory);
    let inventory: Inventory | null = null;

    // ADMIN만 삭제된 기록 조회 가능
    if (session.user.role === 'ADMIN') {
      inventory = await inventoryRepository.findOne({
        where: { id: inventoryId },
        relations: ['created_by', 'updated_by'],
      });
    } else {
      inventory = await inventoryRepository.findOne({
        where: { id: inventoryId, deleted_at: IsNull() },
        relations: ['created_by', 'updated_by'],
      });
    }

    if (!inventory) {
      return NextResponse.json(
        { error: 'Inventory not found' },
        { status: 404 }
      );
    }

    // 3. 이력 조회
    const historyRepository = getRepository(InventoryHistory);
    const histories = await historyRepository
      .createQueryBuilder('h')
      .where('h.inventory_id = :inventoryId', { inventoryId })
      .orderBy('h.changed_at', 'ASC')
      .leftJoinAndSelect('h.changed_by', 'employee')
      .getMany();

    // 4. 과기 판정 계산
    let is_overdue = false;
    let overdue_days: number | null = null;

    if (inventory.current_status === '출고') {
      // 최근 출고 이력에서 expected_checkin_date 찾기
      const checkoutHistories = histories
        .filter((h) => h.change_type === '출고')
        .sort((a, b) => b.changed_at.getTime() - a.changed_at.getTime());

      if (
        checkoutHistories.length > 0 &&
        checkoutHistories[0].expected_checkin_date
      ) {
        const expectedDate = new Date(checkoutHistories[0].expected_checkin_date);
        const today = new Date();
        today.setHours(0, 0, 0, 0);

        if (expectedDate < today) {
          is_overdue = true;
          const diffTime = today.getTime() - expectedDate.getTime();
          overdue_days = Math.floor(diffTime / (1000 * 60 * 60 * 24));
        }
      }
    }

    // 5. 응답 포맷
    return NextResponse.json(
      {
        id: inventory.id,
        category: inventory.category,
        model: inventory.model,
        serial_number: inventory.serial_number,
        purchase_date: inventory.purchase_date.toISOString().split('T')[0],
        purchase_from: inventory.purchase_from,
        current_location: inventory.current_location,
        current_status: inventory.current_status,
        notes: inventory.notes,
        created_by: {
          id: inventory.created_by.id,
          name: inventory.created_by.name,
        },
        updated_by: {
          id: inventory.updated_by.id,
          name: inventory.updated_by.name,
        },
        created_at: inventory.created_at.toISOString(),
        updated_at: inventory.updated_at.toISOString(),
        overdue_days,
        is_overdue,
        histories: histories.map((h) => ({
          id: h.id,
          change_type: h.change_type,
          previous_location: h.previous_location,
          new_location: h.new_location,
          previous_status: h.previous_status,
          new_status: h.new_status,
          checkout_location: h.checkout_location,
          expected_checkin_date: h.expected_checkin_date
            ? h.expected_checkin_date.toISOString().split('T')[0]
            : null,
          reason: h.reason,
          changed_by: {
            id: h.changed_by.id,
            name: h.changed_by.name,
          },
          changed_at: h.changed_at.toISOString(),
        })),
      },
      { status: 200 }
    );
  } catch (error) {
    console.error('GET /api/inventory/[id] error:', error);
    return NextResponse.json(
      { error: 'Internal Server Error' },
      { status: 500 }
    );
  }
}
```
